/-- Summability of the correction for the (1 - p^-s)^(-z) family, p >= 5. -/
theorem cpow_neg_sub_one_summable (s : RR) (hs : 1 LEQ s) (z : CC) (hz : Norm.norm z LEQ 2) :
    Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      Norm.norm ((((1 - ((p:NN):RR) ^ (-s) : RR) : CC)) ^ (-z) - 1)) := by
  have hcomp : Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} => 8 * (((p:NN):RR) ^ (-s))) := by
    have hbase : Summable (fun n : NN => ((n:RR) ^ (-s))) := by
      rw [Real.summable_nat_rpow]
      linarith
    have hsub : Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} => (((p:NN):RR) ^ (-s))) :=
      hbase.subtype _
    exact hsub.mul_left 8
  apply Summable.of_nonneg_of_le
    (fun p => norm_nonneg _)
    (fun p => cpow_neg_sub_one_bound (p:NN) p.property.2 s hs z hz)
    hcomp
