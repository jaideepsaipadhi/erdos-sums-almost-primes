/-- D1: at z = 2, s = 1 the regularised factor is the cast of the real one. -/
theorem reg_factor_at_two (y : RR) (hy : (1:RR) - 2 * y NEQ 0) :
    (1 - (2:CC) * ((y : RR) : CC))INV * (((1 - y : RR)) : CC) ^ (2:CC)
      = ((((1 - 2*y : RR))INV * ((1 - y : RR)) ^ 2 : RR) : CC) := by
  have hcast2 : ((1 - 2*y : RR) : CC) = 1 - (2:CC) * ((y : RR) : CC) := by
    push_cast; ring
  have hne : (1:CC) - (2:CC) * ((y : RR) : CC) NEQ 0 := by
    rw [LARR hcast2]
    simp only [ne_eq, Complex.ofReal_eq_zero]
    exact hy
  have hpow : (((1 - y : RR)) : CC) ^ (2:CC) = (((1 - y : RR)) : CC) ^ (2:NN) := by
    rw [LARR Complex.cpow_natCast]
    norm_num
  rw [hpow]
  push_cast
  rw [LARR hcast2]
  push_cast
  field_simp
