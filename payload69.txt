/-- MANUSCRIPT THEOREM 4.1, UPPER BOUND: delta_m <= a log a + gamma a - a + 1
- 1 - (log2/4) m, where a = m log 2. -/
theorem deltaM_upper_bound (m : <<N>>) (hm : 1 <<LE>> m) :
    deltaM m <<LE>> -1 - (Real.log 2 / 4) * (m : <<R>>)
      + (((m:<<N>>):<<R>>) * Real.log 2) * Real.log (((m:<<N>>):<<R>>) * Real.log 2)
      + Real.eulerMascheroniConstant * (((m:<<N>>):<<R>>) * Real.log 2)
      - ((m:<<N>>):<<R>>) * Real.log 2 + 1 := by
  have hl2 : (0:<<R>>) < Real.log 2 := Real.log_pos (by norm_num)
  have hmpos : (0:<<R>>) < ((m:<<N>>):<<R>>) := by
    have : (1:<<R>>) <<LE>> ((m:<<N>>):<<R>>) := by exact_mod_cast hm
    linarith
  have hapos : (0:<<R>>) < ((m:<<N>>):<<R>>) * Real.log 2 := by positivity
  have hIooIoc : (<<INT>> t in Set.Ioo (0:<<R>>) 1,
      (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2 * t)) - 1
        + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2)
      = <<INT>> t in Set.Ioc (0:<<R>>) 1,
        (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2) * t) - 1
          + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2 := by
    rw [MeasureTheory.integral_Ioc_eq_integral_Ioo]
    apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioo
    intro t _
    simp only
    rw [show -(((m:<<N>>):<<R>>) * Real.log 2) * t = -(((m:<<N>>):<<R>>) * Real.log 2 * t) by ring]
  have hgap := BanksMartin.gap_five_evaluation (((m:<<N>>):<<R>>) * Real.log 2) hapos
  have hchain := sum_betaCirc_le_uniform m
  have hchain2 := sum_betaCirc_le_integral m
  have hsubst := exp_Bfun_integral_eq m
  rw [deltaM_identity m]
  have hfinal : <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i (m - i)
      <<LE>> (((m:<<N>>):<<R>>) * Real.log 2) * Real.log (((m:<<N>>):<<R>>) * Real.log 2)
        + Real.eulerMascheroniConstant * (((m:<<N>>):<<R>>) * Real.log 2)
        - ((m:<<N>>):<<R>>) * Real.log 2 + 1 := by
    calc <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i (m - i)
        <<LE>> <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i m := hchain
      _ <<LE>> <<INT>> u in Set.Ioi (0:<<R>>), Real.exp u * Bfun m u := hchain2
      _ = <<INT>> t in Set.Ioo (0:<<R>>) 1,
          (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2 * t)) - 1
            + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2 := hsubst
      _ = <<INT>> t in Set.Ioc (0:<<R>>) 1,
          (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2) * t) - 1
            + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2 := hIooIoc
      _ <<LE>> _ := hgap
  linarith [hfinal]
