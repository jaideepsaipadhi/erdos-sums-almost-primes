theorem log_tprod_real {IDX : Type*} {f : IDX ARROW RR} (hpos : FORALL i, 0 < f i)
    (hmul : Multipliable f) (hlog : Summable (fun i => Real.log (f i))) :
    Real.log (TPRODN i : IDX, f i) = TSUMN i : IDX, Real.log (f i) := by
  have hpart : FORALL t : Finset IDX, PRODF i IN t, f i
      = Real.exp (SUMF i IN t, Real.log (f i)) := by
    intro t
    rw [Real.exp_sum]
    apply Finset.prod_congr rfl
    intro i _
    rw [Real.exp_log (hpos i)]
  have h1 : HasProd f (TPRODN i : IDX, f i) := hmul.hasProd
  have h2 : HasSum (fun i => Real.log (f i)) (TSUMN i : IDX, Real.log (f i)) := hlog.hasSum
  have h3 : HasProd f (Real.exp (TSUMN i : IDX, Real.log (f i))) := by
    have hcomp := (Real.continuous_exp.tendsto _).comp h2
    apply hcomp.congr
    intro t
    rw [Function.comp_apply, LARR hpart t]
  have heq : (TPRODN i : IDX, f i) = Real.exp (TSUMN i : IDX, Real.log (f i)) :=
    h1.unique h3
  rw [heq, Real.log_exp]
