theorem reg_factors_tprod (s : RR) (hs : 1 LEQ s) (z : CC) (hc : 0 < cFac s)
    (hmul : Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} => (1 - ((p:NN):RR) ^ (-s))INV))
    (hpos : FORALL p : {p : NN // p.Prime AND 3 LEQ p}, 0 < (1 - ((p:NN):RR) ^ (-s))INV)
    (hlog : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((1 - ((p:NN):RR) ^ (-s))INV))) :
    ((cFac s : RR) : CC) ^ z
      = Complex.exp (z * ((TSUMN p : {p : NN // p.Prime AND 3 LEQ p},
          Real.log ((1 - ((p:NN):RR) ^ (-s))INV) : RR) : CC)) := by
  rw [cFac_cpow s hc z]
  congr 2
  have hlogeq : Real.log (cFac s)
      = TSUMN p : {p : NN // p.Prime AND 3 LEQ p},
          Real.log ((1 - ((p:NN):RR) ^ (-s))INV) := by
    unfold cFac
    exact log_tprod_real hpos hmul hlog
  rw [hlogeq]
