/-- s-free version of the per-prime bound.  hasDerivAt_tsum_of_isPreconnected
needs a dominating function independent of the point, so we replace p^(-2s)
by p^(-2), valid since s >= 1 and p >= 3 > 1. -/
theorem deriv_prime_term_bound_uniform (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : RR) (hz : |z| LEQ 2) :
    |(-(z * (z - 1) * Real.log (p:RR))
        / ((((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z)))|
      LEQ 27 * Real.log (p:RR) * ((p:RR)) ^ (-(2:RR)) := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hlogpos : (0:RR) LEQ Real.log (p:RR) := Real.log_nonneg (by linarith)
  have hbase := deriv_prime_term_bound p hp s hs z hz
  have hmono : ((p:RR)) ^ (-2*s) LEQ ((p:RR)) ^ (-(2:RR)) := by
    apply Real.rpow_le_rpow_of_exponent_le (by linarith)
    linarith
  have : 27 * Real.log (p:RR) * ((p:RR)) ^ (-2*s)
      LEQ 27 * Real.log (p:RR) * ((p:RR)) ^ (-(2:RR)) := by
    apply mul_le_mul_of_nonneg_left hmono
    linarith
  linarith

/-- The s-free dominator is summable. -/
theorem deriv_bound_uniform_summable :
    Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      27 * Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-(2:RR))) := by
  have h := deriv_series_summable 1 (le_refl 1)
  have heq : (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-2*(1:RR)))
      = fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-(2:RR)) := by
    funext p
    norm_num
  rw [heq] at h
  have h27 := h.mul_left 27
  apply h27.congr
  intro p
  ring
