/-- The correction series is continuous in s on {s // 1 <= s}. -/
theorem correction_series_continuous_s (z : CC) (hz : Norm.norm z LEQ 2) :
    Continuous (fun s : {s : RR // 1 LEQ s} =>
      TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-Complex.log (1 - z * (((((p:NN):RR)) ^ (-(s:RR)) : RR) : CC))
          + z * ((Real.log (1 - (((p:NN):RR)) ^ (-(s:RR))) : RR) : CC))) := by
  apply continuous_tsum
    (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      correction_continuous_s (p:NN) (by omega) z hz)
    correction_bound_uniform_summable_five
  intro p s
  exact correction_bound_sfree (p:NN) (by omega) (s:RR) s.2 z hz
