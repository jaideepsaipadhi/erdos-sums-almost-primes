/-- log p <= sqrt p for p >= 3.  Lets the differentiated log-series be
compared against a plain p-series rather than needing sum log p / p^2. -/
theorem log_le_sqrt (x : RR) (hx : 3 LEQ x) : Real.log x LEQ x ^ ((1:RR)/2) := by
  have hx0 : (0:RR) < x := by linarith
  have hsq : (0:RR) < x ^ ((1:RR)/2) := Real.rpow_pos_of_pos hx0 _
  have hkey : Real.log x = 2 * Real.log (x ^ ((1:RR)/2)) := by
    rw [Real.log_rpow hx0]
    ring
  rw [hkey]
  have hy := Real.log_le_sub_one_of_pos hsq
  have hsqge : (1:RR) LEQ x ^ ((1:RR)/2) := by
    have h1 : (1:RR) ^ ((1:RR)/2) LEQ x ^ ((1:RR)/2) := by
      apply Real.rpow_le_rpow (by norm_num) (by linarith) (by norm_num)
    simpa using h1
  nlinarith [hy, hsqge]
