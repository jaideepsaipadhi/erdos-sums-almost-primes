/-- The complex form of the p >= 3 Euler family is multipliable for s > 1,
by mapping the real one through the ofReal ring hom. -/
theorem cFac_multipliable_complex (s : RR) (hs : 1 < s) :
    Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (1 - (((p:NN):RR) : CC) ^ (-(s:CC)))INV) := by
  have hm := cFac_multipliable s hs
  have hmap := hm.map Complex.ofRealHom Complex.continuous_ofReal
  apply hmap.congr
  intro p
  have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
  show ((((1 - ((p:NN):RR) ^ (-s))INV : RR)) : CC) = _
  push_cast
  congr 1
  rw [Complex.ofReal_cpow (by linarith : (0:RR) LEQ ((p:NN):RR))]
  push_cast
  ring
