/-- The substitution map s = 1 + e^(-u) sends (0, infty) onto (1, 2). -/
theorem subst_image :
    (fun u : <<R>> => 1 + Real.exp (-u)) '' Set.Ioi 0 = Set.Ioo 1 2 := by
  ext s
  simp only [Set.mem_image, Set.mem_Ioi, Set.mem_Ioo]
  constructor
  <<DOT>> rintro <<LA>>u, hu, rfl<<RA>>
    dsimp only
    constructor
    <<DOT>> have h := Real.exp_pos (-u)
      linarith
    <<DOT>> have h1 : Real.exp (-u) < Real.exp 0 := Real.exp_lt_exp.mpr (by linarith)
      rw [Real.exp_zero] at h1
      linarith
  <<DOT>> rintro <<LA>>h1, h2<<RA>>
    refine <<LA>>-Real.log (s - 1), ?_, ?_<<RA>>
    <<DOT>> have hlog : Real.log (s - 1) < 0 :=
        Real.log_neg (by linarith) (by linarith)
      linarith
    <<DOT>> show 1 + Real.exp (-(-Real.log (s - 1))) = s
      rw [neg_neg, Real.exp_log (by linarith)]
      ring

/-- The substitution map is injective on (0, infty). -/
theorem subst_injOn :
    Set.InjOn (fun u : <<R>> => 1 + Real.exp (-u)) (Set.Ioi 0) := by
  intro u _ v _ huv
  dsimp only at huv
  have h2 : Real.exp (-u) = Real.exp (-v) := by linarith
  have h3 := Real.exp_injective h2
  linarith
