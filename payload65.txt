/-- e^u * Bfun m u is integrable on (0, infty): it equals
(e^(-(m log2) e^-u) - 1) e^u + m log2, times... -- bounded via the Bfun
decomposition against integrable pieces. -/
theorem integrable_exp_Bfun (m : <<N>>) :
    MeasureTheory.IntegrableOn (fun u : <<R>> => Real.exp u * Bfun m u) (Set.Ioi 0) := by
  sorry

/-- The key bound: sum 2^i betaCirc i m <= int e^u Bfun m u. -/
theorem sum_betaCirc_le_integral (m : <<N>>) :
    <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i m
      <<LE>> <<INT>> u in Set.Ioi (0:<<R>>), Real.exp u * Bfun m u := by
  rw [sum_betaCirc_eq_integral m]
  apply MeasureTheory.setIntegral_mono_on
  <<DOT>> have heq : (fun u : <<R>> =>
        (<<SUM>> i <<IN>> Finset.range (m + 1), (2 * u) ^ i / (Nat.factorial i))
          * (Real.exp (-u) * Bfun m u))
        = fun u : <<R>> => <<SUM>> i <<IN>> Finset.range (m + 1),
          ((2:<<R>>) ^ i / (Nat.factorial i)) * (u ^ i * Real.exp (-u) * Bfun m u) := by
      funext u
      rw [Finset.sum_mul]
      apply Finset.sum_congr rfl
      intro i _
      simp only [mul_pow]
      ring
    rw [heq]
    apply MeasureTheory.integrable_finsetSum
    intro i _
    exact (integrable_betaCirc_integrand i m).const_mul _
  <<DOT>> exact integrable_exp_Bfun m
  <<DOT>> exact measurableSet_Ioi
  <<DOT>> intro u hu
    have hu0 : (0:<<R>>) < u := hu
    have hsum := sum_pow_div_factorial_le_exp (2 * u) (by linarith) m
    have hB := Bfun_nonneg m u
    have hE : (0:<<R>>) < Real.exp (-u) := Real.exp_pos _
    have hfac : Real.exp (2 * u) * (Real.exp (-u) * Bfun m u)
        = Real.exp u * Bfun m u := by
      rw [show Real.exp (2 * u) * (Real.exp (-u) * Bfun m u)
          = (Real.exp (2 * u) * Real.exp (-u)) * Bfun m u by ring,
        <<LARR>> Real.exp_add]
      ring_nf
    calc (<<SUM>> i <<IN>> Finset.range (m + 1), (2 * u) ^ i / (Nat.factorial i))
          * (Real.exp (-u) * Bfun m u)
        <<LE>> Real.exp (2 * u) * (Real.exp (-u) * Bfun m u) := by
          apply mul_le_mul_of_nonneg_right hsum
          positivity
      _ = Real.exp u * Bfun m u := hfac
