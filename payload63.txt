/-- Replace the varying index: sum 2^i betaCirc i (m-i) <= sum 2^i betaCirc i m. -/
theorem sum_betaCirc_le_uniform (m : <<N>>) :
    <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i (m - i)
      <<LE>> <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i m := by
  apply Finset.sum_le_sum
  intro i hi
  apply mul_le_mul_of_nonneg_left _ (by positivity)
  exact betaCirc_mono i (m - i) m (Nat.sub_le m i)
