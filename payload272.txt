/-- The complex argument 1 - c*p^(-s) lies in the slit plane, for ||c|| <= 2,
p >= 3, s >= 1: its distance from 1 is at most 2/3, so its real part exceeds
1/3.  This is the side condition Complex.hasDerivAt_log needs. -/
theorem one_sub_mem_slitPlane (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (c : CC) (hc : Norm.norm c LEQ 2) :
    (1 : CC) - c * ((((p:NN):RR) ^ (-s) : RR) : CC) IN Complex.slitPlane := by
  have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < ((p:NN):RR) := by linarith
  have hppos : (0:RR) < ((p:NN):RR) ^ (-s) := Real.rpow_pos_of_pos hp0 _
  have hthird : ((p:NN):RR) ^ (-s) LEQ 1/3 := by
    have h1 : ((p:NN):RR) ^ (-s) LEQ ((p:NN):RR) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [Real.rpow_neg_one] at h1
    have h2 : (((p:NN):RR))INV LEQ 1/3 := by
      rw [inv_le_comm0 hp0 (by norm_num)]
      linarith
    linarith
  have hnorm : Norm.norm (c * ((((p:NN):RR) ^ (-s) : RR) : CC)) LEQ 2/3 := by
    rw [norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_of_pos hppos]
    nlinarith [norm_nonneg c, hc, hthird, hppos]
  rw [Complex.mem_slitPlane_iff]
  left
  have hre : (((1 : CC) - c * ((((p:NN):RR) ^ (-s) : RR) : CC))).re
      = 1 - (c * ((((p:NN):RR) ^ (-s) : RR) : CC)).re := by
    simp
  rw [hre]
  have habs : |(c * ((((p:NN):RR) ^ (-s) : RR) : CC)).re|
      LEQ Norm.norm (c * ((((p:NN):RR) ^ (-s) : RR) : CC)) := Complex.abs_re_le_norm _
  have := abs_le.mp (le_trans habs hnorm)
  linarith [this.2]
