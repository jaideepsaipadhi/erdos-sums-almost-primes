/-- A1 at radius 3/2 (the form Lemma 2.2 uses). -/
theorem zOmega_summable (s : RR) (hs : 1 < s) (z : CC) (hz : Norm.norm z LEQ 3/2) :
    Summable (fun n : NN => Norm.norm (zOmegaHom s z n)) := by
  apply zOmega_summable_radius s hs (3/2) (by norm_num) _ _ z hz
  DOT have h3 : (3:RR) ^ (-s) LEQ (3:RR) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
      linarith
    have h3one : (3:RR) ^ (-(1:RR)) = 1/3 := by
      rw [Real.rpow_neg_one]
      norm_num
    rw [h3one] at h3
    linarith
  DOT have h2 : (2:RR) ^ (1:RR) LEQ (2:RR) ^ s := by
      apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
      linarith
    rw [Real.rpow_one] at h2
    linarith
