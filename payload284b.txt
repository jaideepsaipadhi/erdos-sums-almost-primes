/-- d/ds of kappa(t)^z: the cpow of a positive real function, differentiated
via exp(z log kappa).  Gives kappa^z * z * (log kappa)', and
(log kappa)' = f(s) + log 2/(2^s - 1) by kappaFac_eq_zetaTilde. -/
theorem hasDerivAt_kappa_cpow (z : CC) (s : RR) (hsmem : s IN Set.Ioo (1:RR) 2)
    (f : RR ARROW RR)
    (hf : HasDerivAt (fun t : RR => Real.log (zetaTilde t)) (f s) s) :
    HasDerivAt (fun t : RR => ((kappaFac t : RR) : CC) ^ z)
      (((kappaFac s : RR) : CC) ^ z
        * (z * (((f s + Real.log 2 / ((2:RR) ^ s - 1) : RR)) : CC))) s := by
  have hs1 : 1 < s := hsmem.1
  have hklog : HasDerivAt (fun t : RR => Real.log (kappaFac t))
      (f s + Real.log 2 / ((2:RR) ^ s - 1)) s := by
    have hsplit : FORALL t : RR, t IN Set.Ioo (1:RR) 2 ARROW
        Real.log (kappaFac t) = Real.log (zetaTilde t) + Real.log (1 - (2:RR) ^ (-t)) := by
      intro t ht
      rw [kappaFac_eq_zetaTilde t ht.1,
        Real.log_mul (ne_of_gt (zetaTilde_pos t ht.1)) (ne_of_gt (one_sub_two_pow_pos t ht.1))]
    have hsum := hf.add (deriv_log_one_sub_two_pow s hs1)
    apply hsum.congr_of_eventuallyEq
    filter_upwards [Ioo_mem_nhds hsmem.1 hsmem.2] with t ht
    exact (hsplit t ht).symm
  have hcast : HasDerivAt (fun t : RR => ((Real.log (kappaFac t) : RR) : CC))
      (((f s + Real.log 2 / ((2:RR) ^ s - 1) : RR) : CC)) s := hklog.ofReal_comp
  have hmul : HasDerivAt (fun t : RR => z * ((Real.log (kappaFac t) : RR) : CC))
      (z * (((f s + Real.log 2 / ((2:RR) ^ s - 1) : RR)) : CC)) s := hcast.const_mul z
  have hexp := hmul.cexp
  have heq : FORALL t : RR, t IN Set.Ioo (1:RR) 2 ARROW
      Complex.exp (z * ((Real.log (kappaFac t) : RR) : CC)) = ((kappaFac t : RR) : CC) ^ z := by
    intro t ht
    have hkpos : 0 < kappaFac t := kappaFac_pos t ht.1 (cFac_pos t ht.1)
    rw [ofReal_cpow_eq_exp (kappaFac t) hkpos z]
    ring_nf
  rw [LARR heq s hsmem]
  apply hexp.congr_of_eventuallyEq
  filter_upwards [Ioo_mem_nhds hsmem.1 hsmem.2] with t ht
  exact (heq t ht).symm
