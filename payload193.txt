theorem Fk_eq_coeff_Fbig (k : NN) (hk : 1 LEQ k) (s : RR) (hs : 1 < s) (hs2 : s LEQ 2)
    (hc : 0 < cFac s)
    (hmulR : Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} => (1 - ((p:NN):RR) ^ (-s))INV))
    (hposR : FORALL p : {p : NN // p.Prime AND 3 LEQ p}, 0 < (1 - ((p:NN):RR) ^ (-s))INV)
    (hlogR : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((1 - ((p:NN):RR) ^ (-s))INV))) :
    ((Fk k s : RR) : CC)
      = (iteratedDeriv k (fun z => Fbig s z) 0) / (Nat.factorial k : CC) := by
  have hA4 := Fk_eq_coeff_tsum k hk s hs
  rw [hA4]
  congr 1
  apply Filter.EventuallyEq.iteratedDeriv_eq
  have hball : Metric.ball (0:CC) (3/2) IN nhds (0:CC) := by
    apply Metric.isOpen_ball.mem_nhds
    rw [Metric.mem_ball, dist_self]
    norm_num
  filter_upwards [hball] with z hz
  rw [Metric.mem_ball, dist_zero_right] at hz
  have hzle : Norm.norm z LEQ 3/2 := le_of_lt hz
  have hsum := zOmega_summable s hs z hzle
  have hA2 := zOmega_eulerProduct s hs z hsum
  have hA3 := eulerProduct_eq_Fbig s hs z hzle hc hmulR hposR hlogR
  rw [LARR hA2, hA3]
