/-- cBracket is s-continuous on {s // 1 <= s}: the kappa part (with fs held
fixed), the p = 3 term, and the tail series. -/
theorem cBracket_continuous_s (fs : RR) (z : CC) (hz : Norm.norm z LEQ 2) :
    Continuous (fun s : {s : RR // 1 LEQ s} => cBracket fs (s:RR) z) := by
  have hsub : Continuous (fun s : {s : RR // 1 LEQ s} => (s:RR)) := continuous_subtype_val
  have hrpow2 : Continuous (fun s : {s : RR // 1 LEQ s} => (2:RR) ^ (s:RR)) :=
    (continuous_const_rpow (by norm_num)).comp hsub
  have h2ge : FORALL s : {s : RR // 1 LEQ s}, (2:RR) LEQ (2:RR) ^ (s:RR) := by
    intro s
    calc (2:RR) = (2:RR) ^ (1:RR) := by norm_num
      _ LEQ (2:RR) ^ (s:RR) := Real.rpow_le_rpow_of_exponent_le (by norm_num) s.2
  have hne : FORALL s : {s : RR // 1 LEQ s}, (2:RR) ^ (s:RR) - 1 NEQ 0 := by
    intro s hcon
    linarith [h2ge s]
  have hkap : Continuous (fun s : {s : RR // 1 LEQ s} =>
      z * (((fs + Real.log 2 / ((2:RR) ^ (s:RR) - 1) : RR)) : CC)) := by
    apply continuous_const.mul
    apply Complex.continuous_ofReal.comp
    exact continuous_const.add (continuous_const.div (hrpow2.sub continuous_const) hne)
  unfold cBracket
  apply hkap.add
  apply Continuous.add
  DOT exact cBracket_term_continuous_s 3 (le_refl 3) z hz
  DOT exact cBracket_series_continuous_s z hz
