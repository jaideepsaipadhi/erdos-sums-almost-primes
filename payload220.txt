/-- Probe: does Mathlib's Nat.Primes-indexed Euler product typecheck against
tprod_split_two's {p // p.Prime} spelling without a bridge? -/
theorem zeta_split_probe (s : RR) (hs : 1 < s) :
    (TPRODN p : {p : NN // p.Prime}, (1 - (((p:NN):RR) : CC) ^ (-(s:CC)))INV)
      = riemannZeta (s : CC) := by
  have hre : (1:RR) < ((s : CC)).re := by simpa using hs
  have h := riemannZeta_eulerProduct_tprod hre
  rw [LARR h]
  apply tprod_congr
  intro p
  congr 1
  congr 1
  rw [Complex.ofReal_natCast]
