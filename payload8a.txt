/-- Telescoping bound: partial sums of 1/(j*sqrt j) over [2,N] never exceed 2. -/
theorem telescope_sqrt (N : <<N>>) :
    <<SUM>> j <<IN>> Finset.Icc 2 N, (1:<<R>>) / ((j:<<R>>) * Real.sqrt j) <<LE>> 2 := by
  have hstep : <<ALL>> M : <<N>>,
      <<SUM>> j <<IN>> Finset.Icc 2 M, (1:<<R>>) / ((j:<<R>>) * Real.sqrt j)
        <<LE>> 2 * (1 - 1 / Real.sqrt (M:<<R>>)) := by
    intro M
    induction M with
    | zero => simp
    | succ M ih =>
      rcases Nat.lt_or_ge M 1 with hM0 | hM1
      <<DOT>> have hM0' : M = 0 := by omega
        subst hM0'
        rw [show Finset.Icc 2 1 = (<<EMPTY>> : Finset <<N>>) by decide]
        simp
      <<DOT>> rw [Finset.sum_Icc_succ_top (by omega : 2 <<LE>> M + 1)]
        have hMR : (1:<<R>>) <<LE>> (M:<<R>>) := by exact_mod_cast hM1
        have hsM : (0:<<R>>) < Real.sqrt (M:<<R>>) := Real.sqrt_pos.mpr (by linarith)
        have hsM1 : (0:<<R>>) < Real.sqrt ((M:<<R>>) + 1) := Real.sqrt_pos.mpr (by linarith)
        have hmulM : Real.sqrt (M:<<R>>) * Real.sqrt (M:<<R>>) = (M:<<R>>) :=
          Real.mul_self_sqrt (by linarith)
        have hmulM1 : Real.sqrt ((M:<<R>>) + 1) * Real.sqrt ((M:<<R>>) + 1) = (M:<<R>>) + 1 :=
          Real.mul_self_sqrt (by linarith)
        have hle : Real.sqrt (M:<<R>>) <<LE>> Real.sqrt ((M:<<R>>) + 1) :=
          Real.sqrt_le_sqrt (by linarith)
        have hkey : (Real.sqrt ((M:<<R>>) + 1) - Real.sqrt (M:<<R>>)) *
            (Real.sqrt ((M:<<R>>) + 1) + Real.sqrt (M:<<R>>)) = 1 := by
          nlinarith [hmulM, hmulM1]
        have hcast : (((M + 1 : <<N>>)):<<R>>) = (M:<<R>>) + 1 := by push_cast; ring
        have htail : (1:<<R>>) / (((M + 1 : <<N>>):<<R>>) * Real.sqrt ((M + 1 : <<N>>):<<R>>))
            <<LE>> 2 * (1 / Real.sqrt (M:<<R>>) - 1 / Real.sqrt ((M:<<R>>) + 1)) := by
          rw [hcast]
          have hdenom : (0:<<R>>) < ((M:<<R>>) + 1) * Real.sqrt ((M:<<R>>) + 1) := by positivity
          rw [div_le_iff<<S0>> hdenom]
          have hdiff : 1 / Real.sqrt (M:<<R>>) - 1 / Real.sqrt ((M:<<R>>) + 1)
              = (Real.sqrt ((M:<<R>>) + 1) - Real.sqrt (M:<<R>>)) /
                (Real.sqrt (M:<<R>>) * Real.sqrt ((M:<<R>>) + 1)) := by
            field_simp
            try ring
          rw [hdiff]
          have hab_pos : (0:<<R>>) < Real.sqrt (M:<<R>>) * Real.sqrt ((M:<<R>>) + 1) := by
            positivity
          have hform : 2 * ((Real.sqrt ((M:<<R>>) + 1) - Real.sqrt (M:<<R>>)) /
                (Real.sqrt (M:<<R>>) * Real.sqrt ((M:<<R>>) + 1))) * (((M:<<R>>) + 1) * Real.sqrt ((M:<<R>>) + 1))
              = (2 * (Real.sqrt ((M:<<R>>) + 1) - Real.sqrt (M:<<R>>)) * (((M:<<R>>) + 1) * Real.sqrt ((M:<<R>>) + 1))) /
                (Real.sqrt (M:<<R>>) * Real.sqrt ((M:<<R>>) + 1)) := by
            ring
          rw [hform, le_div_iff<<S0>> hab_pos]
          nlinarith [hkey, hle, hsM, hsM1, hmulM1, hmulM,
            mul_pos hsM hsM1,
            mul_nonneg (mul_nonneg hsM1.le hsM1.le) (sub_nonneg.mpr hle),
            mul_nonneg hsM1.le (sub_nonneg.mpr hle)]
        have hfinal : 2 * (1 - 1 / Real.sqrt (M:<<R>>)) +
            2 * (1 / Real.sqrt (M:<<R>>) - 1 / Real.sqrt ((M:<<R>>) + 1))
            = 2 * (1 - 1 / Real.sqrt ((M:<<R>>) + 1)) := by ring
        calc <<SUM>> j <<IN>> Finset.Icc 2 M, (1:<<R>>) / ((j:<<R>>) * Real.sqrt j)
              + (1:<<R>>) / (((M + 1 : <<N>>):<<R>>) * Real.sqrt ((M + 1 : <<N>>):<<R>>))
            <<LE>> 2 * (1 - 1 / Real.sqrt (M:<<R>>)) +
              2 * (1 / Real.sqrt (M:<<R>>) - 1 / Real.sqrt ((M:<<R>>) + 1)) :=
              add_le_add ih htail
          _ = 2 * (1 - 1 / Real.sqrt ((M:<<R>>) + 1)) := hfinal
          _ = 2 * (1 - 1 / Real.sqrt (((M + 1 : <<N>>)):<<R>>)) := by rw [hcast]
  calc <<SUM>> j <<IN>> Finset.Icc 2 N, (1:<<R>>) / ((j:<<R>>) * Real.sqrt j)
      <<LE>> 2 * (1 - 1 / Real.sqrt (N:<<R>>)) := hstep N
    _ <<LE>> 2 := by
        have h1 : (0:<<R>>) <<LE>> 1 / Real.sqrt (N:<<R>>) := by positivity
        linarith

/-- Power-of-log bound: (log y)^m <<LE>> (2m)^m * sqrt y for y >= 1, m >= 1. -/
theorem log_pow_le_sqrt (m : <<N>>) (hm : 1 <<LE>> m) (y : <<R>>) (hy : 1 <<LE>> y) :
    (Real.log y) ^ m <<LE>> (2 * (m:<<R>>)) ^ m * Real.sqrt y := by
  have hypos : (0:<<R>>) < y := by linarith
  have hm_pos : (0:<<R>>) < (m:<<R>>) := by exact_mod_cast (by omega : 0 < m)
  have hm_ne : (m:<<R>>) <<NE>> 0 := hm_pos.ne'
  have h1 : Real.log (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) = (1 / (2 * (m:<<R>>))) * Real.log y :=
    Real.log_rpow hypos _
  have h2 : Real.log (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) <<LE>> y ^ ((1:<<R>>) / (2 * (m:<<R>>))) :=
    Real.log_le_self (Real.rpow_nonneg hypos.le _)
  have h3 : (1 / (2 * (m:<<R>>))) * Real.log y <<LE>> y ^ ((1:<<R>>) / (2 * (m:<<R>>))) := by
    rw [<<LARR>> h1]
    exact h2
  have hlog_le : Real.log y <<LE>> 2 * (m:<<R>>) * y ^ ((1:<<R>>) / (2 * (m:<<R>>))) := by
    calc Real.log y = 2 * (m:<<R>>) * ((1 / (2 * (m:<<R>>))) * Real.log y) := by
          field_simp
      _ <<LE>> 2 * (m:<<R>>) * y ^ ((1:<<R>>) / (2 * (m:<<R>>))) :=
          mul_le_mul_of_nonneg_left h3 (by positivity)
  have hlog_nn : (0:<<R>>) <<LE>> Real.log y := Real.log_nonneg hy
  have hp : (Real.log y) ^ m <<LE>> (2 * (m:<<R>>) * y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m := by
    apply pow_le_pow_left<<S0>> hlog_nn hlog_le
  have hsplit : (2 * (m:<<R>>) * y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m
      = (2 * (m:<<R>>)) ^ m * (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m := mul_pow _ _ _
  have hrpow : (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m = Real.sqrt y := by
    rw [<<LARR>> Real.rpow_natCast (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) m,
      <<LARR>> Real.rpow_mul hypos.le]
    rw [show (1:<<R>>) / (2 * (m:<<R>>)) * (m:<<R>>) = 1 / 2 by field_simp; try ring]
    rw [Real.sqrt_eq_rpow]
    norm_num
  calc (Real.log y) ^ m <<LE>> (2 * (m:<<R>>) * y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m := hp
    _ = (2 * (m:<<R>>)) ^ m * (y ^ ((1:<<R>>) / (2 * (m:<<R>>)))) ^ m := hsplit
    _ = (2 * (m:<<R>>)) ^ m * Real.sqrt y := by rw [hrpow]

/-- The summand 1/(t log t) is antitone past 2. -/
theorem inv_mul_log_le (a b : <<R>>) (ha : 2 <<LE>> a) (hab : a <<LE>> b) :
    1 / (b * Real.log b) <<LE>> 1 / (a * Real.log a) := by
  have hla : (0:<<R>>) < Real.log a := Real.log_pos (by linarith)
  have hlb : Real.log a <<LE>> Real.log b := Real.log_le_log (by linarith) hab
  have hpa : (0:<<R>>) < a * Real.log a := mul_pos (by linarith) hla
  apply one_div_le_one_div_of_le hpa
  apply mul_le_mul hab hlb hla.le (by linarith)

/-- Nonzero terms in dyadic block j inject into the piK count at 2^(j+1). -/
theorem block_count_le_piK (kk j M : <<N>>) :
    (((Finset.Ioc 1 M).filter (fun n => Nat.log 2 n = j)).filter
        (fun n => 2 <<LE>> n <<AND>> bigOmega n = kk)).card
      <<LE>> piK kk (((2:<<N>>) ^ (j + 1) : <<N>>) : <<R>>) := by
  classical
  unfold piK
  rw [Nat.floor_natCast]
  apply Finset.card_le_card
  intro n hn
  obtain <<LA>>hnf, hn2, hnO<<RA>> := Finset.mem_filter.mp hn
  have hb := dyadic_fiber_mem_bounds 1 M j n hnf
  exact Finset.mem_filter.mpr <<LA>>Finset.mem_Icc.mpr <<LA>>hn2, by omega<<RA>>, hnO<<RA>>
