/-- The Leibniz double sum, flattened to a sum over the sigma set. -/
theorem leibniz_flatten (k : <<N>>) (f : <<N>> <<TO>> <<N>> <<TO>> <<C>>) :
    <<SUM>> n <<IN>> Finset.range (k + 1), <<SUM>> i <<IN>> Finset.range (n + 1), f n i
      = <<SUM>> x <<IN>> (Finset.range (k + 1)).sigma (fun n => Finset.range (n + 1)),
          f x.1 x.2 := by
  rw [Finset.sum_sigma]
