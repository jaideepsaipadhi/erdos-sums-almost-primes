/-- The log^2-weighted series is dominated by a p-series, uniformly for s >= 1:
log^2 p * p^(-2s) <= (64/9) p^(3/4 - 2s) <= (64/9) p^(-5/4).
The r=2 analogue of deriv_series_summable. -/
theorem deriv2_series_summable (s : RR) (hs : 1 LEQ s) :
    Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (Real.log ((p:NN):RR)) ^ 2 * (((p:NN):RR)) ^ (-2*s)) := by
  have hcomp : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (64/9) * (((p:NN):RR)) ^ (-(5:RR)/4)) := by
    have hbase : Summable (fun n : NN => (n:RR) ^ (-(5:RR)/4)) := by
      rw [Real.summable_nat_rpow]
      norm_num
    exact (hbase.subtype _).mul_left (64/9)
  apply Summable.of_nonneg_of_le _ _ hcomp
  DOT intro p
    have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    positivity
  DOT intro p
    have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
    have hlog := log_sq_le (((p:NN):RR)) hp3
    have hpos : (0:RR) < (((p:NN):RR)) ^ (-2*s) := Real.rpow_pos_of_pos hp0 _
    have hstep : (Real.log ((p:NN):RR)) ^ 2 * (((p:NN):RR)) ^ (-2*s)
        LEQ ((64/9) * (((p:NN):RR)) ^ ((3:RR)/4)) * (((p:NN):RR)) ^ (-2*s) :=
      mul_le_mul_of_nonneg_right hlog (le_of_lt hpos)
    have hcomb : ((64/9) * (((p:NN):RR)) ^ ((3:RR)/4)) * (((p:NN):RR)) ^ (-2*s)
        = (64/9) * (((p:NN):RR)) ^ ((3:RR)/4 + -2*s) := by
      rw [mul_assoc, LARR Real.rpow_add hp0]
    have hmono : (((p:NN):RR)) ^ ((3:RR)/4 + -2*s) LEQ (((p:NN):RR)) ^ (-(5:RR)/4) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [hcomb] at hstep
    nlinarith [hstep, hmono, hpos]
