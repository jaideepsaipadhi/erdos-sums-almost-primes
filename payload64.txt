/-- The sum of 2^i betaCirc i m is the integral of the partial exponential sum. -/
theorem sum_betaCirc_eq_integral (m : <<N>>) :
    <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i m
      = <<INT>> u in Set.Ioi (0:<<R>>),
        (<<SUM>> i <<IN>> Finset.range (m + 1), (2 * u) ^ i / (Nat.factorial i))
          * (Real.exp (-u) * Bfun m u) := by
  have hterm : <<ALL>> i <<IN>> Finset.range (m + 1),
      (2:<<R>>) ^ i * betaCirc i m
        = <<INT>> u in Set.Ioi (0:<<R>>),
          ((2 * u) ^ i / (Nat.factorial i)) * (Real.exp (-u) * Bfun m u) := by
    intro i _
    unfold betaCirc
    rw [<<LARR>> MeasureTheory.integral_const_mul, <<LARR>> MeasureTheory.integral_const_mul]
    apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioi
    intro u _
    have hfac : ((Nat.factorial i : <<N>>) : <<R>>) <<NE>> 0 :=
      Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
    rw [mul_pow]
    field_simp
    ring
  rw [Finset.sum_congr rfl hterm]
  rw [<<LARR>> MeasureTheory.integral_finset_sum]
  <<DOT>> apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioi
    intro u _
    rw [Finset.sum_mul]
  <<DOT>> intro i _
    have hb := integrable_betaCirc_integrand i m
    have heq : (fun u : <<R>> => ((2 * u) ^ i / (Nat.factorial i)) * (Real.exp (-u) * Bfun m u))
        = fun u : <<R>> => ((2:<<R>>) ^ i / (Nat.factorial i)) *
          (u ^ i * Real.exp (-u) * Bfun m u) := by
      funext u
      rw [mul_pow]
      ring
    rw [heq]
    exact hb.const_mul _
