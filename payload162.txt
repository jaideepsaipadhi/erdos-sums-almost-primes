/-- At a prime, the z-weighted hom is just z * p^(-s). -/
theorem zOmegaHom_prime (s : RR) (z : CC) {p : NN} (hp : p.Prime) :
    zOmegaHom s z p = z * ((((p:NN):RR) ^ (-s) : RR) : CC) := by
  have hp0 : (p:NN) NEQ 0 := hp.pos.ne'
  show (if (p:NN) = 0 then 0
    else z ^ (bigOmega (p:NN)) * (((p:NN) : CC) ^ (-(s:CC)))) = _
  rw [if_neg hp0]
  have hO : bigOmega (p:NN) = 1 := by
    have := bigOmega_prime_pow hp 1
    rwa [pow_one] at this
  rw [hO, pow_one]
  congr 1
  rw [LARR Complex.ofReal_natCast, LARR Complex.ofReal_cpow (Nat.cast_nonneg p)]
  norm_cast
