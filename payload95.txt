theorem zOmega_prime_summable (s : RR) (hs : 1 < s) (z : CC)
    (hz : Norm.norm z < (2:RR) ^ s) {p : NN} (hp : p.Prime) :
    Summable (fun n : NN => Norm.norm (zOmegaHom s z (p ^ n))) := by
  have hp0 : (0:RR) < (p:RR) := by
    exact_mod_cast hp.pos
  have hp2 : (2:RR) LEQ (p:RR) := by
    exact_mod_cast hp.two_le
  have hratio : Norm.norm z * ((p:RR) ^ (-s)) < 1 := by
    have h2s : (0:RR) < (2:RR) ^ s := Real.rpow_pos_of_pos (by norm_num) s
    have hps : (p:RR) ^ (-s) LEQ (2:RR) ^ (-s) := by
      apply Real.rpow_le_rpow_of_exponent_nonpos
      DOT linarith
      DOT exact hp2
      DOT linarith
    have hinv : (2:RR) ^ (-s) = ((2:RR) ^ s)INV := by
      rw [Real.rpow_neg (by norm_num)]
    have hznn : (0:RR) LEQ Norm.norm z := norm_nonneg z
    calc Norm.norm z * ((p:RR) ^ (-s)) LEQ Norm.norm z * ((2:RR) ^ (-s)) := by
          apply mul_le_mul_of_nonneg_left hps hznn
      _ = Norm.norm z * ((2:RR) ^ s)INV := by rw [hinv]
      _ < 1 := by
          rw [mul_inv_lt_one_iff_of_pos h2s]
          exact hz
  sorry
