/-- The tail integrand e^(-v)/v is integrable on [a, infty) for a > 0,
dominated by e^(-v)/a. -/
theorem exp_div_integrableOn_Ici (a : <<R>>) (ha : 0 < a) :
    MeasureTheory.IntegrableOn (fun v : <<R>> => Real.exp (-v) / v) (Set.Ici a) := by
  have hexp : MeasureTheory.IntegrableOn (fun v : <<R>> => Real.exp (-v) / a) (Set.Ici a) := by
    have hbase : MeasureTheory.IntegrableOn (fun v : <<R>> => Real.exp (-v)) (Set.Ici a) := by
      have h := Real.GammaIntegral_convergent (s := (1:<<R>>)) (by norm_num)
      have hsub : Set.Ici a <<SS>> Set.Ioi (0:<<R>>) := by
        intro x hx
        simp only [Set.mem_Ici] at hx
        simp only [Set.mem_Ioi]
        linarith
      have h2 : MeasureTheory.IntegrableOn
          (fun u : <<R>> => Real.exp (-u) * u ^ ((1:<<R>>) - 1)) (Set.Ioi 0) := h
      have h3 : MeasureTheory.IntegrableOn (fun u : <<R>> => Real.exp (-u)) (Set.Ioi 0) := by
        apply h2.congr_fun ?_ measurableSet_Ioi
        intro u hu
        simp only
        rw [show (1:<<R>>) - 1 = 0 by ring, Real.rpow_zero, mul_one]
      exact h3.mono hsub le_rfl
    exact hbase.div_const a
  apply MeasureTheory.Integrable.mono hexp
  <<DOT>> apply Measurable.aestronglyMeasurable
    fun_prop
  <<DOT>> filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ici] with v hv
    have hva : a <<LE>> v := hv
    have hv0 : (0:<<R>>) < v := by linarith
    have hE : (0:<<R>>) < Real.exp (-v) := Real.exp_pos _
    rw [Real.norm_eq_abs, Real.norm_eq_abs,
      abs_of_nonneg (by positivity : (0:<<R>>) <<LE>> Real.exp (-v) / v),
      abs_of_nonneg (by positivity : (0:<<R>>) <<LE>> Real.exp (-v) / a)]
    apply div_le_div_of_nonneg_left hE.le ha hva
