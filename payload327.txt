theorem deriv2_clog_prime_term (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : CC) (hz : Norm.norm z LEQ 2) :
    HasDerivAt (fun t : RR =>
        -((z - 1) * ((Real.log (p:RR) : RR) : CC))
          / (((((p:RR)) ^ t : RR) : CC) - 1) / (((((p:RR)) ^ t : RR) : CC) - z))
      ((z - 1) * ((Real.log (p:RR) : RR) : CC) ^ 2 * ((((p:RR)) ^ s : RR) : CC)
        * ((((((p:RR)) ^ s : RR) : CC) - z) + (((((p:RR)) ^ s : RR) : CC) - 1))
        / ((((((p:RR)) ^ s : RR) : CC) - 1) ^ 2
          * ((((((p:RR)) ^ s : RR) : CC) - z) ^ 2))) s := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < (p:RR) := by linarith
  have hps : (3:RR) LEQ ((p:RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ ((p:RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hpsC : Norm.norm ((((p:RR)) ^ s : RR) : CC) = ((p:RR)) ^ s := by
    rw [Complex.norm_real, Real.norm_eq_abs, abs_of_pos (by linarith)]
  have hd1 : ((((p:RR)) ^ s : RR) : CC) - 1 NEQ 0 := by
    intro hcon
    rw [sub_eq_zero] at hcon
    have : ((p:RR)) ^ s = 1 := by exact_mod_cast hcon
    linarith
  have hd2 : ((((p:RR)) ^ s : RR) : CC) - z NEQ 0 := by
    intro hcon
    rw [sub_eq_zero] at hcon
    rw [LARR hcon, hpsC] at hz
    linarith
  have hbase : HasDerivAt (fun t : RR => ((((p:RR)) ^ t : RR) : CC))
      (((((p:RR)) ^ s * Real.log (p:RR) : RR)) : CC) s :=
    (hasDerivAt_rpow_pos p hp s).ofReal_comp
  have hA : HasDerivAt (fun t : RR => ((((p:RR)) ^ t : RR) : CC) - 1)
      (((((p:RR)) ^ s * Real.log (p:RR) : RR)) : CC) s := by
    simpa using hbase.sub_const 1
  have hB : HasDerivAt (fun t : RR => ((((p:RR)) ^ t : RR) : CC) - z)
      (((((p:RR)) ^ s * Real.log (p:RR) : RR)) : CC) s := by
    simpa using hbase.sub_const z
  have hstep1 := (hasDerivAt_const s (-((z - 1) * ((Real.log (p:RR) : RR) : CC)))).div hA hd1
  have hstep2 := hstep1.div hB hd2
  convert hstep2 using 1
  sorry
