/-- The j-th derivative of a single term. -/
theorem iteratedDeriv_zOmega_term (s : RR) (n : NN) (hn : n NEQ 0) (j : NN) (w : CC) :
    iteratedDeriv j (fun z : CC => zOmegaHom s z n) w
      = ((Nat.descFactorial (bigOmega n) j : NN) : CC) * w ^ (bigOmega n - j)
        * (((n:CC)) ^ (-(s:CC))) := by
  have hterm : (fun z : CC => zOmegaHom s z n)
      = fun z : CC => z ^ (bigOmega n) * (((n:CC)) ^ (-(s:CC))) := by
    funext z
    show (if n = 0 then 0 else z ^ (bigOmega n) * (((n : NN) : CC) ^ (-(s:CC)))) = _
    rw [if_neg hn]
  rw [hterm, iteratedDeriv_mul_const_field, iteratedDeriv_pow]
  ring
