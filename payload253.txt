theorem Ikfrozen_re_expand (k : NN) :
    (Ikfrozen k).re
      = 2 * (1/2 - wk k)
        - (Real.log 2 / 4)
            * ((k:RR)^2 * (dConst - Wk k) - 2 * (k:RR) * Uk k + Vk k) / (2:RR) ^ k
        + DeltaConv k / (2:RR) ^ k := by
  rw [Ikfrozen_re_eq]
  rw [Finset.sum_congr rfl (fun i hi => Ikfrozen_term k i (Nat.lt_succ_iff.mp (Finset.mem_range.mp hi)))]
  rw [Finset.sum_add_distrib, Finset.sum_sub_distrib]
  congr 1
  DOT congr 1
    DOT rw [LARR Finset.mul_sum, cCoeff_partial_sum]
    DOT rw [LARR Finset.sum_div, LARR Finset.mul_sum]
      congr 1
      rw [LARR Finset.sum_div]
      sorry
  DOT rw [LARR Finset.sum_div]
    unfold DeltaConv
    apply Finset.sum_congr rfl
    intro i _
    ring
