/-- The derivative of Lemma 3.9's bracket C(s) = z c(s,z).
Sum of the kappa-part, the p = 3 term, and the tail series. -/
theorem hasDerivAt_c_bracket (z : CC) (hz : Norm.norm z LEQ 2) (s : RR)
    (hsmem : s IN Set.Ioo (1:RR) 2) (f fp : RR ARROW RR)
    (hfp : HasDerivAt f (fp s) s) :
    HasDerivAt (fun t : RR =>
        z * (((f t + Real.log 2 / ((2:RR) ^ t - 1) : RR)) : CC)
          + ((-(z * (z - 1) * ((Real.log ((3:NN):RR) : RR) : CC))
              / ((((((3:NN):RR)) ^ t : RR) : CC) - 1)
                / ((((((3:NN):RR)) ^ t : RR) : CC) - z))
            + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
                (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
                  / ((((((p:NN):RR)) ^ t : RR) : CC) - 1)
                    / ((((((p:NN):RR)) ^ t : RR) : CC) - z))))
      (z * (((fp s + -((Real.log 2) ^ 2 * (2:RR) ^ s) / ((2:RR) ^ s - 1) ^ 2 : RR)) : CC)
        + ((z * ((z - 1) * ((Real.log ((3:NN):RR) : RR) : CC) ^ 2
              * (((((3:NN):RR)) ^ s : RR) : CC)
              * ((((((3:NN):RR)) ^ s : RR) : CC) - z
                + ((((((3:NN):RR)) ^ s : RR) : CC) - 1))
            / (((((((3:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
              * (((((((3:NN):RR)) ^ s : RR) : CC) - z) ^ 2))))
          + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
              (z * ((z - 1) * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2
                  * (((((p:NN):RR)) ^ s : RR) : CC)
                  * ((((((p:NN):RR)) ^ s : RR) : CC) - z
                    + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))
                / (((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
                  * (((((((p:NN):RR)) ^ s : RR) : CC) - z) ^ 2)))))) s := by
  have hkp := hasDerivAt_kappa_part z s hsmem f fp hfp
  have hp3 := deriv2_clog_prime_term_z 3 (le_refl 3) s (le_of_lt hsmem.1) z hz
  have htail := hasDerivAt_clog_series2 z hz s hsmem
  exact hkp.add (hp3.add htail)
