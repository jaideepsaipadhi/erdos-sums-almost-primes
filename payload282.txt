/-- d/ds of the p = 3 Euler factor, via exp of its log.
Same route as hasDerivAt_G2tailC applied to a single term:
prime_factor_eq_exp_complex expresses the factor as exp(correction), and
deriv_clog_prime_factor_canon (now at p >= 3) differentiates the exponent. -/
theorem hasDerivAt_prime_three_factor (z : CC) (hz : Norm.norm z LEQ 2) (s : RR)
    (hsmem : s IN Set.Ioo (1:RR) 2) :
    HasDerivAt (fun t : RR =>
        (1 - z * ((((3:RR)) ^ (-t) : RR) : CC))INV * (((1 - ((3:RR)) ^ (-t) : RR) : CC)) ^ z)
      (((1 - z * ((((3:RR)) ^ (-s) : RR) : CC))INV * (((1 - ((3:RR)) ^ (-s) : RR) : CC)) ^ z)
        * (-(z * (z - 1) * ((Real.log ((3:NN):RR) : RR) : CC))
            / (((((((3:NN):RR)) ^ s : RR)) : CC) - 1)
              / (((((((3:NN):RR)) ^ s : RR)) : CC) - z))) s := by
  have hder := deriv_clog_prime_factor_canon 3 (le_refl 3) s (le_of_lt hsmem.1) z hz
  have hexp := hder.cexp
  have heq : FORALL t : RR, t IN Set.Ioo (1:RR) 2 ARROW
      Complex.exp (-Complex.log (1 - z * ((((((3:NN):RR)) ^ (-t) : RR)) : CC))
          + z * ((Real.log (1 - (((3:NN):RR)) ^ (-t)) : RR) : CC))
        = (1 - z * ((((((3:NN):RR)) ^ (-t) : RR)) : CC))INV
            * (((1 - (((3:NN):RR)) ^ (-t) : RR) : CC)) ^ z := by
    intro t ht
    exact (prime_factor_eq_exp_complex 3 (le_refl 3) t (le_of_lt ht.1) z hz).symm
  rw [LARR heq s hsmem]
  apply hexp.congr_of_eventuallyEq
  filter_upwards [Ioo_mem_nhds hsmem.1 hsmem.2] with t ht
  exact (heq t ht).symm
