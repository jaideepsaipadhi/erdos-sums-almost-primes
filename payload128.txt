      have hsummable_nat : Summable (fun n : NN => Norm.norm z * ((n:RR) ^ (-s))) := by
        have hb : Summable (fun n : NN => ((n:RR) ^ (-s))) := by
          rw [Real.summable_nat_rpow]
          linarith
        exact hb.mul_left _

        intro n
        positivity
