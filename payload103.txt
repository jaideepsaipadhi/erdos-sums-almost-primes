/-- The set of primes where the scaled term exceeds 1/2 is finite. -/
theorem zOmega_bad_primes_finite (s : RR) (hs : 1 < s) (z : CC) :
    {p : Nat.Primes | NOT (Norm.norm z * (((p:NN):RR) ^ (-s)) LEQ 1/2)}.Finite := by
  have hsum := zOmega_prime_sum_summable s hs z
  have hnn : FORALL p : Nat.Primes, 0 LEQ Norm.norm z * (((p:NN):RR) ^ (-s)) := by
    intro p
    positivity
  have hsub : {p : Nat.Primes | NOT (Norm.norm z * (((p:NN):RR) ^ (-s)) LEQ 1/2)}
      SUBSET {p : Nat.Primes | 1/2 < Norm.norm z * (((p:NN):RR) ^ (-s))} := by
    intro p hp
    simp only [Set.mem_setOf_eq, not_le] at hp
    exact hp
  apply Set.Finite.subset _ hsub
  exact hsum.finite_of_lt (by norm_num : (0:RR) < 1/2)
