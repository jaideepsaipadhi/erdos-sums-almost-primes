/-- G2C is continuous in s on {s // 1 <= s}: the p = 3 factor times the tail.
The p = 3 factor is continuous because 1 - z*3^(-s) stays in the slit plane
and 1 - 3^(-s) stays positive for s >= 1. -/
theorem G2C_continuous_s (z : CC) (hz : Norm.norm z LEQ 2) :
    Continuous (fun s : {s : RR // 1 LEQ s} => G2C (s:RR) z) := by
  have hrpow : Continuous (fun s : {s : RR // 1 LEQ s} => ((3:RR)) ^ (-(s:RR))) :=
    (continuous_const_rpow (by norm_num)).comp continuous_subtype_val.neg
  have hlt : FORALL s : {s : RR // 1 LEQ s}, ((3:RR)) ^ (-(s:RR)) LEQ 1/3 := by
    intro s
    have h1 : ((3:RR)) ^ (-(s:RR)) LEQ ((3:RR)) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
      linarith [s.2]
    rw [Real.rpow_neg_one] at h1
    norm_num at h1
    linarith
  have hpos : FORALL s : {s : RR // 1 LEQ s}, (0:RR) < 1 - ((3:RR)) ^ (-(s:RR)) := by
    intro s
    linarith [hlt s]
  have hlogr : Continuous (fun s : {s : RR // 1 LEQ s} =>
      Real.log (1 - ((3:RR)) ^ (-(s:RR)))) :=
    Real.continuousOn_log.comp_continuous (continuous_const.sub hrpow)
      (fun s => ne_of_gt (hpos s))
  have hcpow : Continuous (fun s : {s : RR // 1 LEQ s} =>
      (((1 - ((3:RR)) ^ (-(s:RR)) : RR)) : CC) ^ z) := by
    have heq : (fun s : {s : RR // 1 LEQ s} =>
        (((1 - ((3:RR)) ^ (-(s:RR)) : RR)) : CC) ^ z)
        = fun s : {s : RR // 1 LEQ s} =>
          Complex.exp (z * ((Real.log (1 - ((3:RR)) ^ (-(s:RR))) : RR) : CC)) := by
      funext s
      rw [ofReal_cpow_eq_exp (1 - ((3:RR)) ^ (-(s:RR))) (hpos s) z]
      ring_nf
    rw [heq]
    exact Complex.continuous_exp.comp
      (continuous_const.mul (Complex.continuous_ofReal.comp hlogr))
  have hzne : FORALL s : {s : RR // 1 LEQ s},
      (1 : CC) - z * (((((3:RR)) ^ (-(s:RR)) : RR)) : CC) NEQ 0 := by
    intro s
    have := one_sub_mem_slitPlane 3 (le_refl 3) (s:RR) s.2 z hz
    norm_num at this
    exact Complex.slitPlane_ne_zero this
  have hinv : Continuous (fun s : {s : RR // 1 LEQ s} =>
      (1 - z * (((((3:RR)) ^ (-(s:RR)) : RR)) : CC))INV) := by
    apply Continuous.inv0 _ hzne
    exact continuous_const.sub (continuous_const.mul (Complex.continuous_ofReal.comp hrpow))
  unfold G2C
  exact (hinv.mul hcpow).mul (G2tailC_continuous_s z hz)
