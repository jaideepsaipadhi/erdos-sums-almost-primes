theorem hasDerivAt_log_series (z : RR) (hz : |z| LEQ 2) (s : RR)
    (hsmem : s IN Set.Ioo (1:RR) 2)
    (hconv : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      -Real.log (1 - z * (((p:NN):RR)) ^ (-(3/2:RR)))
        + z * Real.log (1 - (((p:NN):RR)) ^ (-(3/2:RR))))) :
    HasDerivAt (fun t : RR => TSUMN p : {p : NN // p.Prime AND 3 LEQ p},
        (-Real.log (1 - z * (((p:NN):RR)) ^ (-t))
          + z * Real.log (1 - (((p:NN):RR)) ^ (-t))))
      (TSUMN p : {p : NN // p.Prime AND 3 LEQ p},
        (-(z * (z - 1) * Real.log ((p:NN):RR))
          / (((((p:NN):RR)) ^ s - 1) * ((((p:NN):RR)) ^ s - z)))) s := by
  have hy0 : (3/2 : RR) IN Set.Ioo (1:RR) 2 := by
    constructor <;> norm_num
  apply hasDerivAt_tsum_of_isPreconnected
    deriv_bound_uniform_summable isOpen_Ioo isPreconnected_Ioo _ _ hy0 hconv hsmem
  DOT intro p y hy
    exact deriv_log_prime_factor (p:NN) p.2.2 y (le_of_lt hy.1) z hz
  DOT intro p y hy
    rw [Real.norm_eq_abs]
    exact deriv_prime_term_bound_uniform (p:NN) p.2.2 y (le_of_lt hy.1) z hz
