theorem deriv_geom_family (c : <<C>>) (hc : c <<NE>> 0) (k : <<N>>) :
    deriv (fun z : <<C>> => (Nat.factorial k : <<C>>) / (c ^ k * (1 - z / c) ^ (k + 1)))
      = fun z : <<C>> => (Nat.factorial (k + 1) : <<C>>) / (c ^ (k + 1) * (1 - z / c) ^ (k + 2)) := by
  funext z
  by_cases hz : (1 : <<C>>) - z / c = 0
  <<DOT>> sorry
  <<DOT>> have hbase : HasDerivAt (fun w : <<C>> => 1 - w / c) (-(1 / c)) z := by
      have h1 : HasDerivAt (fun w : <<C>> => w / c) (1 / c) z := by
        simpa using (hasDerivAt_id z).div_const c
      simpa using (hasDerivAt_const z (1:<<C>>)).sub h1
    have hpow : HasDerivAt (fun w : <<C>> => (1 - w / c) ^ (k + 1))
        (((k:<<C>>) + 1) * (1 - z / c) ^ k * (-(1 / c))) z := by
      have := hbase.pow (k + 1)
      simpa using this
    have hden : c ^ k * (1 - z / c) ^ (k + 1) <<NE>> 0 := by
      apply mul_ne_zero (pow_ne_zero _ hc)
      exact pow_ne_zero _ hz
    have hmul : HasDerivAt (fun w : <<C>> => c ^ k * (1 - w / c) ^ (k + 1))
        (c ^ k * (((k:<<C>>) + 1) * (1 - z / c) ^ k * (-(1 / c)))) z :=
      hpow.const_mul _
    have hdiv := (hasDerivAt_const z ((Nat.factorial k : <<C>>))).div hmul hden
    rw [hdiv.deriv]
    have hknz : (1 - z / c) ^ k <<NE>> 0 := pow_ne_zero _ hz
    push_cast [Nat.factorial_succ]
    field_simp
    ring
