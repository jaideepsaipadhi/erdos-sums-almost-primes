theorem ein_eq (a : <<R>>) (ha : 0 < a) :
    Ein a = Real.eulerMascheroniConstant + Real.log a + EinTailFun a := by
  set C := Ein 1 - Real.log 1 - EinTailFun 1 with hC
  have hconst : <<ALL>> n : <<N>>, 1 <<LE>> n <<TO>>
      Ein (n:<<R>>) - Real.log (n:<<R>>) - EinTailFun (n:<<R>>) = C := by
    intro n hn
    have hn0 : (0:<<R>>) < (n:<<R>>) := by
      have : (1:<<R>>) <<LE>> (n:<<R>>) := by exact_mod_cast hn
      linarith
    exact einCombo_const (n:<<R>>) 1 hn0 (by norm_num)
  have hseq : Filter.Tendsto
      (fun n : <<N>> => Ein (n:<<R>>) - Real.log (n:<<R>>) - EinTailFun (n:<<R>>))
      Filter.atTop (nhds (-(<<INT>> v in Set.Ioi (0:<<R>>), Real.exp (-v) * Real.log v))) := by
    have hrewrite : <<ALLF>> n : <<N>> in Filter.atTop,
        Ein (n:<<R>>) - Real.log (n:<<R>>) - EinTailFun (n:<<R>>)
          = -(<<INT>> v in Set.Ioc (0:<<R>>) (n:<<R>>), Real.exp (-v) * Real.log v)
            - Real.exp (-(n:<<R>>)) * Real.log (n:<<R>>) - EinTailFun (n:<<R>>) := by
      filter_upwards [Filter.eventually_ge_atTop 1] with n hn
      have hn0 : (0:<<R>>) < (n:<<R>>) := by
        have : (1:<<R>>) <<LE>> (n:<<R>>) := by exact_mod_cast hn
        linarith
      have hpl := euler_parts_limit (n:<<R>>) hn0
      have hexpand : (1 - Real.exp (-(n:<<R>>))) * Real.log (n:<<R>>)
          = Real.log (n:<<R>>) - Real.exp (-(n:<<R>>)) * Real.log (n:<<R>>) := by ring
      rw [hexpand] at hpl
      linarith [hpl]
    have hlim := ((expLog_partial_tendsto.neg.sub expNeg_log_tendsto_zero).sub
      EinTailFun_tendsto_zero)
    have hlim2 : Filter.Tendsto
        (fun n : <<N>> => -(<<INT>> v in Set.Ioc (0:<<R>>) (n:<<R>>), Real.exp (-v) * Real.log v)
          - Real.exp (-(n:<<R>>)) * Real.log (n:<<R>>) - EinTailFun (n:<<R>>))
        Filter.atTop (nhds (-(<<INT>> v in Set.Ioi (0:<<R>>), Real.exp (-v) * Real.log v))) := by
      simpa using hlim
    exact hlim2.congr' (Filter.EventuallyEq.symm hrewrite)
  have hCval : C = -(<<INT>> v in Set.Ioi (0:<<R>>), Real.exp (-v) * Real.log v) := by
    have hev : <<ALLF>> n : <<N>> in Filter.atTop,
        Ein (n:<<R>>) - Real.log (n:<<R>>) - EinTailFun (n:<<R>>) = C := by
      filter_upwards [Filter.eventually_ge_atTop 1] with n hn
      exact hconst n hn
    have hcst : Filter.Tendsto
        (fun _ : <<N>> => C) Filter.atTop
        (nhds (-(<<INT>> v in Set.Ioi (0:<<R>>), Real.exp (-v) * Real.log v))) :=
      hseq.congr' hev
    exact tendsto_nhds_unique tendsto_const_nhds hcst
  have hgamma : Real.eulerMascheroniConstant
      = -(<<INT>> v in Set.Ioi (0:<<R>>), Real.exp (-v) * Real.log v) :=
    eulerMascheroni_eq_integral
  have hCa : Ein a - Real.log a - EinTailFun a = C :=
    einCombo_const a 1 ha (by norm_num)
  rw [hCval] at hCa
  rw [<<LARR>> hgamma] at hCa
  linarith [hCa]
