/-- MANUSCRIPT section 3.5: |E_1(k)| <= alpha_+ (k+1).
From (10), since (k-i+1) <= (k+1) for every i in the sum.  alpha_+ bounds
(1/2) sum_{i<=k} beta_i 2^i uniformly in k -- the manuscript evaluates it by
splitting at 30 and using the Cauchy tail beta_i <= M1/2.9^i, which is a
certified computation, so it enters here as a hypothesis. -/
theorem E1k_le_alpha (k : NN) (hk : 1 LEQ k) (beta : NN ARROW RR)
    (hbeta0 : FORALL i, 0 LEQ beta i)
    (hderiv : FORALL i : NN, FORALL x IN Set.Ioo (1:RR) 2,
      HasDerivAt (fun t : RR => gManu i t) (deriv (fun t : RR => gManu i t) x) x)
    (hbnd : FORALL i : NN, FORALL x IN Set.Ioo (1:RR) 2,
      Norm.norm (deriv (fun t : RR => gManu i t) x) LEQ beta i)
    (M0 : RR) (hM00 : 0 LEQ M0)
    (hM0 : FORALL u : RR, u IN Set.Ioi (1:RR) ARROW FORALL theta : RR,
      Norm.norm (G2manu u (circleMap 0 (2:RR) theta)) LEQ M0)
    (alpha : RR) (halpha : FORALL m : NN,
      (1/2) * SUMF i IN Finset.range (m + 1), beta i * (2:RR) ^ i LEQ alpha) :
    |E1k k| LEQ alpha * ((k : RR) + 1) := by
  have hb := E1k_bound k hk beta hbeta0 hderiv hbnd M0 hM00 hM0
  have hstep : (1/2) * SUMF i IN Finset.range (k + 1),
        beta i * (2:RR) ^ i * ((k : RR) - (i : RR) + 1)
      LEQ ((k : RR) + 1) * ((1/2) * SUMF i IN Finset.range (k + 1), beta i * (2:RR) ^ i) := by
    rw [Finset.mul_sum, Finset.mul_sum, Finset.mul_sum]
    apply Finset.sum_le_sum
    intro i hi
    have hik : i LEQ k := Nat.lt_succ_iff.mp (Finset.mem_range.mp hi)
    have hikR : (i : RR) LEQ (k : RR) := by exact_mod_cast hik
    have hi0 : (0:RR) LEQ (i : RR) := by positivity
    have hnn : (0:RR) LEQ beta i * (2:RR) ^ i := by
      apply mul_nonneg (hbeta0 i)
      positivity
    have hfac : (k : RR) - (i : RR) + 1 LEQ (k : RR) + 1 := by linarith
    nlinarith [hnn, hfac]
  calc |E1k k| LEQ (1/2) * SUMF i IN Finset.range (k + 1),
        beta i * (2:RR) ^ i * ((k : RR) - (i : RR) + 1) := hb
    _ LEQ ((k : RR) + 1) * ((1/2) * SUMF i IN Finset.range (k + 1),
        beta i * (2:RR) ^ i) := hstep
    _ LEQ ((k : RR) + 1) * alpha := by
        apply mul_le_mul_of_nonneg_left (halpha k)
        positivity
    _ = alpha * ((k : RR) + 1) := by ring
