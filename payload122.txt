/-- With ||z|| <= 3/2 and s > 1, every prime p >= 3 satisfies ||z|| p^(-s) <= 1/2. -/
theorem zOmega_half_of_three_le (s : RR) (hs : 1 < s) (z : CC)
    (hz : Norm.norm z LEQ 3/2) {p : NN} (hp3 : 3 LEQ p) :
    Norm.norm z * ((p:RR) ^ (-s)) LEQ 1/2 := by
  have hp3R : (3:RR) LEQ (p:RR) := by exact_mod_cast hp3
  have hppos : (0:RR) < (p:RR) := by linarith
  have hps : (p:RR) ^ (-s) LEQ (3:RR) ^ (-s) := by
    rw [Real.rpow_neg (by linarith : (0:RR) LEQ (p:RR)),
      Real.rpow_neg (by norm_num : (0:RR) LEQ (3:RR))]
    gcongr
  have h3s : (3:RR) ^ (-s) LEQ (3:RR) ^ (-(1:RR)) := by
    apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
    linarith
  have h3one : (3:RR) ^ (-(1:RR)) = 1/3 := by
    rw [Real.rpow_neg_one]
    norm_num
  have hznn : (0:RR) LEQ Norm.norm z := norm_nonneg z
  calc Norm.norm z * ((p:RR) ^ (-s)) LEQ (3/2) * ((3:RR) ^ (-s)) := by
        apply mul_le_mul hz hps (by positivity) (by norm_num)
    _ LEQ (3/2) * (1/3) := by
        rw [LARR h3one]
        apply mul_le_mul_of_nonneg_left h3s (by norm_num)
    _ = 1/2 := by norm_num
