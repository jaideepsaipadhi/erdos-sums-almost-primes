/-- The per-prime derivative sum is continuous in theta on the contour.
continuous_tsum needs a GLOBAL bound (all theta), which holds here precisely
because circleMap always lands on |z| = 2, where
deriv_clog_term_bound_uniform applies. -/
theorem clog_series_continuous_theta (s : RR) (hs : 1 LEQ s) :
    Continuous (fun theta : RR =>
      TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
            * ((Real.log ((p:NN):RR) : RR) : CC))
          / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
            / (((((((p:NN):RR)) ^ s : RR)) : CC) - circleMap 0 (2:RR) theta))) := by
  have hnorm : FORALL theta : RR, Norm.norm (circleMap 0 (2:RR) theta) LEQ 2 := by
    intro theta
    simp [circleMap]
  apply continuous_tsum
    (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      clog_term_continuous_theta (p:NN) (by omega) s hs)
    deriv_bound_uniform_summable_five
  intro p theta
  exact deriv_clog_term_bound_uniform (p:NN) (by omega) s hs
    (circleMap 0 (2:RR) theta) (hnorm theta)
