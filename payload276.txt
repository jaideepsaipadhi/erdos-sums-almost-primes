/-- Complex analogue of deriv_prime_term_bound_uniform: the s-free dominating
bound for the per-prime derivative.  Same constant 27, same crude route, with
norms in place of absolute values. -/
theorem deriv_clog_term_bound_uniform (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : CC) (hz : Norm.norm z LEQ 2) :
    Norm.norm (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
        / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
          / (((((((p:NN):RR)) ^ s : RR)) : CC) - z))
      LEQ 27 * Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-(2:RR)) := by
  have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp
  have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
  have hps : (3:RR) LEQ (((p:NN):RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ (((p:NN):RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hps0 : (0:RR) < (((p:NN):RR)) ^ s := by linarith
  have hlogpos : (0:RR) LEQ Real.log ((p:NN):RR) := Real.log_nonneg (by linarith)
  have hnum : Norm.norm (z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
      LEQ 6 * Real.log ((p:NN):RR) := by
    rw [norm_mul, norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_of_nonneg hlogpos]
    have h1 : Norm.norm (z - 1) LEQ 3 := by
      calc Norm.norm (z - 1) LEQ Norm.norm z + Norm.norm (1:CC) := norm_sub_le z 1
        _ LEQ 2 + 1 := by simp only [norm_one]; linarith
        _ = 3 := by norm_num
    nlinarith [norm_nonneg z, norm_nonneg (z - 1), hz, h1, hlogpos]
  have hd1 : (2/3) * (((p:NN):RR)) ^ s
      LEQ Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC)) - 1) := by
    have hre : ((((((((p:NN):RR)) ^ s : RR)) : CC)) - 1)
        = (((((((p:NN):RR)) ^ s - 1 : RR))) : CC) := by push_cast; ring
    rw [hre, Complex.norm_real, Real.norm_eq_abs, abs_of_pos (by linarith)]
    linarith
  have hd2 : (1/3) * (((p:NN):RR)) ^ s
      LEQ Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC)) - z) := by
    have hge : Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC)) - z)
        GEQ Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC))) - Norm.norm z :=
      norm_sub_norm_le _ _
    rw [Complex.norm_real, Real.norm_eq_abs, abs_of_pos hps0] at hge
    linarith
  rw [norm_div, norm_div]
  rw [div_div]
  rw [div_le_iff0 (by positivity)]
  have hprod : (2/9) * ((((p:NN):RR)) ^ s) ^ 2
      LEQ Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC)) - 1)
          * Norm.norm ((((((((p:NN):RR)) ^ s : RR)) : CC)) - z) := by
    nlinarith [hd1, hd2, hps0, norm_nonneg ((((((((p:NN):RR)) ^ s : RR)) : CC)) - 1)]
  have hpow2 : (((p:NN):RR)) ^ (-(2:RR)) * ((((p:NN):RR)) ^ s) ^ 2
      GEQ (((p:NN):RR)) ^ (-(2:RR)) * ((((p:NN):RR)) ^ (1:RR)) ^ 2 := by
    have hmono : ((((p:NN):RR)) ^ (1:RR)) ^ 2 LEQ ((((p:NN):RR)) ^ s) ^ 2 := by
      have := Real.rpow_le_rpow_of_exponent_le (by linarith : (1:RR) LEQ (((p:NN):RR))) hs
      nlinarith [this, Real.rpow_pos_of_pos hp0 (1:RR)]
    have : (0:RR) < (((p:NN):RR)) ^ (-(2:RR)) := Real.rpow_pos_of_pos hp0 _
    nlinarith [hmono, this]
  rw [norm_neg]
  nlinarith [hnum, hprod, hpow2, hlogpos, hps0]
