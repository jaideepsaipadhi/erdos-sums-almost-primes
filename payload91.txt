theorem L2b_summand (k : NN) (s : RR) (hs : 1 < s) (n i : NN) (hin : i LEQ n) (hnk : n LEQ k) :
    (Nat.choose k n : CC) *
        ((Nat.choose n i : CC)
          * ((Nat.factorial i : CC) * ((((-Real.log (s - 1)) ^ i / (Nat.factorial i) : RR)) : CC))
          * ((Nat.factorial (n - i) : CC) * ((((2:RR) ^ (-((n - i : NN):RR) * s) : RR)) : CC)))
        * ((Nat.factorial (k - n) : CC) * gCoeff (k - n) s)
      / (Nat.factorial k : CC)
      = gCoeff (k - n) s * ((((2:RR) ^ (-((n - i : NN):RR) * s) : RR)) : CC)
        * ((((-Real.log (s - 1)) ^ (k - (n - i) - (k - n))
            / (Nat.factorial (k - (n - i) - (k - n))) : RR)) : CC) := by
  have hidx : k - (n - i) - (k - n) = i := by omega
  rw [hidx]
  have hkfac : ((Nat.factorial k : NN) : CC) NEQ 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero k)
  have htri := choose_mul_choose_div_factorial k n i hnk hin
  rw [div_eq_iff hkfac]
  rw [htri]
  have hfi : ((Nat.factorial i : NN) : CC) NEQ 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
  have hfni : ((Nat.factorial (n - i) : NN) : CC) NEQ 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero (n - i))
  have hfkn : ((Nat.factorial (k - n) : NN) : CC) NEQ 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero (k - n))
  field_simp
  ring
