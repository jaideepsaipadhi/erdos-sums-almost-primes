/-- G2manu is differentiable in z on the closed disc of radius 12/5, for each
fixed s.  The s-independent half of Lemma 3.8; generalises
G2at1_differentiableOn_wide from s = 1. -/
theorem G2manu_differentiableOn_wide (s : RR) (hs : 1 LEQ s) (hk : 0 < kappaFac s) :
    DifferentiableOn CC (fun z : CC => G2manu s z) (Metric.closedBall (0:CC) (12/5 : RR)) := by
  have hsub : Metric.closedBall (0:CC) (12/5 : RR) SUBSET Metric.ball (0:CC) (5/2) := by
    intro z hz
    rw [Metric.mem_closedBall, dist_zero_right] at hz
    rw [Metric.mem_ball, dist_zero_right]
    linarith
  have hcpow : DifferentiableOn CC (fun z : CC => ((kappaFac s : RR) : CC) ^ z)
      (Metric.closedBall (0:CC) (12/5 : RR)) := by
    apply Differentiable.differentiableOn
    intro z
    apply DifferentiableAt.const_cpow differentiableAt_id
    left
    simp only [ne_eq, Complex.ofReal_eq_zero]
    exact ne_of_gt hk
  have hG2C : DifferentiableOn CC (fun z : CC => G2C s z)
      (Metric.closedBall (0:CC) (12/5 : RR)) :=
    (G2C_differentiableOn_wide s hs).mono hsub
  unfold G2manu
  exact hcpow.mul hG2C
