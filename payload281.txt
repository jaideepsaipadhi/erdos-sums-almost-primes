/-- d/ds G2tailC = G2tailC * (sum of per-prime derivatives).
The key step of the complex chain: G2tailC is exp of the log-series
(G2tailC_eq_exp_sum), so differentiating goes through HasDerivAt.cexp and
never requires Complex.log of the product -- which would be false in general
because of branch cuts. -/
theorem hasDerivAt_G2tailC (z : CC) (hz : Norm.norm z LEQ 2) (s : RR)
    (hsmem : s IN Set.Ioo (1:RR) 2) :
    HasDerivAt (fun t : RR => G2tailC t z)
      (G2tailC s z * TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
          / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
            / (((((((p:NN):RR)) ^ s : RR)) : CC) - z))) s := by
  have hser := hasDerivAt_clog_series z hz s hsmem
  have hexp := hser.cexp
  have heq : FORALL t : RR, t IN Set.Ioo (1:RR) 2 ARROW
      Complex.exp (TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-Complex.log (1 - z * (((((p:NN):RR)) ^ (-t) : RR) : CC))
          + z * ((Real.log (1 - (((p:NN):RR)) ^ (-t)) : RR) : CC)))
      = G2tailC t z := by
    intro t ht
    exact (G2tailC_eq_exp_sum t (le_of_lt ht.1) z hz).symm
  have hval : Complex.exp (TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-Complex.log (1 - z * (((((p:NN):RR)) ^ (-s) : RR) : CC))
          + z * ((Real.log (1 - (((p:NN):RR)) ^ (-s)) : RR) : CC)))
      = G2tailC s z := heq s hsmem
  rw [LARR hval]
  apply hexp.congr_of_eventuallyEq
  filter_upwards [Ioo_mem_nhds hsmem.1 hsmem.2] with t ht
  exact (heq t ht).symm
