/-- The log of the real regularised p >= 3 product is the sum of the logs.
This is what turns hasDerivAt_log_series (a statement about the tsum) into a
statement about the product, i.e. about log G2 restricted to its p >= 3 part. -/
theorem log_reg_tprod_real (s : RR) (hs : 1 LEQ s) (z : RR) (hz : |z| LEQ 2) :
    Real.log (TPRODN p : {p : NN // p.Prime AND 3 LEQ p},
        (1 - z * (((p:NN):RR)) ^ (-s))INV * (1 - (((p:NN):RR)) ^ (-s)) ^ z)
      = TSUMN p : {p : NN // p.Prime AND 3 LEQ p},
        (-Real.log (1 - z * (((p:NN):RR)) ^ (-s))
          + z * Real.log (1 - (((p:NN):RR)) ^ (-s))) := by
  have hlogeq : FORALL p : {p : NN // p.Prime AND 3 LEQ p},
      Real.log ((1 - z * (((p:NN):RR)) ^ (-s))INV * (1 - (((p:NN):RR)) ^ (-s)) ^ z)
        = -Real.log (1 - z * (((p:NN):RR)) ^ (-s))
          + z * Real.log (1 - (((p:NN):RR)) ^ (-s)) := by
    intro p
    have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
    have hppos : (0:RR) < (((p:NN):RR)) ^ (-s) := Real.rpow_pos_of_pos hp0 _
    have hthird : (((p:NN):RR)) ^ (-s) LEQ 1/3 := by
      have h1 : (((p:NN):RR)) ^ (-s) LEQ (((p:NN):RR)) ^ (-(1:RR)) := by
        apply Real.rpow_le_rpow_of_exponent_le (by linarith)
        linarith
      rw [Real.rpow_neg_one] at h1
      have h2 : ((((p:NN):RR)))INV LEQ 1/3 := by
        rw [inv_le_comm0 hp0 (by norm_num)]
        linarith
      linarith
    have hzabs := abs_le.mp hz
    have hden : (0:RR) < 1 - z * (((p:NN):RR)) ^ (-s) := by
      nlinarith [hzabs.2, hppos, hthird]
    have hbase : (0:RR) < 1 - (((p:NN):RR)) ^ (-s) := by linarith
    rw [Real.log_mul (by positivity) (by positivity), Real.log_inv, Real.log_rpow hbase]
  rw [log_tprod_real (fun p => reg_factor_real_pos (p:NN) p.2.2 s hs z hz)
    (reg_multipliable_three_real s hs z hz)
    (by
      have hsum := (prime_correction_summable s hs z hz).of_abs
      apply hsum.congr
      intro p
      exact (hlogeq p).symm)]
  exact tsum_congr hlogeq
