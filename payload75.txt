/-- Pointwise version away from the pole. -/
theorem deriv_geom_family_at (c : <<C>>) (hc : c <<NE>> 0) (k : <<N>>) (z : <<C>>)
    (hz : 1 - z / c <<NE>> 0) :
    deriv (fun w : <<C>> => (Nat.factorial k : <<C>>) / (c ^ k * (1 - w / c) ^ (k + 1))) z
      = (Nat.factorial (k + 1) : <<C>>) / (c ^ (k + 1) * (1 - z / c) ^ (k + 2)) := by
  have hbase : HasDerivAt (fun w : <<C>> => 1 - w / c) (-(1 / c)) z := by
    have h1 : HasDerivAt (fun w : <<C>> => w / c) (1 / c) z := by
      simpa using (hasDerivAt_id z).div_const c
    simpa [Pi.sub_def] using (hasDerivAt_const z (1:<<C>>)).sub h1
  have hpow : HasDerivAt (fun w : <<C>> => (1 - w / c) ^ (k + 1))
      (((k:<<C>>) + 1) * (1 - z / c) ^ k * (-(1 / c))) z := by
    have hp := hbase.pow (k + 1)
    simpa [Pi.pow_def] using hp
  have hden : c ^ k * (1 - z / c) ^ (k + 1) <<NE>> 0 := by
    apply mul_ne_zero (pow_ne_zero _ hc)
    exact pow_ne_zero _ hz
  have hmul : HasDerivAt (fun w : <<C>> => c ^ k * (1 - w / c) ^ (k + 1))
      (c ^ k * (((k:<<C>>) + 1) * (1 - z / c) ^ k * (-(1 / c)))) z :=
    hpow.const_mul _
  have hdiv0 := (hasDerivAt_const z ((Nat.factorial k : <<C>>))).div hmul hden
  have hdiv : HasDerivAt (fun w : <<C>> => (Nat.factorial k : <<C>>) / (c ^ k * (1 - w / c) ^ (k + 1)))
      ((0 * (c ^ k * (1 - z / c) ^ (k + 1))
        - (Nat.factorial k : <<C>>) * (c ^ k * (((k:<<C>>) + 1) * (1 - z / c) ^ k * (-(1 / c)))))
        / (c ^ k * (1 - z / c) ^ (k + 1)) ^ 2) z := by
    simpa [Pi.div_def] using hdiv0
  rw [hdiv.deriv]
  set w := 1 - z / c with hw
  have hwnz : w <<NE>> 0 := hz
  have hnum : (0 * (c ^ k * w ^ (k + 1))
      - (Nat.factorial k : <<C>>) * (c ^ k * (((k:<<C>>) + 1) * w ^ k * (-(1 / c)))))
      / (c ^ k * w ^ (k + 1)) ^ 2
      = ((Nat.factorial k : <<C>>) * ((k:<<C>>) + 1)) / (c ^ (k + 1) * w ^ (k + 2)) := by
    rw [div_eq_div_iff (by positivity) (by positivity)]
    field_simp
    ring
  rw [hnum]
  congr 1
  push_cast [Nat.factorial_succ]
  ring
