/-- The differentiated log-series is dominated by a p-series, uniformly for
s >= 1: log p * p^(-2s) <= p^(1/2 - 2s) <= p^(-3/2).
This is the estimate behind both Lemma 3.8's "product converging locally
uniformly" and Lemma 3.9's termwise differentiation. -/
theorem deriv_series_summable (s : RR) (hs : 1 LEQ s) :
    Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-2*s)) := by
  have hcomp : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (((p:NN):RR)) ^ (-(3:RR)/2) := by
    have hbase : Summable (fun n : NN => (n:RR) ^ (-(3:RR)/2)) := by
      rw [Real.summable_nat_rpow]
      norm_num
    exact hbase.subtype _
  apply Summable.of_nonneg_of_le _ _ hcomp
  DOT intro p
    have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
    have hlog : (0:RR) LEQ Real.log ((p:NN):RR) := Real.log_nonneg (by linarith)
    positivity
  DOT intro p
    have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < ((p:NN):RR) := by linarith
    have hlog := log_le_sqrt ((p:NN):RR) hp3
    have hstep : Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-2*s)
        LEQ (((p:NN):RR)) ^ ((1:RR)/2) * (((p:NN):RR)) ^ (-2*s) := by
      apply mul_le_mul_of_nonneg_right hlog (by positivity)
    have hcomb : (((p:NN):RR)) ^ ((1:RR)/2) * (((p:NN):RR)) ^ (-2*s)
        = (((p:NN):RR)) ^ ((1:RR)/2 - 2*s) := by
      rw [LARR Real.rpow_add hp0]
      ring_nf
    have hmono : (((p:NN):RR)) ^ ((1:RR)/2 - 2*s) LEQ (((p:NN):RR)) ^ (-(3:RR)/2) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [hcomb] at hstep
    linarith
