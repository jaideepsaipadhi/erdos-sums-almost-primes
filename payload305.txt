/-- Interior form of the mean value bound: for 1 < t, s < 2,
||g_i(s) - g_i(t)|| <= beta * |s - t|, given a derivative bound beta on (1,2).
The endpoint version follows by letting t -> 1+ and using gManu_tendsto_cCoeff. -/
theorem gManu_lipschitz_interior (i : NN) (beta : RR)
    (hderiv : FORALL x IN Set.Ioo (1:RR) 2,
      HasDerivAt (fun t : RR => gManu i t) (deriv (fun t : RR => gManu i t) x) x)
    (hbound : FORALL x IN Set.Ioo (1:RR) 2,
      Norm.norm (deriv (fun t : RR => gManu i t) x) LEQ beta)
    (t s : RR) (ht : t IN Set.Ioo (1:RR) 2) (hs : s IN Set.Ioo (1:RR) 2) :
    Norm.norm (gManu i s - gManu i t) LEQ beta * Norm.norm (s - t) := by
  apply Convex.norm_image_sub_le_of_norm_hasDerivWithin_le
    (f' := fun x : RR => deriv (fun u : RR => gManu i u) x)
    (fun x hx => (hderiv x hx).hasDerivWithinAt) hbound (convex_Ioo 1 2) ht hs
