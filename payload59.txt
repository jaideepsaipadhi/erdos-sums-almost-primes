/-- The substitution t = v/a: the gap-five integral over (0,1] scales to
a times the integral over (0,a]. -/
theorem gap_substitution (a : <<R>>) (ha : 0 < a) :
    (<<INT>> t in Set.Ioc (0:<<R>>) 1, (Real.exp (-(a * t)) - 1 + a * t) / t ^ 2)
      = a * <<INT>> v in Set.Ioc (0:<<R>>) a, (Real.exp (-v) - 1 + v) / v ^ 2 := by
  have hane : a <<NE>> 0 := ne_of_gt ha
  have hlhs : (<<INT>> t in Set.Ioc (0:<<R>>) 1, (Real.exp (-(a * t)) - 1 + a * t) / t ^ 2)
      = <<INT>> t in (0:<<R>>)..1, (Real.exp (-(a * t)) - 1 + a * t) / t ^ 2 := by
    rw [intervalIntegral.integral_of_le (by norm_num : (0:<<R>>) <<LE>> 1)]
  have hrhs : (<<INT>> v in Set.Ioc (0:<<R>>) a, (Real.exp (-v) - 1 + v) / v ^ 2)
      = <<INT>> v in (0:<<R>>)..a, (Real.exp (-v) - 1 + v) / v ^ 2 := by
    rw [intervalIntegral.integral_of_le ha.le]
  rw [hlhs, hrhs]
  have hcong : (<<INT>> t in (0:<<R>>)..1, (Real.exp (-(a * t)) - 1 + a * t) / t ^ 2)
      = <<INT>> t in (0:<<R>>)..1,
        a * ((fun v : <<R>> => (Real.exp (-v) - 1 + v) / v ^ 2) (a * t)) := by
    apply intervalIntegral.integral_congr
    intro t ht
    rw [Set.uIcc_of_le (by norm_num : (0:<<R>>) <<LE>> 1)] at ht
    simp only
    rcases eq_or_lt_of_le ht.1 with h0 | h0
    <<DOT>> rw [<<LARR>> h0]
      simp
    <<DOT>> field_simp
      ring
  rw [hcong, intervalIntegral.integral_const_mul,
    intervalIntegral.integral_comp_mul_left _ hane]
  rw [mul_zero, mul_one]
  rw [smul_eq_mul]
  field_simp
