/-- cBracket's series is s-continuous on {s // 1 <= s}. -/
theorem cBracket_series_continuous_s (z : CC) (hz : Norm.norm z LEQ 2) :
    Continuous (fun s : {s : RR // 1 LEQ s} =>
      TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
          / (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC) - 1)
            / (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC) - z))) := by
  apply continuous_tsum
    (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      cBracket_term_continuous_s (p:NN) (by omega) z hz)
    deriv_bound_uniform_summable_five
  intro p s
  exact deriv_clog_term_bound_uniform (p:NN) (by omega) (s:RR) s.2 z hz
