/-- Ein(a) = int_0^a (1 - e^(-v))/v dv (the entire exponential integral).
The integrand extends continuously by 1 at v = 0. -/
noncomputable def Ein (a : <<R>>) : <<R>> :=
  <<INT>> v in Set.Ioc (0:<<R>>) a, (1 - Real.exp (-v)) / v

/-- The Ein integrand is bounded by 1 on positives: 0 <= (1-e^(-v))/v <= 1. -/
theorem ein_integrand_bounds (v : <<R>>) (hv : 0 < v) :
    0 <<LE>> (1 - Real.exp (-v)) / v <<AND>> (1 - Real.exp (-v)) / v <<LE>> 1 := by
  have h1 : Real.exp (-v) <<LE>> 1 := by
    apply Real.exp_le_one_iff.mpr
    linarith
  have h2 : (1:<<R>>) - v <<LE>> Real.exp (-v) := by
    have := Real.add_one_le_exp (-v)
    linarith
  constructor
  <<DOT>> apply div_nonneg _ hv.le
    linarith
  <<DOT>> rw [div_le_one hv]
    linarith

/-- The Ein integrand is integrable on any (0, a]. -/
theorem ein_integrable (a : <<R>>) (ha : 0 < a) :
    MeasureTheory.IntegrableOn (fun v : <<R>> => (1 - Real.exp (-v)) / v)
      (Set.Ioc 0 a) := by
  have hmeas : MeasureTheory.AEStronglyMeasurable
      (fun v : <<R>> => (1 - Real.exp (-v)) / v)
      (MeasureTheory.volume.restrict (Set.Ioc 0 a)) := by
    apply Measurable.aestronglyMeasurable
    fun_prop
  have hbound : MeasureTheory.IntegrableOn (fun _ : <<R>> => (1:<<R>>)) (Set.Ioc 0 a) := by
    apply MeasureTheory.integrableOn_const.mpr
    right
    rw [Real.volume_Ioc]
    exact ENNReal.ofReal_lt_top
  apply MeasureTheory.Integrable.mono hbound hmeas
  filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioc] with v hv
  obtain <<LA>>hv0, _<<RA>> := hv
  obtain <<LA>>hlow, hhigh<<RA>> := ein_integrand_bounds v hv0
  rw [Real.norm_eq_abs, Real.norm_eq_abs, abs_of_nonneg hlow, abs_one]
  exact hhigh
