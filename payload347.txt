/-- Per-term bound for S_2: the j-th term is at most 2^(-j) 2^(-l-1).
Uses 2^(-js) <= 2^(-j) on [1,2] and the exact integral
integral_log_pow_sub_one. -/
theorem S2_term_le (l j : NN) :
    (INTEGRAL s in Set.Ioc (1:RR) 2,
      (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1)
        * (2:RR) ^ (-(j:RR) * s))
      LEQ (2:RR) ^ (-(j:RR)) * (1 / (2:RR) ^ (l + 1)) := by
  have hleft := integrable_S2_integrand l j
  have hright := (integrable_log_pow_sub_one l).const_mul ((2:RR) ^ (-(j:RR)))
  have hmono : (INTEGRAL s in Set.Ioc (1:RR) 2,
        (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1)
          * (2:RR) ^ (-(j:RR) * s))
      LEQ INTEGRAL s in Set.Ioc (1:RR) 2,
        (2:RR) ^ (-(j:RR)) * ((-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1)) := by
    apply MeasureTheory.integral_mono_of_nonneg _ hright
    DOT filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioc] with s hs
      have hs1 : (1:RR) < s := hs.1
      have hL : (0:RR) LEQ (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1) := by
        apply mul_nonneg _ (by linarith)
        apply div_nonneg _ (by positivity)
        apply pow_nonneg
        simp only [neg_nonneg]
        exact Real.log_nonpos (by linarith) (by linarith [hs.2])
      positivity
    DOT filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioc] with s hs
      have hs1 : (1:RR) < s := hs.1
      have hL : (0:RR) LEQ (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1) := by
        apply mul_nonneg _ (by linarith)
        apply div_nonneg _ (by positivity)
        apply pow_nonneg
        simp only [neg_nonneg]
        exact Real.log_nonpos (by linarith) (by linarith [hs.2])
      have hmonoexp : (2:RR) ^ (-(j:RR) * s) LEQ (2:RR) ^ (-(j:RR)) := by
        apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
        have hj : (0:RR) LEQ (j:RR) := by positivity
        nlinarith [hs1]
      calc (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1) * (2:RR) ^ (-(j:RR) * s)
          LEQ (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1) * (2:RR) ^ (-(j:RR)) :=
            mul_le_mul_of_nonneg_left hmonoexp hL
        _ = (2:RR) ^ (-(j:RR))
            * ((-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1)) := by ring
  rw [MeasureTheory.integral_const_mul, integral_log_pow_sub_one] at hmono
  exact hmono
