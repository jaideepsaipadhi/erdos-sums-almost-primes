/-- C' is continuous in theta on the contour: the kappa-part is constant in
theta apart from the z factor, the p=3 term is clog2_term_continuous_theta at
p = 3, and the tail is clog2_series_continuous_theta. -/
theorem c_bracket_deriv_continuous_theta (s : RR) (hs : 1 LEQ s) (fps : RR) :
    Continuous (fun theta : RR =>
      circleMap 0 (2:RR) theta * (((fps + -((Real.log 2) ^ 2 * (2:RR) ^ s)
            / ((2:RR) ^ s - 1) ^ 2 : RR)) : CC)
        + ((circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
              * ((Real.log ((3:NN):RR) : RR) : CC) ^ 2 * (((((3:NN):RR)) ^ s : RR) : CC)
              * ((((((3:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
                + ((((((3:NN):RR)) ^ s : RR) : CC) - 1))
            / (((((((3:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
              * (((((((3:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta) ^ 2))))
          + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
              (circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
                  * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2
                  * (((((p:NN):RR)) ^ s : RR) : CC)
                  * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
                    + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))
                / (((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
                  * (((((((p:NN):RR)) ^ s : RR) : CC)
                    - circleMap 0 (2:RR) theta) ^ 2)))))) := by
  have hcm : Continuous (circleMap 0 (2:RR)) := continuous_circleMap 0 _
  apply Continuous.add (hcm.mul continuous_const)
  apply Continuous.add
  DOT exact clog2_term_continuous_theta 3 (le_refl 3) s hs
  DOT exact clog2_series_continuous_theta s hs
