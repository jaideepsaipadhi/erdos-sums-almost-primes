/-- Sm's integrand is integrable on (1,2].
Transfers from integrable_Jint_integrand through integrableOn_Ioc_subst.
Without this, Sm would be Lean's junk value for a non-integrable integral, and
Am, deltaM and Theorem 4.1 would all be vacuous. -/
theorem integrable_Sm_integrand (l j : NN) :
    MeasureTheory.IntegrableOn
      (fun s : RR => (-Real.log (s - 1)) ^ l / (Nat.factorial l) * (2:RR) ^ (-(j:RR) * s))
      (Set.Ioc (1:RR) 2) := by
  apply integrableOn_Ioc_subst
  have hJ := integrable_Jint_integrand (j:RR) (by positivity) l
  have hconst := hJ.const_mul (((Nat.factorial l : RR))INV * (2:RR) ^ (-(j:RR)))
  apply hconst.congr_fun _ measurableSet_Ioi
  intro u hu
  have hexp : (0:RR) < Real.exp (-u) := Real.exp_pos _
  simp only [add_sub_cancel_left, Real.log_exp, neg_neg]
  rw [show (-(j:RR)) * (1 + Real.exp (-u)) = -(j:RR) + -(j:RR) * Real.exp (-u) by ring]
  rw [Real.rpow_add (by norm_num : (0:RR) < 2)]
  field_simp
  ring
