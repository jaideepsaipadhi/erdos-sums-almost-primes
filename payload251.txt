/-- Per-term expansion for the frozen integral, for i <= k. -/
theorem Ikfrozen_term (k i : NN) (hi : i LEQ k) :
    (cCoeff i).re * Sm (k - i)
      = 2 * (cCoeff i).re
        - (Real.log 2 / 4) * ((2:RR)^i * (cCoeff i).re * ((k:RR) - (i:RR))^2) / (2:RR)^k
        + ((2:RR)^i * (cCoeff i).re * deltaM (k - i)) / (2:RR)^k := by
  have hpow : (2:RR)^(k-i) * (2:RR)^i = (2:RR)^k := by
    rw [LARR pow_add, Nat.sub_add_cancel hi]
  have hcast : ((k - i : NN) : RR) = (k:RR) - (i:RR) := Nat.cast_sub hi
  have h1 : ((2:RR)^(k-i)) NEQ 0 := by positivity
  have h2 : ((2:RR)^i) NEQ 0 := by positivity
  rw [Sm_eq_Am_div, Am_eq_delta, hcast, LARR hpow]
  field_simp
  ring
