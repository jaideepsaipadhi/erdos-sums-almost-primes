/-- Finite-interval by-parts for the gap-five integrand, 0 < b <= a. -/
theorem gap_parts_finite (b a : <<R>>) (hb : 0 < b) (hba : b <<LE>> a) :
    (<<INT>> v in b..a, ((Real.exp (-v) - 1 + v) / v ^ 2 - (1 - Real.exp (-v)) / v))
      = -((Real.exp (-a) - 1 + a) / a) + (Real.exp (-b) - 1 + b) / b := by
  have hderiv : <<ALL>> x <<IN>> Set.uIcc b a,
      HasDerivAt (fun t : <<R>> => -((Real.exp (-t) - 1 + t) / t))
        ((Real.exp (-x) - 1 + x) / x ^ 2 - (1 - Real.exp (-x)) / x) x := by
    intro x hx
    rw [Set.uIcc_of_le hba] at hx
    exact hasDerivAt_gapAntideriv x (by linarith [hx.1])
  have hint : IntervalIntegrable
      (fun v : <<R>> => (Real.exp (-v) - 1 + v) / v ^ 2 - (1 - Real.exp (-v)) / v)
      MeasureTheory.volume b a := by
    apply ContinuousOn.intervalIntegrable
    rw [Set.uIcc_of_le hba]
    apply ContinuousOn.sub
    <<DOT>> apply ContinuousOn.div
      <<DOT>> exact (((Real.continuous_exp.comp continuous_neg).sub continuous_const).add
          continuous_id).continuousOn
      <<DOT>> exact (continuous_pow 2).continuousOn
      <<DOT>> intro x hx
        have : (0:<<R>>) < x := by linarith [hx.1]
        positivity
    <<DOT>> apply ContinuousOn.div
      <<DOT>> exact (continuous_const.sub (Real.continuous_exp.comp continuous_neg)).continuousOn
      <<DOT>> exact continuousOn_id
      <<DOT>> intro x hx
        exact ne_of_gt (by linarith [hx.1])
  rw [intervalIntegral.integral_eq_sub_of_hasDerivAt hderiv hint]
  ring
