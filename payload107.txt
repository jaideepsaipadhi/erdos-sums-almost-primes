/-- A1 step 4b: each Euler factor is at least 1. -/
theorem zOmega_factor_ge_one (s : RR) (hs : 1 < s) (z : CC)
    (hz : Norm.norm z < (2:RR) ^ s) (p : Nat.Primes) :
    (1:RR) LEQ (1 - Norm.norm z * (((p:NN):RR) ^ (-s)))INV := by
  have hlt : Norm.norm z * (((p:NN):RR) ^ (-s)) < 1 := by
    have hp2 : (2:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.two_le
    have h2s : (0:RR) < (2:RR) ^ s := Real.rpow_pos_of_pos (by norm_num) s
    have hps : ((p:NN):RR) ^ (-s) LEQ (2:RR) ^ (-s) := by
      rw [Real.rpow_neg (by linarith : (0:RR) LEQ ((p:NN):RR)),
        Real.rpow_neg (by norm_num : (0:RR) LEQ (2:RR))]
      gcongr
    have hznn : (0:RR) LEQ Norm.norm z := norm_nonneg z
    have hinvpos : (0:RR) < ((2:RR) ^ s)INV := inv_pos.mpr h2s
    calc Norm.norm z * (((p:NN):RR) ^ (-s)) LEQ Norm.norm z * ((2:RR) ^ (-s)) :=
          mul_le_mul_of_nonneg_left hps hznn
      _ = Norm.norm z * ((2:RR) ^ s)INV := by rw [Real.rpow_neg (by norm_num : (0:RR) LEQ (2:RR))]
      _ < 1 := by
          have hkey := mul_lt_mul_of_pos_right hz hinvpos
          rwa [mul_inv_cancelZ (ne_of_gt h2s)] at hkey
  have hnn : (0:RR) LEQ Norm.norm z * (((p:NN):RR) ^ (-s)) := by positivity
  rw [le_inv_comm0 (by norm_num) (by linarith)]
  linarith
