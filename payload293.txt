/-- s-free dominator for the correction series at p >= 5, for the
s-continuity argument. -/
theorem correction_bound_uniform_summable_five :
    Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      18 * (((p:NN):RR)) ^ (-(2:RR))) := by
  have h2 : (-(2:RR)) < -1 := by norm_num
  have hbase : Summable (fun n : NN => (n:RR) ^ (-(2:RR))) :=
    Real.summable_nat_rpow.mpr h2
  exact (hbase.subtype _).mul_left 18

/-- The correction bound, s-free: 18 p^(-2s) <= 18 p^(-2) for s >= 1. -/
theorem correction_bound_sfree (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : CC) (hz : Norm.norm z LEQ 2) :
    Norm.norm ((-Complex.log (1 - z * (((((p:NN):RR)) ^ (-s) : RR) : CC)))
        + z * ((Real.log (1 - (((p:NN):RR)) ^ (-s)) : RR) : CC))
      LEQ 18 * (((p:NN):RR)) ^ (-(2:RR)) := by
  have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp
  have hbase := prime_correction_bound_complex p hp s hs z hz
  have hmono : (((p:NN):RR)) ^ (-2*s) LEQ (((p:NN):RR)) ^ (-(2:RR)) := by
    apply Real.rpow_le_rpow_of_exponent_le (by linarith)
    linarith
  linarith
