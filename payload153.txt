/-- Pointwise factorisation: the plain Euler factor splits into the
cpow-negative factor times the regularised factor. -/
theorem euler_factor_split (a : RR) (ha : 0 < 1 - a) (z w : CC) :
    (1 - w)INV = ((((1 - a : RR)) : CC)) ^ (-z)
      * ((1 - w)INV * (((1 - a : RR)) : CC) ^ z) := by
  have hane : (((1 - a : RR)) : CC) NEQ 0 := by
    simp only [ne_eq, Complex.ofReal_eq_zero]
    linarith
  rw [Complex.cpow_neg]
  field_simp
