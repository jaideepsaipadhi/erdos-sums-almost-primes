/-- Complex analogue of deriv2_prime_term_bound: the r=2 per-prime derivative
is bounded by 200 log^2 p p^(-2), s-free for s >= 1.
Same crude route: ||z-1|| <= 3, ||(p^s-z)+(p^s-1)|| <= 3p^s,
||p^s-1|| >= (2/3)p^s, ||p^s-z|| >= (1/3)p^s. -/
theorem deriv2_clog_term_bound (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : CC) (hz : Norm.norm z LEQ 2) :
    Norm.norm ((z - 1) * ((Real.log (p:RR) : RR) : CC) ^ 2 * ((((p:RR)) ^ s : RR) : CC)
        * ((((((p:RR)) ^ s : RR) : CC) - z) + (((((p:RR)) ^ s : RR) : CC) - 1))
        / ((((((p:RR)) ^ s : RR) : CC) - 1) ^ 2
          * ((((((p:RR)) ^ s : RR) : CC) - z) ^ 2)))
      LEQ 200 * (Real.log (p:RR)) ^ 2 * (((p:RR)) ^ (-(2:RR))) := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < (p:RR) := by linarith
  have hps : (3:RR) LEQ ((p:RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ ((p:RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hps0 : (0:RR) < ((p:RR)) ^ s := by linarith
  have hpsC : Norm.norm ((((p:RR)) ^ s : RR) : CC) = ((p:RR)) ^ s := by
    rw [Complex.norm_real, Real.norm_eq_abs, abs_of_pos hps0]
  have hlog0 : (0:RR) LEQ (Real.log (p:RR)) ^ 2 := sq_nonneg _
  have hlogC : Norm.norm (((Real.log (p:RR) : RR) : CC) ^ 2) = (Real.log (p:RR)) ^ 2 := by
    rw [norm_pow, Complex.norm_real, Real.norm_eq_abs, LARR abs_pow,
      abs_of_nonneg (sq_nonneg _)]
  have hz1 : Norm.norm (z - 1) LEQ 3 := by
    calc Norm.norm (z - 1) LEQ Norm.norm z + Norm.norm (1:CC) := norm_sub_le z 1
      _ LEQ 2 + 1 := by simp only [norm_one]; linarith
      _ = 3 := by norm_num
  have hsum3 : Norm.norm ((((((p:RR)) ^ s : RR) : CC) - z)
      + (((((p:RR)) ^ s : RR) : CC) - 1)) LEQ 3 * ((p:RR)) ^ s := by
    calc Norm.norm ((((((p:RR)) ^ s : RR) : CC) - z) + (((((p:RR)) ^ s : RR) : CC) - 1))
        LEQ Norm.norm ((((((p:RR)) ^ s : RR) : CC) - z))
          + Norm.norm ((((((p:RR)) ^ s : RR) : CC) - 1)) := norm_add_le _ _
      _ LEQ (((p:RR)) ^ s + 2) + (((p:RR)) ^ s + 1) := by
          have ha : Norm.norm ((((((p:RR)) ^ s : RR) : CC) - z)) LEQ ((p:RR)) ^ s + 2 := by
            calc Norm.norm ((((((p:RR)) ^ s : RR) : CC) - z))
                LEQ Norm.norm ((((p:RR)) ^ s : RR) : CC) + Norm.norm z := norm_sub_le _ _
              _ LEQ ((p:RR)) ^ s + 2 := by rw [hpsC]; linarith
          have hb : Norm.norm ((((((p:RR)) ^ s : RR) : CC) - 1)) LEQ ((p:RR)) ^ s + 1 := by
            calc Norm.norm ((((((p:RR)) ^ s : RR) : CC) - 1))
                LEQ Norm.norm ((((p:RR)) ^ s : RR) : CC) + Norm.norm (1:CC) := norm_sub_le _ _
              _ LEQ ((p:RR)) ^ s + 1 := by rw [hpsC]; simp only [norm_one]
          linarith
      _ LEQ 3 * ((p:RR)) ^ s := by linarith
  sorry
