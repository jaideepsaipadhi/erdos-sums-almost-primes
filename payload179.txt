/-- Summability of the derivative bound family. -/
theorem deriv_bound_summable (s : RR) (hs : 1 < s) (j : NN)
    (r0 r : RR) (hr01 : 1 LEQ r0) (hr0r : r0 < r)
    (hr0 : 0 LEQ r) (hr3 : r * ((3:RR) ^ (-s)) LEQ 1/2) (hr2 : r < (2:RR) ^ s) :
    Summable (fun n : NN =>
      ((bigOmega n : NN) : RR) ^ j * r0 ^ (bigOmega n - j) * ((n:RR) ^ (-s))) := by
  have hbase := omega_pow_summable_radius s hs j r0 r (by linarith) hr0r hr0 hr3 hr2
  apply Summable.of_nonneg_of_le _ _ hbase
  DOT intro n
    positivity
  DOT intro n
    apply mul_le_mul_of_nonneg_right _ (by positivity)
    apply mul_le_mul_of_nonneg_left _ (by positivity)
    apply pow_le_pow_right_of_le_one_or_le hr01
    exact Nat.sub_le _ _
