/-- The manuscript's c1* = log2 (log log2 + gamma - 1 - 1/4) = -0.720386... -/
noncomputable def cStarOne : <<R>> :=
  Real.log 2 * (Real.log (Real.log 2) + Real.eulerMascheroniConstant - 1 - 1/4)

/-- MANUSCRIPT THEOREM 4.1 (upper bound, manuscript form):
delta_m <= (log 2) m log m + c1* m, for m >= 1. -/
theorem deltaM_upper_bound_manuscript (m : <<N>>) (hm : 1 <<LE>> m) :
    deltaM m <<LE>> Real.log 2 * (m : <<R>>) * Real.log (m : <<R>>) + cStarOne * (m : <<R>>) := by
  have hl2 : (0:<<R>>) < Real.log 2 := Real.log_pos (by norm_num)
  have hmR : (1:<<R>>) <<LE>> ((m:<<N>>):<<R>>) := by exact_mod_cast hm
  have hmpos : (0:<<R>>) < ((m:<<N>>):<<R>>) := by linarith
  have hub := deltaM_upper_bound m hm
  have hlogsplit : Real.log (((m:<<N>>):<<R>>) * Real.log 2)
      = Real.log ((m:<<N>>):<<R>>) + Real.log (Real.log 2) := by
    rw [Real.log_mul (ne_of_gt hmpos) (ne_of_gt hl2)]
  rw [hlogsplit] at hub
  unfold cStarOne
  nlinarith [hub, hl2, hmpos]
