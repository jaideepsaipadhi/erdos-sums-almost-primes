theorem hasDerivAt_EinTailFun (a : <<R>>) (ha : 0 < a) :
    HasDerivAt EinTailFun (-(Real.exp (-a) / a)) a := by
  have hcont : ContinuousAt (fun v : <<R>> => Real.exp (-v) / v) a := by
    apply ContinuousAt.div
    <<DOT>> exact (Real.continuous_exp.comp continuous_neg).continuousAt
    <<DOT>> exact continuousAt_id
    <<DOT>> exact ne_of_gt ha
  have hmeas : StronglyMeasurableAtFilter
      (fun v : <<R>> => Real.exp (-v) / v) (nhds a) := by
    refine <<LA>>Set.univ, Filter.univ_mem, ?_<<RA>>
    apply Measurable.aestronglyMeasurable
    fun_prop
  have hint : IntervalIntegrable (fun v : <<R>> => Real.exp (-v) / v)
      MeasureTheory.volume 1 a := by
    rcases le_total (1:<<R>>) a with h1 | h1
    <<DOT>> rw [intervalIntegrable_iff_integrableOn_Ioc_of_le h1]
      exact ((exp_div_integrableOn_Ici 1 (by norm_num)).mono
        (fun x hx => Set.mem_Ici.mpr (le_of_lt hx.1)) le_rfl)
    <<DOT>> rw [IntervalIntegrable.symm_iff, intervalIntegrable_iff_integrableOn_Ioc_of_le h1]
      exact ((exp_div_integrableOn_Ici a ha).mono
        (fun x hx => Set.mem_Ici.mpr (le_of_lt hx.1)) le_rfl)
  have hFTC := intervalIntegral.integral_hasDerivAt_right hint hmeas hcont
  have hEq : EinTailFun =<<FEQ>>[nhds a]
      fun t : <<R>> => EinTailFun 1 - <<INT>> v in (1:<<R>>)..t, Real.exp (-v) / v := by
    filter_upwards [Ioi_mem_nhds ha] with t ht
    exact EinTailFun_split t ht
  have hderiv := (hasDerivAt_const a (EinTailFun 1)).sub hFTC
  rw [zero_sub] at hderiv
  exact hderiv.congr_of_eventuallyEq hEq
