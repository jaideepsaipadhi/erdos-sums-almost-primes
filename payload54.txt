/-- The gap-five boundary term vanishes at 0: (e^(-b) - 1 + b)/b -> 0. -/
theorem gapBoundary_tendsto_zero :
    Filter.Tendsto (fun b : <<R>> => (Real.exp (-b) - 1 + b) / b)
      (nhdsWithin 0 (Set.Ioi 0)) (nhds 0) := by
  have hmaj : Filter.Tendsto (fun b : <<R>> => b) (nhdsWithin 0 (Set.Ioi 0)) (nhds 0) :=
    Filter.tendsto_id.mono_left nhdsWithin_le_nhds
  apply squeeze_zero_norm' _ hmaj
  filter_upwards [self_mem_nhdsWithin] with b hb
  have hb0 : (0:<<R>>) < b := hb
  have hupper : Real.exp (-b) - 1 + b <<LE>> b * b := by
    nlinarith [Real.add_one_le_exp (-b), Real.exp_pos (-b),
      Real.add_one_le_exp b, sq_nonneg b, sq_nonneg (Real.exp (-b) - 1)]
  have hlower : (0:<<R>>) <<LE>> Real.exp (-b) - 1 + b := by
    nlinarith [Real.add_one_le_exp (-b)]
  rw [Real.norm_eq_abs, Real.norm_eq_abs, abs_of_nonneg hb0.le,
    abs_of_nonneg (by positivity : (0:<<R>>) <<LE>> (Real.exp (-b) - 1 + b) / b)]
  rw [div_le_iff<<S0>> hb0]
  nlinarith [hupper]
