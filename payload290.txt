/-- Per-term continuity in theta on the contour |z| = 2.
Both denominators are nonvanishing there: p^s - 1 >= 2 is constant in theta,
and |p^s - z| >= p^s - 2 >= 1 since |z| = 2 and p^s >= 3. -/
theorem clog_term_continuous_theta (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s) :
    Continuous (fun theta : RR =>
      -(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
          * ((Real.log ((p:NN):RR) : RR) : CC))
        / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
          / (((((((p:NN):RR)) ^ s : RR)) : CC) - circleMap 0 (2:RR) theta)) := by
  have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp
  have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
  have hps : (3:RR) LEQ (((p:NN):RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ (((p:NN):RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hcm : Continuous (circleMap 0 (2:RR)) := continuous_circleMap 0 _
  have hnorm : FORALL theta : RR, Norm.norm (circleMap 0 (2:RR) theta) = 2 := by
    intro theta
    simp [circleMap]
  have hd1 : (((((((p:NN):RR)) ^ s : RR)) : CC) - 1) NEQ 0 := by
    intro hcon
    rw [sub_eq_zero] at hcon
    have : ((((p:NN):RR)) ^ s : RR) = 1 := by exact_mod_cast hcon
    linarith
  have hd2 : FORALL theta : RR,
      (((((((p:NN):RR)) ^ s : RR)) : CC) - circleMap 0 (2:RR) theta) NEQ 0 := by
    intro theta hcon
    rw [sub_eq_zero] at hcon
    have hn := hnorm theta
    rw [LARR hcon, Complex.norm_real, Real.norm_eq_abs,
      abs_of_pos (by linarith : (0:RR) < (((p:NN):RR)) ^ s)] at hn
    linarith
  apply Continuous.div _ continuous_const (fun theta => hd2 theta)
  apply Continuous.div _ continuous_const (fun _ => hd1)
  apply Continuous.neg
  apply Continuous.mul _ continuous_const
  exact hcm.mul (hcm.sub continuous_const)
