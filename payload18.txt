/-- Closed form: sum_{i <= m} (m - i) = m(m+1)/2, over the reals. -/
theorem sum_range_sub_cast (m : <<N>>) :
    <<SUM>> i <<IN>> Finset.range (m + 1), ((m - i : <<N>>) : <<R>>)
      = (m : <<R>>) * ((m : <<R>>) + 1) / 2 := by
  induction m with
  | zero => simp
  | succ M ih =>
    rw [Finset.sum_range_succ']
    have hcong : <<ALL>> i <<IN>> Finset.range (M + 1),
        (((M + 1 - (i + 1) : <<N>>)) : <<R>>) = ((M - i : <<N>>) : <<R>>) := by
      intro i _
      congr 1
      omega
    rw [Finset.sum_congr rfl hcong, ih]
    push_cast
    ring

/-- Identity (11): delta_m = -1 - (log 2 / 4) m + sum_{i <= m} 2^i beta-circle(i, m-i). -/
theorem deltaM_identity (m : <<N>>) :
    deltaM m = -1 - (Real.log 2 / 4) * (m : <<R>>)
      + <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i (m - i) := by
  have hterm : <<ALL>> i <<IN>> Finset.range (m + 1),
      (2:<<R>>) ^ i / (Nat.factorial i) * Jint m i
        = (2:<<R>>) ^ i * betaCirc i (m - i) + (2:<<R>>) ^ i
          - ((m - i : <<N>>) : <<R>>) * Real.log 2 / 2 := by
    intro i _
    rw [Jint_split m i]
    have hfacne : ((Nat.factorial i : <<N>>) : <<R>>) <<NE>> 0 :=
      Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
    have h2 : (2:<<R>>) ^ (i + 1) = 2 ^ i * 2 := by ring
    field_simp [h2]
    ring
  have hsum : <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i / (Nat.factorial i) * Jint m i
      = (<<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i * betaCirc i (m - i))
        + (<<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i)
        - (<<SUM>> i <<IN>> Finset.range (m + 1), ((m - i : <<N>>) : <<R>>) * Real.log 2 / 2) := by
    rw [<<LARR>> Finset.sum_add_distrib, <<LARR>> Finset.sum_sub_distrib]
    exact Finset.sum_congr rfl hterm
  have hgeom : <<SUM>> i <<IN>> Finset.range (m + 1), (2:<<R>>) ^ i = 2 ^ (m + 1) - 1 := by
    have := Finset.geom_sum_eq (by norm_num : (2:<<R>>) <<NE>> 1) (m + 1)
    rw [this]
    ring
  have hlin : <<SUM>> i <<IN>> Finset.range (m + 1), ((m - i : <<N>>) : <<R>>) * Real.log 2 / 2
      = (Real.log 2 / 2) * ((m : <<R>>) * ((m : <<R>>) + 1) / 2) := by
    rw [<<LARR>> sum_range_sub_cast m, Finset.mul_sum]
    apply Finset.sum_congr rfl
    intro i _
    ring
  unfold deltaM
  rw [Am_eq_sum_Jint m, hsum, hgeom, hlin]
  ring
