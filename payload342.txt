/-- C is continuous in theta on the contour: kappa-part, p=3 term, and the
r=1 tail series.  The r=1 analogue of c_bracket_deriv_continuous_theta. -/
theorem c_bracket_continuous_theta (s : RR) (hs : 1 LEQ s) (fs : RR) :
    Continuous (fun theta : RR =>
      circleMap 0 (2:RR) theta * (((fs + Real.log 2 / ((2:RR) ^ s - 1) : RR)) : CC)
        + ((-(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
              * ((Real.log ((3:NN):RR) : RR) : CC))
            / ((((((3:NN):RR)) ^ s : RR) : CC) - 1)
              / ((((((3:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta))
          + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
              (-(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
                  * ((Real.log ((p:NN):RR) : RR) : CC))
                / ((((((p:NN):RR)) ^ s : RR) : CC) - 1)
                  / ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta)))) := by
  have hcm : Continuous (circleMap 0 (2:RR)) := continuous_circleMap 0 _
  apply Continuous.add (hcm.mul continuous_const)
  apply Continuous.add
  DOT exact clog_term_continuous_theta 3 (le_refl 3) s hs
  DOT exact clog_series_continuous_theta s hs
