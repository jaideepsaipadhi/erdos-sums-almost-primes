theorem log_tprod_real {f : NN ARROW RR} (hpos : FORALL i, 0 < f i)
    (hmul : Multipliable f) (hlog : Summable (fun i => Real.log (f i))) :
    Real.log (TPRODN i : NN, f i) = TSUMN i : NN, Real.log (f i) := by
  have hpart : FORALL n : NN, PRODF i IN Finset.range n, f i
      = Real.exp (SUMF i IN Finset.range n, Real.log (f i)) := by
    intro n
    rw [Real.exp_sum]
    apply Finset.prod_congr rfl
    intro i _
    rw [Real.exp_log (hpos i)]
  have htend1 : Filter.Tendsto (fun n : NN => PRODF i IN Finset.range n, f i)
      Filter.atTop (nhds (TPRODN i : NN, f i)) := hmul.hasProd.tendsto_prod_nat
  have htend2 : Filter.Tendsto (fun n : NN => SUMF i IN Finset.range n, Real.log (f i))
      Filter.atTop (nhds (TSUMN i : NN, Real.log (f i))) := hlog.hasSum.tendsto_sum_nat
  have htend3 : Filter.Tendsto (fun n : NN => PRODF i IN Finset.range n, f i)
      Filter.atTop (nhds (Real.exp (TSUMN i : NN, Real.log (f i)))) := by
    have hcomp := (Real.continuous_exp.tendsto _).comp htend2
    apply hcomp.congr
    intro n
    rw [Function.comp_apply, LARR hpart n]
  have heq : (TPRODN i : NN, f i) = Real.exp (TSUMN i : NN, Real.log (f i)) :=
    tendsto_nhds_unique htend1 htend3
  rw [heq, Real.log_exp]
