/-- kappa(s) = (s-1) cFac s tends to 1/2 as s -> 1 from above.
This is the limit that section 3.5's mean value step needs, and the reason
G2(1,z) = 2^(-z) prod_{p>2}(...) has the 2^(-z) prefactor. -/
theorem kappaFac_tendsto_half :
    Filter.Tendsto kappaFac (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds (1/2 : RR)) := by
  rw [LARR Filter.tendsto_ofReal_iff]
  have hres : Filter.Tendsto (fun s : RR => ((s : CC) - 1) * riemannZeta (s : CC))
      (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds 1) :=
    zeta_residue_halfplane.comp ofReal_tendsto_halfplane
  have htwo : Filter.Tendsto (fun s : RR => 1 - ((2:CC)) ^ (-(s:CC)))
      (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds (1/2 : CC)) := by
    sorry
  have hprod := hres.mul htwo
  simp only [one_mul] at hprod
  apply hprod.congr'
  filter_upwards [self_mem_nhdsWithin] with s hs
  have hs1 : (1:RR) < s := hs
  unfold kappaFac
  push_cast
  rw [cFac_eq_zeta s hs1]
  ring
