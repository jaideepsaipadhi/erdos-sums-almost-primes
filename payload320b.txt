/-- The inner sum of (10) is constant in j: each term is 2^(-(k-i)-1),
so the sum over j <= k-i is (k-i+1) * 2^(-(k-i)-1). -/
theorem inner_sum_const (k i : NN) (hik : i LEQ k) (b : RR) :
    SUMF j IN Finset.range (k - i + 1),
        b * (2:RR) ^ (-(j:RR)) * (1 / (2:RR) ^ ((k - i - j) + 1))
      = ((k : RR) - (i : RR) + 1) * (b * (1 / (2:RR) ^ ((k - i) + 1))) := by
  have hterm : FORALL j IN Finset.range (k - i + 1),
      b * (2:RR) ^ (-(j:RR)) * (1 / (2:RR) ^ ((k - i - j) + 1))
        = b * (1 / (2:RR) ^ ((k - i) + 1)) := by
    intro j hj
    have hjk : j LEQ k - i := Nat.lt_succ_iff.mp (Finset.mem_range.mp hj)
    have hpow : (2:RR) ^ (-(j:RR)) = 1 / (2:RR) ^ (j:NN) := by
      rw [Real.rpow_neg (by norm_num), Real.rpow_natCast]
      ring
    rw [hpow]
    have hsplit : (2:RR) ^ ((k - i) + 1) = (2:RR) ^ (j:NN) * (2:RR) ^ ((k - i - j) + 1) := by
      rw [LARR pow_add]
      congr 1
      omega
    rw [hsplit]
    field_simp
  rw [Finset.sum_congr rfl hterm, Finset.sum_const, Finset.card_range]
  rw [nsmul_eq_mul]
  congr 1
  have : ((k - i + 1 : NN) : RR) = (k : RR) - (i : RR) + 1 := by
    rw [Nat.cast_add, Nat.cast_sub hik]
    norm_num
  exact this
