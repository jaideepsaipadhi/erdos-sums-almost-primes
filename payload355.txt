/-- Per-term s-continuity for cBracket's series on {s // 1 <= s}.
The denominators are nonvanishing there: p^s - 1 >= 2 and |p^s - z| >= 1
since p^s >= 3 and ||z|| <= 2. -/
theorem cBracket_term_continuous_s (p : NN) (hp : 3 LEQ p) (z : CC)
    (hz : Norm.norm z LEQ 2) :
    Continuous (fun s : {s : RR // 1 LEQ s} =>
      -(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
        / (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC) - 1)
          / (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC) - z)) := by
  have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp
  have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
  have hsub : Continuous (fun s : {s : RR // 1 LEQ s} => (s:RR)) := continuous_subtype_val
  have hrpow : Continuous (fun s : {s : RR // 1 LEQ s} => (((p:NN):RR)) ^ (s:RR)) :=
    (continuous_const_rpow (ne_of_gt hp0)).comp hsub
  have hps : FORALL s : {s : RR // 1 LEQ s}, (3:RR) LEQ (((p:NN):RR)) ^ (s:RR) := by
    intro s
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ (s:RR) := Real.rpow_le_rpow_of_exponent_le (by norm_num) s.2
      _ LEQ (((p:NN):RR)) ^ (s:RR) := Real.rpow_le_rpow (by norm_num) hp3 (by linarith [s.2])
  have hcast : Continuous (fun s : {s : RR // 1 LEQ s} =>
      (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC))) := Complex.continuous_ofReal.comp hrpow
  have hd1 : FORALL s : {s : RR // 1 LEQ s},
      (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC)) - 1 NEQ 0 := by
    intro s hcon
    rw [sub_eq_zero] at hcon
    have : ((((p:NN):RR)) ^ (s:RR) : RR) = 1 := by exact_mod_cast hcon
    linarith [hps s]
  have hd2 : FORALL s : {s : RR // 1 LEQ s},
      (((((((p:NN):RR)) ^ (s:RR) : RR)) : CC)) - z NEQ 0 := by
    intro s hcon
    rw [sub_eq_zero] at hcon
    rw [LARR hcon, Complex.norm_real, Real.norm_eq_abs,
      abs_of_pos (by linarith [hps s] : (0:RR) < (((p:NN):RR)) ^ (s:RR))] at hz
    linarith [hps s]
  exact ((continuous_const.div (hcast.sub continuous_const) hd1).div
    (hcast.sub continuous_const) hd2)
