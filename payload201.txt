/-- Factor A in the manuscript normalisation: (s-1)^(-z) is ContDiff of all orders. -/
theorem contDiffAt_factorAmanu (s : RR) (hs1 : 0 < s - 1) (k : NN) (z : CC) :
    ContDiffAt CC k (fun w : CC => ((s - 1 : RR) : CC) ^ (-w)) z := by
  have hrewrite : (fun w : CC => ((s - 1 : RR) : CC) ^ (-w))
      = fun w : CC => Complex.exp ((-w) * ((Real.log (s - 1) : RR) : CC)) := by
    funext w
    exact ofReal_cpow_eq_exp (s - 1) hs1 (-w)
  rw [hrewrite]
  apply ContDiff.contDiffAt
  apply ContDiff.cexp
  exact (contDiff_id.neg).mul contDiff_const
