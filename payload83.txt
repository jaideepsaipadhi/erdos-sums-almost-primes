/-- G2C is ContDiff at 0 (from holomorphy on ball 0 2). -/
theorem contDiffAt_factorC (s : <<R>>) (hs : 1 <<LE>> s) (k : <<N>>) :
    ContDiffAt <<C>> k (fun w : <<C>> => G2C s w) 0 := by
  have hball : Metric.ball (0:<<C>>) 2 <<IN>> nhds (0:<<C>>) :=
    Metric.ball_mem_nhds _ (by norm_num)
  have han : AnalyticAt <<C>> (fun w : <<C>> => G2C s w) 0 :=
    (G2C_differentiableOn s hs).analyticAt hball
  exact han.contDiffAt

/-- Fbig factors as (A * B) * C as functions. -/
theorem Fbig_eq_prod (s : <<R>>) :
    (fun z : <<C>> => Fbig s z)
      = ((fun z : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-z))
          * (fun z : <<C>> => (1 - z / ((2:<<C>>) ^ (s:<<C>>)))<<INV>>))
        * (fun z : <<C>> => G2C s z) := by
  funext z
  unfold Fbig
  simp only [Pi.mul_apply]
