theorem Fk_expansion (k : <<N>>) (s : <<R>>) (hs : 1 < s) (hs2 : s <<LE>> 2) :
    Fk k s = <<SUM>> p <<IN>> Finset.range (k + 1),
      <<SUM>> q <<IN>> Finset.range (k - p + 1),
        (gCoeff q s).re * (2:<<R>>) ^ (-((p:<<R>>)) * s)
          * ((-Real.log (s - 1)) ^ (k - p - q) / (Nat.factorial (k - p - q))) := by
  have hC := (Fk_eq_coeff_Fbig k s hs hs2).trans (coeff_Fbig_eq_convolution k s hs hs2)
  have hre : ((Fk k s : <<R>>) : <<C>>).re = Fk k s := Complex.ofReal_re _
  rw [<<LARR>> hre, hC]
  rw [Complex.re_sum]
  apply Finset.sum_congr rfl
  intro p _
  rw [Complex.re_sum]
  apply Finset.sum_congr rfl
  intro q _
  rw [Complex.mul_re, Complex.mul_re]
  simp only [Complex.ofReal_re, Complex.ofReal_im]
  ring
