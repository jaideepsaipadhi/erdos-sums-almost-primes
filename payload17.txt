/-- The J-split (manuscript Thm 4.1 engine): J(i) decomposes exactly into the
Bfun piece, the Gamma piece, and the log2-weighted scaled piece. -/
theorem Jint_split (m i : <<N>>) :
    Jint m i = (Nat.factorial i : <<R>>) * betaCirc i (m - i)
      + (Nat.factorial i : <<R>>)
      - ((m - i : <<N>>) : <<R>>) * Real.log 2 * ((Nat.factorial i : <<R>>) / 2 ^ (i + 1)) := by
  have hfacne : ((Nat.factorial i : <<N>>) : <<R>>) <<NE>> 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
  have hpt : <<ALL>> u : <<R>>,
      u ^ i * Real.exp (-u) * (2:<<R>>) ^ (-(((m - i : <<N>>)):<<R>>) * Real.exp (-u))
        = u ^ i * Real.exp (-u) * Bfun (m - i) u
          + u ^ i * Real.exp (-u)
          - ((m - i : <<N>>) : <<R>>) * Real.log 2 * (u ^ i * Real.exp (-(2 * u))) := by
    intro u
    have hexp : Real.exp (-u) * Real.exp (-u) = Real.exp (-(2 * u)) := by
      rw [<<LARR>> Real.exp_add]
      ring_nf
    unfold Bfun
    linear_combination ((m - i : <<N>>) : <<R>>) * Real.log 2 * u ^ i * hexp
  have hI1 := integrable_pow_exp i
  have hI2 := integrable_pow_exp_two i
  have hI3 := integrable_Jint_integrand (((m - i : <<N>>)):<<R>>) (by positivity) i
  have hIB : MeasureTheory.IntegrableOn
      (fun u : <<R>> => u ^ i * Real.exp (-u) * Bfun (m - i) u) (Set.Ioi 0) := by
    have hfeq : (fun u : <<R>> => u ^ i * Real.exp (-u) * Bfun (m - i) u)
        = fun u : <<R>> =>
          (u ^ i * Real.exp (-u) * (2:<<R>>) ^ (-(((m - i : <<N>>)):<<R>>) * Real.exp (-u))
            - u ^ i * Real.exp (-u))
          + ((m - i : <<N>>) : <<R>>) * Real.log 2 * (u ^ i * Real.exp (-(2 * u))) := by
      funext u
      have h := hpt u
      linarith [h]
    rw [hfeq]
    exact (hI3.sub hI1).add (hI2.const_mul _)
  have hbase : <<INT>> u in Set.Ioi (0:<<R>>), u ^ i * Real.exp (-u)
      = (Nat.factorial i : <<R>>) := by
    rw [<<LARR>> integral_exp_neg_mul_pow i]
    apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioi
    intro u _
    simp only
    ring
  have hbeta : <<INT>> u in Set.Ioi (0:<<R>>), u ^ i * Real.exp (-u) * Bfun (m - i) u
      = (Nat.factorial i : <<R>>) * betaCirc i (m - i) := by
    unfold betaCirc
    field_simp
  unfold Jint
  calc <<INT>> u in Set.Ioi (0:<<R>>),
        u ^ i * Real.exp (-u) * (2:<<R>>) ^ (-(((m - i : <<N>>)):<<R>>) * Real.exp (-u))
      = <<INT>> u in Set.Ioi (0:<<R>>),
        (u ^ i * Real.exp (-u) * Bfun (m - i) u + u ^ i * Real.exp (-u)
          - ((m - i : <<N>>) : <<R>>) * Real.log 2 * (u ^ i * Real.exp (-(2 * u)))) := by
        apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioi
        intro u _
        simp only [hpt u]
    _ = (<<INT>> u in Set.Ioi (0:<<R>>), u ^ i * Real.exp (-u) * Bfun (m - i) u)
        + (<<INT>> u in Set.Ioi (0:<<R>>), u ^ i * Real.exp (-u))
        - <<INT>> u in Set.Ioi (0:<<R>>),
            ((m - i : <<N>>) : <<R>>) * Real.log 2 * (u ^ i * Real.exp (-(2 * u))) := by
        rw [MeasureTheory.integral_sub (hIB.add hI1) (hI2.const_mul _),
          MeasureTheory.integral_add hIB hI1]
    _ = (Nat.factorial i : <<R>>) * betaCirc i (m - i) + (Nat.factorial i : <<R>>)
        - ((m - i : <<N>>) : <<R>>) * Real.log 2 * ((Nat.factorial i : <<R>>) / 2 ^ (i + 1)) := by
        rw [hbeta, hbase, MeasureTheory.integral_const_mul, integral_pow_exp_neg_two_mul]
