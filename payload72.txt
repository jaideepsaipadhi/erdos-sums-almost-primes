/-- BRIDGE: for a function with a power series on a ball, analytic everywhere,
the n-th iterated (one-dimensional) derivative at the center is n! times the
n-th coefficient. This is the Taylor-coefficient relation Mathlib does not
state directly for `iteratedDeriv`. -/
theorem iteratedDeriv_eq_factorial_smul_coeff
    {f : <<C>> <<TO>> <<C>>} {p : FormalMultilinearSeries <<C>> <<C>> <<C>>} {r : ENNReal}
    (h : HasFPowerSeriesOnBall f p 0 r) (h' : AnalyticOn <<C>> f Set.univ) (n : <<N>>) :
    iteratedDeriv n f 0 = (Nat.factorial n : <<C>>) * p.coeff n := by
  have hv := h.iteratedFDeriv_eq_sum h' (v := fun _ : Fin n => (1:<<C>>))
  rw [iteratedDeriv_eq_iteratedFDeriv]
  rw [hv]
  have hconst : <<ALL>> <<SIGMA>> : Equiv.Perm (Fin n),
      p n (fun i => (fun _ : Fin n => (1:<<C>>)) (<<SIGMA>> i)) = p.coeff n := by
    intro <<SIGMA>>
    rfl
  rw [Finset.sum_congr rfl (fun <<SIGMA>> _ => hconst <<SIGMA>>)]
  rw [Finset.sum_const, Finset.card_univ, Fintype.card_perm, Fintype.card_fin]
  rw [nsmul_eq_mul]
