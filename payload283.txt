/-- d/ds G2C = G2C * (per-prime sum over p >= 3), assembling the p = 3 factor
and the p >= 5 tail by the product rule.  The p = 3 term appears separately
because G2C splits it out; combining gives the full p >= 3 sum of Lemma 3.9. -/
theorem hasDerivAt_G2C (z : CC) (hz : Norm.norm z LEQ 2) (s : RR)
    (hsmem : s IN Set.Ioo (1:RR) 2) :
    HasDerivAt (fun t : RR => G2C t z)
      (G2C s z *
        ((-(z * (z - 1) * ((Real.log ((3:NN):RR) : RR) : CC))
            / (((((((3:NN):RR)) ^ s : RR)) : CC) - 1)
              / (((((((3:NN):RR)) ^ s : RR)) : CC) - z))
          + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
              (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
                / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
                  / (((((((p:NN):RR)) ^ s : RR)) : CC) - z)))) s := by
  have h3 := hasDerivAt_prime_three_factor z hz s hsmem
  have htail := hasDerivAt_G2tailC z hz s hsmem
  have hcast : FORALL t : RR, ((((3:NN):RR)) ^ (-t) : RR) = (((3:RR)) ^ (-t) : RR) := by
    intro t
    norm_num
  simp only [hcast] at h3
  have hmul := h3.mul htail
  have hG2C : FORALL t : RR, G2C t z
      = ((1 - z * ((((3:RR)) ^ (-t) : RR) : CC))INV
          * (((1 - ((3:RR)) ^ (-t) : RR) : CC)) ^ z) * G2tailC t z := by
    intro t
    unfold G2C
    ring
  simp only [hG2C]
  convert hmul using 1
  rw [hG2C s]
  ring
