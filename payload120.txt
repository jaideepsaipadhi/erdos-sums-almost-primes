/-- Pointwise: (1-x)^{-1} <= exp(2x) for 0 <= x <= 1/2. -/
theorem inv_one_sub_le_exp (x : RR) (hx0 : 0 LEQ x) (hx : x LEQ 1/2) :
    (1 - x)INV LEQ Real.exp (2 * x) := by
  have hone : (0:RR) < 1 - x := by linarith
  have hlog : -Real.log (1 - x) LEQ 2 * x := neg_log_one_sub_le x hx0 hx
  have hkey : Real.log ((1 - x)INV) LEQ 2 * x := by
    rw [Real.log_inv]
    exact hlog
  have hpos : (0:RR) < (1 - x)INV := inv_pos.mpr hone
  calc (1 - x)INV = Real.exp (Real.log ((1 - x)INV)) := (Real.exp_log hpos).symm
    _ LEQ Real.exp (2 * x) := Real.exp_le_exp.mpr hkey
