/-- The r=2 s-free dominator, indexed at p >= 5.
Rebuilt from the N-indexed p-series rather than restricted from the p >= 3
version, per the file's idiom. -/
theorem deriv2_bound_uniform_summable_five :
    Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      200 * (Real.log ((p:NN):RR)) ^ 2 * (((p:NN):RR)) ^ (-(2:RR))) := by
  have hbase : Summable (fun n : NN => (n:RR) ^ (-(5:RR)/4)) := by
    rw [Real.summable_nat_rpow]
    norm_num
  have hsub : Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      (((p:NN):RR)) ^ (-(5:RR)/4)) := hbase.subtype _
  have hc := hsub.mul_left ((200:RR) * (64/9))
  apply Summable.of_nonneg_of_le _ _ hc
  DOT intro p
    have hp5 : (5:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have : (0:RR) LEQ (Real.log ((p:NN):RR)) ^ 2 := sq_nonneg _
    positivity
  DOT intro p
    have hp5 : (5:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
    have hlog := log_sq_le (((p:NN):RR)) (by linarith)
    have hpos : (0:RR) < (((p:NN):RR)) ^ (-(2:RR)) := Real.rpow_pos_of_pos hp0 _
    have hstep : (Real.log ((p:NN):RR)) ^ 2 * (((p:NN):RR)) ^ (-(2:RR))
        LEQ ((64/9) * (((p:NN):RR)) ^ ((3:RR)/4)) * (((p:NN):RR)) ^ (-(2:RR)) :=
      mul_le_mul_of_nonneg_right hlog (le_of_lt hpos)
    have hcomb : ((64/9) * (((p:NN):RR)) ^ ((3:RR)/4)) * (((p:NN):RR)) ^ (-(2:RR))
        = (64/9) * (((p:NN):RR)) ^ ((3:RR)/4 + -(2:RR)) := by
      rw [mul_assoc, LARR Real.rpow_add hp0]
    have hmono : (((p:NN):RR)) ^ ((3:RR)/4 + -(2:RR)) LEQ (((p:NN):RR)) ^ (-(5:RR)/4) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      norm_num
    rw [hcomb] at hstep
    nlinarith [hstep, hmono, hpos, sq_nonneg (Real.log ((p:NN):RR))]
