/-- kappa(s)^z -> (1/2)^z as s -> 1 from above. -/
theorem kappa_cpow_tendsto (z : CC) :
    Filter.Tendsto (fun s : RR => ((kappaFac s : RR) : CC) ^ z)
      (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds ((((1/2 : RR)) : CC) ^ z)) := by
  have hk := kappaFac_tendsto_half
  have heq : FORALL s : RR, s IN Set.Ioi (1:RR) ARROW
      ((kappaFac s : RR) : CC) ^ z = Complex.exp (z * ((Real.log (kappaFac s) : RR) : CC)) := by
    intro s hs
    exact ofReal_cpow_eq_exp (kappaFac s) (kappaFac_pos s hs (cFac_pos s hs)) z
  have hlog : Filter.Tendsto (fun s : RR => Real.log (kappaFac s))
      (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds (Real.log (1/2 : RR))) := by
    apply (Real.continuousAt_log (by norm_num)).tendsto.comp hk
  have hcomp : Filter.Tendsto (fun s : RR => Complex.exp (z * ((Real.log (kappaFac s) : RR) : CC)))
      (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds (Complex.exp (z * ((Real.log (1/2 : RR) : RR) : CC)))) := by
    apply Complex.continuous_exp.continuousAt.tendsto.comp
    apply Filter.Tendsto.const_mul
    exact (Complex.continuous_ofReal.continuousAt).tendsto.comp hlog
  have hval : Complex.exp (z * ((Real.log (1/2 : RR) : RR) : CC)) = (((1/2 : RR)) : CC) ^ z :=
    (ofReal_cpow_eq_exp (1/2 : RR) (by norm_num) z).symm
  rw [LARR hval]
  apply hcomp.congr'
  filter_upwards [self_mem_nhdsWithin] with s hs
  exact (heq s hs).symm
