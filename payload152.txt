theorem cpow_neg_multipliable_five (s : RR) (hs : 1 < s) (z : CC) (hz : Norm.norm z LEQ 2) :
    Multipliable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      (((1 - ((p:NN):RR) ^ (-s) : RR) : CC)) ^ (-z)) := by
  have hsum := cpow_neg_sub_one_summable s hs z hz
  have hmult : Multipliable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      1 + ((((1 - ((p:NN):RR) ^ (-s) : RR) : CC)) ^ (-z) - 1)) := by
    apply Complex.multipliable_one_add_of_summable
    exact hsum.of_norm
  apply hmult.congr
  intro p
  ring
