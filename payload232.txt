/-- The p = 3 Euler factor is differentiable on the wide disc |z| < 5/2.
Simpler than the general per-prime lemma: no log estimates are needed, only
that 1 - z*3^(-s) is nonvanishing there (|z*3^(-s)| < 5/6 at s = 1) and that
the cpow factor is entire.  The general wide lemma requires p >= 5, so the
p = 3 factor of G2C needs this separate treatment. -/
theorem prime_three_differentiableOn_wide (s : RR) (hs : 1 LEQ s) :
    DifferentiableOn CC (fun z : CC => (1 - z * ((((3:RR)) ^ (-s) : RR) : CC))INV *
      ((((1 - ((3:RR)) ^ (-s) : RR)) : CC)) ^ z) (Metric.ball 0 (5/2)) := by
  set a : RR := ((3:RR)) ^ (-s) with ha_def
  have ha_pos : (0:RR) < a := by rw [ha_def]; positivity
  have ha_le : a LEQ 1/3 := by
    rw [ha_def]
    have h1 : ((3:RR)) ^ (-s) LEQ ((3:RR)) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by norm_num)
      linarith
    rw [Real.rpow_neg_one] at h1
    norm_num at h1
    linarith
  have h1a_pos : (0:RR) < 1 - a := by linarith
  have h1a_ne : ((((1 - a : RR))) : CC) NEQ 0 := by
    simp only [ne_eq, Complex.ofReal_eq_zero]
    linarith
  intro z hz
  rw [Metric.mem_ball, dist_zero_right] at hz
  have hza : Norm.norm (z * ((a : RR) : CC)) < 1 := by
    rw [norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_of_pos ha_pos]
    calc Norm.norm z * a < (5/2) * a := by
          apply mul_lt_mul_of_pos_right hz ha_pos
      _ LEQ (5/2) * (1/3) := by
          apply mul_le_mul_of_nonneg_left ha_le (by norm_num)
      _ < 1 := by norm_num
  have hne : (1:CC) - z * ((a : RR) : CC) NEQ 0 := by
    intro hcon
    have := sub_eq_zero.mp hcon
    rw [LARR this] at hza
    simp at hza
  apply DifferentiableAt.differentiableWithinAt
  apply DifferentiableAt.mul
  DOT apply DifferentiableAt.inv _ hne
    exact (differentiableAt_const 1).sub (differentiableAt_id.mul (differentiableAt_const _))
  DOT exact DifferentiableAt.const_cpow differentiableAt_id (Or.inl h1a_ne)
