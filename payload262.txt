/-- Lemma 3.9's per-prime term, assembled from two applications of
deriv_log_one_sub_const_rpow (at c = z and c = 1).
The cancellation between the two logs is what produces p^(-2s) decay;
bounding them separately would give only p^(-s), which is not summable. -/
theorem deriv_log_prime_factor (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : RR) (hz : |z| LEQ 2) :
    HasDerivAt (fun t : RR =>
        -Real.log (1 - z * ((p:RR)) ^ (-t)) + z * Real.log (1 - ((p:RR)) ^ (-t)))
      (-(z * (z - 1) * Real.log (p:RR))
        / ((((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z))) s := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < (p:RR) := by linarith
  have hps : (3:RR) LEQ ((p:RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ ((p:RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hneg : ((p:RR)) ^ (-s) = (((p:RR)) ^ s)INV := Real.rpow_neg (le_of_lt hp0) s
  have hpsinv : ((p:RR)) ^ (-s) LEQ 1/3 := by
    rw [hneg, inv_le_comm0 (by linarith) (by norm_num)]
    linarith
  have hppos : (0:RR) < ((p:RR)) ^ (-s) := Real.rpow_pos_of_pos hp0 _
  have hzabs := abs_le.mp hz
  have hne1 : (1:RR) - z * ((p:RR)) ^ (-s) NEQ 0 := by
    have : z * ((p:RR)) ^ (-s) LEQ 2/3 := by nlinarith [hzabs.2, hppos, hpsinv]
    intro hcon
    have : z * ((p:RR)) ^ (-s) = 1 := by linarith
    linarith
  have hne2 : (1:RR) - (1:RR) * ((p:RR)) ^ (-s) NEQ 0 := by
    rw [one_mul]
    intro hcon
    have : ((p:RR)) ^ (-s) = 1 := by linarith
    linarith
  have hA := (deriv_log_one_sub_const_rpow p hp z s hne1).neg
  have hB := (deriv_log_one_sub_const_rpow p hp 1 s hne2).const_mul z
  simp only [one_mul] at hB
  have hsum := hA.add hB
  have hval : -(z * (z - 1) * Real.log (p:RR))
        / ((((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z))
      = -(z * Real.log (p:RR) * ((p:RR)) ^ (-s) / (1 - z * ((p:RR)) ^ (-s)))
        + z * (1 * Real.log (p:RR) * ((p:RR)) ^ (-s) / (1 - 1 * ((p:RR)) ^ (-s))) := by
    rw [one_mul, one_mul, hneg]
    have hps0 : ((p:RR)) ^ s NEQ 0 := by linarith
    have hd1 : (1:RR) - z * (((p:RR)) ^ s)INV NEQ 0 := by rw [LARR hneg]; exact hne1
    have hd2 : (1:RR) - (((p:RR)) ^ s)INV NEQ 0 := by
      rw [LARR hneg]; simpa using hne2
    field_simp
    ring
  rw [hval]
  exact hsum
