/-- Taylor bridge without a global analyticity hypothesis: over a complete
space, HasFPowerSeriesOnBall alone determines the iterated derivatives.
Weakening of iteratedDeriv_eq_factorial_smul_coeff, which required
AnalyticOn C f univ and so could not be applied to functions with poles
(e.g. G2C, whose nearest pole is at z = 3). -/
theorem iteratedDeriv_eq_factorial_smul_coeff'
    {f : CC ARROW CC} {p : FormalMultilinearSeries CC CC CC} {r : ENNReal}
    (h : HasFPowerSeriesOnBall f p 0 r) (n : NN) :
    iteratedDeriv n f 0 = (Nat.factorial n : CC) * p.coeff n := by
  have hv := h.iteratedFDeriv_eq_sum_of_completeSpace (v := fun _ : Fin n => (1:CC))
  rw [iteratedDeriv_eq_iteratedFDeriv]
  rw [hv]
  have hconst : FORALL SIGMA : Equiv.Perm (Fin n),
      p n (fun i => (fun _ : Fin n => (1:CC)) (SIGMA i)) = p.coeff n := by
    intro SIGMA
    rfl
  rw [Finset.sum_congr rfl (fun SIGMA _ => hconst SIGMA)]
  rw [Finset.sum_const, Finset.card_univ, Fintype.card_perm, Fintype.card_fin]
  rw [nsmul_eq_mul]
