theorem EinTailFun_split (a : <<R>>) (ha : 0 < a) :
    EinTailFun a = EinTailFun 1 - <<INT>> v in (1:<<R>>)..a, Real.exp (-v) / v := by
  unfold EinTailFun
  rcases le_total (1:<<R>>) a with h1 | h1
  <<DOT>> have hunion : Set.Ici (1:<<R>>) = Set.Ico 1 a <<UN>> Set.Ici a :=
      (Set.Ico_union_Ici_eq_Ici h1).symm
    have hdisj : Disjoint (Set.Ico (1:<<R>>) a) (Set.Ici a) := by
      rw [Set.disjoint_left]
      intro x hx hx2
      exact absurd (Set.mem_Ici.mp hx2) (not_le.mpr hx.2)
    have hint1 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ico 1 a) :=
      (exp_div_integrableOn_Ici 1 (by norm_num)).mono
        (fun x hx => Set.mem_Ici.mpr hx.1) le_rfl
    have hint2 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ici a) :=
      exp_div_integrableOn_Ici a ha
    have hadd : (<<INT>> v in Set.Ici (1:<<R>>), Real.exp (-v) / v)
        = (<<INT>> v in Set.Ico (1:<<R>>) a, Real.exp (-v) / v)
          + <<INT>> v in Set.Ici a, Real.exp (-v) / v := by
      rw [hunion, MeasureTheory.setIntegral_union hdisj measurableSet_Ici hint1 hint2]
    have hIcoIoc : (<<INT>> v in Set.Ico (1:<<R>>) a, Real.exp (-v) / v)
        = <<INT>> v in Set.Ioc (1:<<R>>) a, Real.exp (-v) / v := by
      rw [MeasureTheory.integral_Ico_eq_integral_Ioo,
        MeasureTheory.integral_Ioc_eq_integral_Ioo]
    rw [intervalIntegral.integral_of_le h1, <<LARR>> hIcoIoc]
    linarith [hadd]
  <<DOT>> have hunion : Set.Ici a = Set.Ico a 1 <<UN>> Set.Ici 1 :=
      (Set.Ico_union_Ici_eq_Ici h1).symm
    have hdisj : Disjoint (Set.Ico a (1:<<R>>)) (Set.Ici 1) := by
      rw [Set.disjoint_left]

    have hint2 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ici 1) :=
      exp_div_integrableOn_Ici 1 (by norm_num)
    have hadd : (<<INT>> v in Set.Ici a, Real.exp (-v) / v)
        = (<<INT>> v in Set.Ico a (1:<<R>>), Real.exp (-v) / v)
