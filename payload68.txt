/-- The change of variables t = e^(-u): int_{(0,inf)} e^u Bfun m u du
= int_{(0,1)} (e^(-at) - 1 + at)/t^2 dt with a = m log 2. -/
theorem exp_Bfun_integral_eq (m : <<N>>) :
    (<<INT>> u in Set.Ioi (0:<<R>>), Real.exp u * Bfun m u)
      = <<INT>> t in Set.Ioo (0:<<R>>) 1,
        (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2 * t)) - 1
          + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2 := by
  have hderiv : <<ALL>> u <<IN>> Set.Ioi (0:<<R>>),
      HasDerivWithinAt (fun u : <<R>> => Real.exp (-u))
        (Real.exp (-u) * (-1)) (Set.Ioi 0) u := by
    intro u _
    exact (((hasDerivAt_neg u).exp)).hasDerivWithinAt
  have himg := MeasureTheory.integral_image_eq_integral_abs_deriv_smul
    measurableSet_Ioi hderiv expNeg_injOn
    (fun t : <<R>> => (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2 * t)) - 1
      + ((m:<<N>>):<<R>>) * Real.log 2 * t) / t ^ 2)
  rw [expNeg_image] at himg
  rw [himg]
  apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioi
  intro u hu
  have hu0 : (0:<<R>>) < u := hu
  have hE : (0:<<R>>) < Real.exp (-u) := Real.exp_pos _
  simp only [smul_eq_mul, mul_neg_one, abs_neg, abs_of_pos hE]
  have hBeq : Bfun m u
      = (Real.exp (-(((m:<<N>>):<<R>>) * Real.log 2 * Real.exp (-u))) - 1
        + ((m:<<N>>):<<R>>) * Real.log 2 * Real.exp (-u)) := by
    unfold Bfun
    rw [Real.rpow_def_of_pos (by norm_num : (0:<<R>>) < 2)]
    ring_nf
  rw [hBeq]
  have hsq : Real.exp (-u) ^ 2 <<NE>> 0 := by positivity
  field_simp
  rw [<<LARR>> Real.exp_add]
  ring_nf
