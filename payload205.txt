/-- The cFac family is multipliable for s > 1.
Needed because 0 < cFac s has been threaded as a hypothesis throughout;
without multipliability that would be an assumption about a tprod junk value. -/
theorem cFac_multipliable (s : RR) (hs : 1 < s) :
    Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (1 - ((p:NN):RR) ^ (-s))INV) := by
  have hthird : FORALL p : {p : NN // p.Prime AND 3 LEQ p},
      ((p:NN):RR) ^ (-s) LEQ 1/3 := by
    intro p
    have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
    have h1 : ((p:NN):RR) ^ (-s) LEQ ((p:NN):RR) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [Real.rpow_neg_one] at h1
    have h2 : (((p:NN):RR))INV LEQ 1/3 := by
      rw [inv_le_comm0 (by linarith) (by norm_num)]
      linarith
    linarith
  have hpos : FORALL p : {p : NN // p.Prime AND 3 LEQ p},
      (0:RR) < ((p:NN):RR) ^ (-s) := by
    intro p
    have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
    positivity
  have hcomp : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (3/2) * (((p:NN):RR) ^ (-s))) := by
    have hbase : Summable (fun n : NN => (n:RR) ^ (-s)) :=
      Real.summable_nat_rpow.mpr (by linarith)
    exact (hbase.subtype _).mul_left _
  have hsum : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (1 - ((p:NN):RR) ^ (-s))INV - 1) := by
    apply Summable.of_nonneg_of_le _ _ hcomp
    DOT intro p
      have h3 := hthird p
      have hp := hpos p
      have hden : (0:RR) < 1 - ((p:NN):RR) ^ (-s) := by linarith
      rw [sub_nonneg, le_inv_comm0 (by norm_num) hden]
      linarith
    DOT intro p
      have h3 := hthird p
      have hp := hpos p
      have hden : (0:RR) < 1 - ((p:NN):RR) ^ (-s) := by linarith
      have hkey : (1 - ((p:NN):RR) ^ (-s))INV - 1
          = ((p:NN):RR) ^ (-s) / (1 - ((p:NN):RR) ^ (-s)) := by
        field_simp
      rw [hkey, div_le_iff0 hden]
      nlinarith
  have h := Real.multipliable_one_add_of_summable hsum
  apply h.congr
  intro p
  ring
