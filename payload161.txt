/-- Splitting a product over all primes into p = 2 and the p >= 3 tail. -/
theorem tprod_split_two {M : Type*} [CommMonoid M] [TopologicalSpace M] [T2Space M]
    [ContinuousMul M] (f : NN ARROW M)
    (hf3 : Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} => f p)) :
    Multipliable (fun p : {p : NN // p.Prime} => f p)
      AND (TPRODN p : {p : NN // p.Prime}, f p)
        = f 2 * TPRODN p : {p : NN // p.Prime AND 3 LEQ p}, f p := by
  classical
  set b2 : {p : NN // p.Prime} := ANGL 2, Nat.prime_two ANGR with hb2
  set g : {p : NN // p.Prime} ARROW M := fun p => f p with hg
  have hsingle : HasProd ((({b2} : Set {p : NN // p.Prime}).restrict g)) (g b2) :=
    hasProd_singleton b2 g
  have hcompl : HasProd (((({b2} : Set {p : NN // p.Prime})COMPL).restrict g))
      (TPRODN p : {p : NN // p.Prime AND 3 LEQ p}, f p) :=
    (compl_two_equiv.hasProd_iff).mpr hf3.hasProd
  have hall := hsingle.mul_compl hcompl
  exact ANGL hall.multipliable, hall.tprod_eq ANGR
