/-- The regularised product over p >= 3 is exactly G2C. -/
theorem reg_tprod_eq_G2C (s : RR) (hs : 1 LEQ s) (z : CC) (hz : Norm.norm z LEQ 2) :
    (TPRODN p : {p : NN // p.Prime AND 3 LEQ p},
        (1 - z * ((((p:NN):RR) ^ (-s) : RR) : CC))INV
          * (((1 - ((p:NN):RR) ^ (-s) : RR) : CC)) ^ z)
      = G2C s z := by
  have h5 := prime_factor_multipliable_complex s hs z hz
  have hsplit := tprod_split_three
    (fun n : NN => (1 - z * (((n:RR) ^ (-s) : RR) : CC))INV
      * (((1 - (n:RR) ^ (-s) : RR) : CC)) ^ z) h5
  rw [hsplit.2]
  unfold G2C G2tailC
  norm_num
