/-- The norm of the family at a prime power is a geometric term. -/
theorem zOmegaNorm_prime_pow (s : RR) (z : CC) {p : NN} (hp : p.Prime) (n : NN) :
    zOmegaNormHom s z (p ^ n) = (Norm.norm z * ((p:RR) ^ (-s))) ^ n := by
  rw [zOmegaNormHom_apply]
  have hpn : (p:NN) ^ n NEQ 0 := pow_ne_zero n hp.pos.ne'
  have hppos : 0 < (p:NN) ^ n := Nat.pos_of_ne_zero hpn
  show Norm.norm (if (p:NN) ^ n = 0 then 0
    else z ^ (bigOmega ((p:NN) ^ n)) * ((((p:NN) ^ n : NN) : CC) ^ (-(s:CC)))) = _
  rw [if_neg hpn, bigOmega_prime_pow hp n, norm_mul, norm_pow]
  rw [Complex.norm_natCast_cpow_of_pos hppos]
  rw [Complex.neg_re, Complex.ofReal_re]
  rw [Nat.cast_pow]
  rw [LARR Real.rpow_natCast ((p:RR)) n, LARR Real.rpow_mul (Nat.cast_nonneg p)]
  rw [mul_pow]
  congr 1
  rw [LARR Real.rpow_natCast ((p:RR) ^ (-s)) n, LARR Real.rpow_mul (Nat.cast_nonneg p)]
  congr 1
  ring
