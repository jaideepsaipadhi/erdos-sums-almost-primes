theorem coeff_pow_neg (s : <<R>>) (hs : 1 < s) (l : <<N>>) :
    (iteratedDeriv l (fun z : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-z)) 0)
        / (Nat.factorial l : <<C>>)
      = ((-Real.log (s - 1)) ^ l / (Nat.factorial l) : <<R>>) := by
  have hspos : (0:<<R>>) < s - 1 := by linarith
  have hsne : ((s - 1 : <<R>>) : <<C>>) <<NE>> 0 := by
    simp only [ne_eq, Complex.ofReal_eq_zero]
    linarith
  have hrewrite : (fun z : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-z))
      = fun z : <<C>> => Complex.exp ((-(Real.log (s - 1) : <<C>>)) * z) := by
    funext z
    rw [Complex.cpow_def_of_ne_zero hsne]
    congr 1
    rw [Complex.log_ofReal_of_pos hspos]
    ring
  rw [hrewrite, iteratedDeriv_cexp_const_mul l (-(Real.log (s - 1) : <<C>>))]
  simp only [mul_zero, Complex.exp_zero, mul_one]
  have hfacnz : ((Nat.factorial l : <<N>>) : <<C>>) <<NE>> 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero l)
  push_cast
  ring
