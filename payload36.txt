/-- The antiderivative identity behind integration by parts for the Euler
integrand: d/dv [-e^(-v) log v] = e^(-v) log v - e^(-v)/v, for v > 0. -/
theorem hasDerivAt_negExpLog (v : <<R>>) (hv : 0 < v) :
    HasDerivAt (fun t : <<R>> => -(Real.exp (-t) * Real.log t))
      (Real.exp (-v) * Real.log v - Real.exp (-v) / v) v := by
  have hexp : HasDerivAt (fun t : <<R>> => Real.exp (-t)) (-(Real.exp (-v))) v := by
    have h := (Real.hasDerivAt_exp (-v)).comp v (hasDerivAt_neg v)
    simpa using h
  have hlog : HasDerivAt Real.log v<<INV>> v := Real.hasDerivAt_log (ne_of_gt hv)
  have hmul := hexp.mul hlog
  have hneg := hmul.neg
  have harith : -(-(Real.exp (-v)) * Real.log v + Real.exp (-v) * v<<INV>>)
      = Real.exp (-v) * Real.log v - Real.exp (-v) / v := by
    field_simp
    try ring
  rw [harith] at hneg
  exact hneg

/-- Finite-interval integration by parts for the Euler integrand. -/
theorem euler_parts_core (b c : <<R>>) (hb : 0 < b) (hbc : b <<LE>> c) :
    (<<INT>> v in b..c, (Real.exp (-v) * Real.log v - Real.exp (-v) / v))
      = -(Real.exp (-c) * Real.log c) + Real.exp (-b) * Real.log b := by
  have hderiv : <<ALL>> x <<IN>> Set.uIcc b c,
      HasDerivAt (fun t : <<R>> => -(Real.exp (-t) * Real.log t))
        (Real.exp (-x) * Real.log x - Real.exp (-x) / x) x := by
    intro x hx
    rw [Set.uIcc_of_le hbc] at hx
    exact hasDerivAt_negExpLog x (by linarith [hx.1])
  have hint : IntervalIntegrable
      (fun v : <<R>> => Real.exp (-v) * Real.log v - Real.exp (-v) / v)
      MeasureTheory.volume b c := by
    apply ContinuousOn.intervalIntegrable
    rw [Set.uIcc_of_le hbc]
    apply ContinuousOn.sub
    <<DOT>> apply ContinuousOn.mul
      <<DOT>> exact (Real.continuous_exp.comp continuous_neg).continuousOn
      <<DOT>> apply ContinuousOn.log continuousOn_id
        intro x hx
        exact ne_of_gt (by linarith [hx.1])
    <<DOT>> apply ContinuousOn.div
      <<DOT>> exact (Real.continuous_exp.comp continuous_neg).continuousOn
      <<DOT>> exact continuousOn_id
      <<DOT>> intro x hx
        exact ne_of_gt (by linarith [hx.1])
  rw [intervalIntegral.integral_eq_sub_of_hasDerivAt hderiv hint]
