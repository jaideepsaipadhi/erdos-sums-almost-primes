/-- File-canonical form of the complex per-prime derivative: second term as a
real log cast to C (matching G2tailC_eq_exp_sum and
prime_correction_summable_complex), indexed from p >= 5 to match G2tailC. -/
theorem deriv_clog_prime_factor_canon (p : NN) (hp : 5 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : CC) (hz : Norm.norm z LEQ 2) :
    HasDerivAt (fun t : RR =>
        -Complex.log (1 - z * (((((p:NN):RR)) ^ (-t) : RR) : CC))
          + z * ((Real.log (1 - (((p:NN):RR)) ^ (-t)) : RR) : CC))
      (-(z * (z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
        / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
          / (((((((p:NN):RR)) ^ s : RR)) : CC) - z)) s := by
  have hp3 : (3:NN) LEQ p := by omega
  have hbase := deriv_clog_prime_factor p hp3 s hs z hz
  apply hbase.congr_of_eventuallyEq
  have hnhds : Set.Ioi (0:RR) IN nhds s := by
    apply Ioi_mem_nhds
    linarith
  filter_upwards [hnhds] with t ht
  congr 1
  congr 1
  have hpr : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp3
  have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
  have hlt : (((p:NN):RR)) ^ (-t) < 1 :=
    Real.rpow_lt_one_of_one_lt_of_neg (by linarith) (by linarith [ht])
  have hpos : (0:RR) < 1 - (((p:NN):RR)) ^ (-t) := by linarith
  rw [Complex.ofReal_log (le_of_lt hpos)]
  push_cast
  ring
