/-- The derivative contour integral converges as s -> 1+, by dominated
convergence.  M is the same first-derivative bound lemma_3_8_integral_r1
carries -- no new hypothesis. -/
theorem deriv_contour_integral_tendsto (i : NN)
    (f : RR ARROW RR) (f1 : RR)
    (hfc : Filter.Tendsto f (nhdsWithin 1 (Set.Ioi (1:RR))) (nhds f1))
    (M : RR) (hM0 : 0 LEQ M)
    (hM : FORALL s : RR, s IN Set.Ioi (1:RR) ARROW FORALL theta : RR,
      Norm.norm (G2manu s (circleMap 0 (2:RR) theta)
        * cBracket (f s) s (circleMap 0 (2:RR) theta)) LEQ M)
    (hcont : FORALL s : RR, s IN Set.Ioi (1:RR) ARROW Continuous (fun theta : RR =>
      deriv (circleMap 0 (2:RR)) theta
        * ((1 / circleMap 0 (2:RR) theta) ^ i
          * ((circleMap 0 (2:RR) theta)INV
            * (G2manu s (circleMap 0 (2:RR) theta)
              * cBracket (f s) s (circleMap 0 (2:RR) theta)))))) :
    Filter.Tendsto (fun s : RR =>
        INTEGRAL theta in Set.Ioc (0:RR) (2 * Real.pi),
          deriv (circleMap 0 (2:RR)) theta
            * ((1 / circleMap 0 (2:RR) theta) ^ i
              * ((circleMap 0 (2:RR) theta)INV
                * (G2manu s (circleMap 0 (2:RR) theta)
                  * cBracket (f s) s (circleMap 0 (2:RR) theta)))))
      (nhdsWithin 1 (Set.Ioi (1:RR)))
      (nhds (INTEGRAL theta in Set.Ioc (0:RR) (2 * Real.pi),
        deriv (circleMap 0 (2:RR)) theta
          * ((1 / circleMap 0 (2:RR) theta) ^ i
            * ((circleMap 0 (2:RR) theta)INV
              * G2deriv1 f1 (circleMap 0 (2:RR) theta))))) := by
  apply MeasureTheory.tendsto_integral_filter_of_dominated_convergence
    (bound := fun _ : RR => M)
  DOT filter_upwards [self_mem_nhdsWithin] with s hs
    exact (hcont s hs).aestronglyMeasurable
  DOT filter_upwards [self_mem_nhdsWithin] with s hs
    filter_upwards with theta
    exact contour_integrand_bound_gen i M hM0 theta
      (G2manu s (circleMap 0 (2:RR) theta)
        * cBracket (f s) s (circleMap 0 (2:RR) theta)) (hM s hs theta)
  DOT exact MeasureTheory.integrable_const M
  DOT filter_upwards with theta
    exact deriv_integrand_tendsto i theta f f1 hfc
