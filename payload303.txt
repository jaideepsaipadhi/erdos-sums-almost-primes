/-- cCoeff as a contour integral at radius 2, the G2at1 analogue of
gManu_eq_circleIntegral.  G2at1 is differentiable on closedBall 0 (12/5)
(G2at1_differentiableOn_wide), which contains closedBall 0 2. -/
theorem cCoeff_eq_circleIntegral (i : NN) :
    cCoeff i
      = (2 * ((Real.pi : RR) : CC) * Complex.I)INV
        SMUL (circleIntegral (fun z : CC => (1 / z) ^ i SMUL zINV SMUL G2at1 z)
          0 (2 : RR)) := by
  have hdiff : DifferentiableOn CC G2at1 (Metric.closedBall (0:CC) ((2 : NNReal) : RR)) := by
    apply G2at1_differentiableOn_wide.mono
    intro w hw
    rw [Metric.mem_closedBall, dist_zero_right] at hw TURNSTILE
    norm_num at hw TURNSTILE
    linarith
  have hR : (0:NNReal) < 2 := by norm_num
  have hser := hdiff.hasFPowerSeriesOnBall hR
  have hbridge := iteratedDeriv_eq_factorial_smul_coeff' hser i
  have hcoeff : cCoeff i
      = (cauchyPowerSeries G2at1 0 ((2 : NNReal) : RR)).coeff i := by
    unfold cCoeff
    rw [hbridge]
    have hfac : ((Nat.factorial i : NN) : CC) NEQ 0 :=
      Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
    field_simp
  rw [hcoeff]
  have h := cauchyPowerSeries_apply G2at1 0 ((2 : NNReal) : RR) i 1
  show ((cauchyPowerSeries G2at1 0 ((2 : NNReal) : RR) i) fun _ => 1) = _
  rw [h]
  norm_num
