/-- gManu i is bounded on (1,2) by ||c_i|| + beta_i, from the mean value step.
This is what makes the Ik-side integrands dominated, hence integrable, hence
the sum-integral interchange of Dk_eq_triple_sum available. -/
theorem gManu_bounded (i : NN) (beta : RR) (hbeta0 : 0 LEQ beta)
    (hderiv : FORALL x IN Set.Ioo (1:RR) 2,
      HasDerivAt (fun t : RR => gManu i t) (deriv (fun t : RR => gManu i t) x) x)
    (hbound : FORALL x IN Set.Ioo (1:RR) 2,
      Norm.norm (deriv (fun t : RR => gManu i t) x) LEQ beta)
    (M0 : RR) (hM00 : 0 LEQ M0)
    (hM0 : FORALL u : RR, u IN Set.Ioi (1:RR) ARROW FORALL theta : RR,
      Norm.norm (G2manu u (circleMap 0 (2:RR) theta)) LEQ M0)
    (s : RR) (hs : s IN Set.Ioo (1:RR) 2) :
    Norm.norm (gManu i s) LEQ Norm.norm (cCoeff i) + beta := by
  have hmv := gManu_sub_cCoeff_bound i beta hbeta0 hderiv hbound M0 hM00 hM0 s hs
  have hs1 : s - 1 LEQ 1 := by linarith [hs.2]
  have hbs : beta * (s - 1) LEQ beta := by
    calc beta * (s - 1) LEQ beta * 1 := by
          apply mul_le_mul_of_nonneg_left hs1 hbeta0
      _ = beta := mul_one beta
  calc Norm.norm (gManu i s)
      = Norm.norm ((gManu i s - cCoeff i) + cCoeff i) := by ring_nf
    _ LEQ Norm.norm (gManu i s - cCoeff i) + Norm.norm (cCoeff i) := norm_add_le _ _
    _ LEQ beta * (s - 1) + Norm.norm (cCoeff i) := by linarith [hmv]
    _ LEQ beta + Norm.norm (cCoeff i) := by linarith [hbs]
    _ = Norm.norm (cCoeff i) + beta := by ring
