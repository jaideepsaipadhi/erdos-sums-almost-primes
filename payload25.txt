/-- Manuscript Thm 4.1: B_j is increasing in j (the 2^(-j e^-u) factor shrinks
faster than the linear term grows). Stated as the comparison needed there. -/
theorem Bfun_mono (j m : <<N>>) (hjm : j <<LE>> m) (u : <<R>>) :
    Bfun j u <<LE>> Bfun m u := by
  unfold Bfun
  have hE : (0:<<R>>) < Real.exp (-u) := Real.exp_pos _
  have hjR : ((j:<<N>>):<<R>>) <<LE>> ((m:<<N>>):<<R>>) := by exact_mod_cast hjm
  have hgap : (2:<<R>>) ^ (-((m:<<N>>):<<R>>) * Real.exp (-u))
      - (2:<<R>>) ^ (-((j:<<N>>):<<R>>) * Real.exp (-u))
      + (((m:<<N>>):<<R>>) - ((j:<<N>>):<<R>>)) * Real.log 2 * Real.exp (-u) <<GE>> 0 := by
    set c := ((m:<<N>>):<<R>>) - ((j:<<N>>):<<R>>) with hc
    have hc0 : (0:<<R>>) <<LE>> c := by rw [hc]; linarith
    have hrw1 : (2:<<R>>) ^ (-((m:<<N>>):<<R>>) * Real.exp (-u))
        = (2:<<R>>) ^ (-((j:<<N>>):<<R>>) * Real.exp (-u)) *
          (2:<<R>>) ^ (-c * Real.exp (-u)) := by
      rw [<<LARR>> Real.rpow_add (by norm_num : (0:<<R>>) < 2)]
      congr 1
      rw [hc]
      ring
    have hbase : (0:<<R>>) < (2:<<R>>) ^ (-((j:<<N>>):<<R>>) * Real.exp (-u)) :=
      Real.rpow_pos_of_pos (by norm_num) _
    have hbase_le : (2:<<R>>) ^ (-((j:<<N>>):<<R>>) * Real.exp (-u)) <<LE>> 1 := by
      apply Real.rpow_le_one_of_one_le_of_nonpos (by norm_num)
      have : (0:<<R>>) <<LE>> ((j:<<N>>):<<R>>) * Real.exp (-u) :=
        mul_nonneg (Nat.cast_nonneg j) hE.le
      linarith
    have hB := Bfun_nonneg 0 u
    have hkey : (1:<<R>>) - (2:<<R>>) ^ (-c * Real.exp (-u))
        <<LE>> c * Real.log 2 * Real.exp (-u) := by
      have hexp : (2:<<R>>) ^ (-c * Real.exp (-u))
          = Real.exp (-c * Real.exp (-u) * Real.log 2) := by
        rw [Real.rpow_def_of_pos (by norm_num : (0:<<R>>) < 2)]
        ring_nf
      rw [hexp]
      have h := Real.add_one_le_exp (-c * Real.exp (-u) * Real.log 2)
      linarith
    rw [hrw1]
    nlinarith [hbase, hbase_le, hkey, hE]
  linarith [hgap]

/-- Partial sums of the exponential series are bounded by the exponential. -/
theorem sum_pow_div_factorial_le_exp (x : <<R>>) (hx : 0 <<LE>> x) (m : <<N>>) :
    <<SUM>> i <<IN>> Finset.range (m + 1), x ^ i / (Nat.factorial i) <<LE>> Real.exp x := by
  have hle : <<SUM>> i <<IN>> Finset.range (m + 1), x ^ i / (Nat.factorial i)
      <<LE>> <<SUM>>' i : <<N>>, x ^ i / (Nat.factorial i) := by
    apply Summable.sum_le_tsum
    <<DOT>> intro i _
      positivity
    <<DOT>> exact Real.summable_pow_div_factorial x
  rw [Real.exp_eq_tsum_div] at *
  exact hle
