/-- The r=2 series is continuous in theta on the contour.
Mirrors clog_series_continuous_theta; the global bound holds because circleMap
always lands on |z| = 2, where deriv2_clog_term_bound_z applies. -/
theorem clog2_series_continuous_theta (s : RR) (hs : 1 LEQ s) :
    Continuous (fun theta : RR =>
      TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
        (circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
            * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2 * (((((p:NN):RR)) ^ s : RR) : CC)
            * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
              + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))
          / (((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
            * (((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta) ^ 2))))) := by
  have hnorm : FORALL theta : RR, Norm.norm (circleMap 0 (2:RR) theta) LEQ 2 := by
    intro theta
    simp [circleMap]
  apply continuous_tsum
    (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      clog2_term_continuous_theta (p:NN) (by omega) s hs)
    deriv2_bound_uniform_summable_five_z
  intro p theta
  exact deriv2_clog_term_bound_z (p:NN) (by omega) s hs
    (circleMap 0 (2:RR) theta) (hnorm theta)
