/-- Each regularised factor is positive for real z with |z| <= 2, p >= 3, s >= 1. -/
theorem reg_factor_real_pos (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : RR) (hz : |z| LEQ 2) :
    0 < (1 - z * ((p:RR)) ^ (-s))INV * (1 - ((p:RR)) ^ (-s)) ^ z := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < (p:RR) := by linarith
  have hppos : (0:RR) < ((p:RR)) ^ (-s) := Real.rpow_pos_of_pos hp0 _
  have hthird : ((p:RR)) ^ (-s) LEQ 1/3 := by
    have h1 : ((p:RR)) ^ (-s) LEQ ((p:RR)) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [Real.rpow_neg_one] at h1
    have h2 : ((p:RR))INV LEQ 1/3 := by
      rw [inv_le_comm0 hp0 (by norm_num)]
      linarith
    linarith
  have hzabs := abs_le.mp hz
  have hden : (0:RR) < 1 - z * ((p:RR)) ^ (-s) := by
    nlinarith [hzabs.2, hppos, hthird]
  have hbase : (0:RR) < 1 - ((p:RR)) ^ (-s) := by linarith
  have h1 : (0:RR) < (1 - z * ((p:RR)) ^ (-s))INV := inv_pos.mpr hden
  have h2 : (0:RR) < (1 - ((p:RR)) ^ (-s)) ^ z := Real.rpow_pos_of_pos hbase z
  exact mul_pos h1 h2

/-- The real regularised family is multipliable: its logs are summable
(prime_correction_summable) and each factor is positive. -/
theorem reg_multipliable_three_real (s : RR) (hs : 1 LEQ s) (z : RR) (hz : |z| LEQ 2) :
    Multipliable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (1 - z * (((p:NN):RR)) ^ (-s))INV * (1 - (((p:NN):RR)) ^ (-s)) ^ z) := by
  apply Real.multipliable_of_summable_log
  DOT intro p
    exact reg_factor_real_pos (p:NN) p.2.2 s hs z hz
  DOT have hsum := (prime_correction_summable s hs z hz).of_abs
    apply hsum.congr
    intro p
    have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
    have hpos := reg_factor_real_pos (p:NN) p.2.2 s hs z hz
    have hppos : (0:RR) < (((p:NN):RR)) ^ (-s) := Real.rpow_pos_of_pos hp0 _
    have hzabs := abs_le.mp hz
    have hthird : (((p:NN):RR)) ^ (-s) LEQ 1/3 := by
      have h1 : (((p:NN):RR)) ^ (-s) LEQ (((p:NN):RR)) ^ (-(1:RR)) := by
        apply Real.rpow_le_rpow_of_exponent_le (by linarith)
        linarith
      rw [Real.rpow_neg_one] at h1
      have h2 : ((((p:NN):RR)))INV LEQ 1/3 := by
        rw [inv_le_comm0 hp0 (by norm_num)]
        linarith
      linarith
    have hden : (0:RR) < 1 - z * (((p:NN):RR)) ^ (-s) := by
      nlinarith [hzabs.2, hppos, hthird]
    have hbase : (0:RR) < 1 - (((p:NN):RR)) ^ (-s) := by linarith
    rw [Real.log_mul (by positivity) (by positivity), Real.log_inv, Real.log_rpow hbase]
    ring
