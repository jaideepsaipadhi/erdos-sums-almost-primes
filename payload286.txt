/-- The contour integrand is continuous in theta.  Discharges the two
measurability obligations and the integrability of F x0 in
hasDerivAt_integral_of_dominated_loc_of_deriv_le: on the circle |z| = 12/5 the
factor (1/z)^i * z^(-1) is continuous (z never vanishes) and G2manu s . is
continuous there by G2manu_differentiableOn_wide. -/
theorem contour_integrand_continuous (s : RR) (hs : 1 LEQ s) (hk : 0 < kappaFac s) (i : NN) :
    Continuous (fun t : RR =>
      deriv (circleMap 0 (12/5 : RR)) t
        SMUL ((1 / circleMap 0 (12/5 : RR) t) ^ i
          SMUL (circleMap 0 (12/5 : RR) t)INV SMUL G2manu s (circleMap 0 (12/5 : RR) t))) := by
  have hR : (0:RR) < 12/5 := by norm_num
  have hne : FORALL t : RR, circleMap 0 (12/5 : RR) t NEQ 0 :=
    fun t => circleMap_ne_center (ne_of_gt hR)
  have hcm : Continuous (circleMap 0 (12/5 : RR)) := continuous_circleMap 0 _
  have hdcm : Continuous (deriv (circleMap 0 (12/5 : RR))) := by
    simp only [deriv_circleMap]
    exact (continuous_circleMap 0 _).mul continuous_const
  have hmem : FORALL t : RR, circleMap 0 (12/5 : RR) t IN
      Metric.closedBall (0:CC) (12/5 : RR) := by
    intro t
    rw [Metric.mem_closedBall, dist_zero_right, abs_circleMap_zero]
    simp
  have hG2 : Continuous (fun t : RR => G2manu s (circleMap 0 (12/5 : RR) t)) := by
    have hcont := (G2manu_differentiableOn_wide s hs hk).continuousOn
    exact hcont.comp_continuous hcm hmem
  apply hdcm.smul
  apply Continuous.smul
  DOT exact (continuous_const.div hcm (fun t => hne t)).pow i
  DOT exact (hcm.inv0 (fun t => hne t)).smul hG2
