/-- The trinomial identity: C(k,n) * C(n,i) / k! = 1/(i! (n-i)! (k-n)!)
for i <= n <= k.  Verified numerically before encoding. -/
theorem choose_mul_choose_div_factorial (k n i : <<N>>) (hnk : n <<LE>> k) (hin : i <<LE>> n) :
    (Nat.choose k n : <<C>>) * (Nat.choose n i : <<C>>)
      = (Nat.factorial k : <<C>>)
        / ((Nat.factorial i : <<C>>) * (Nat.factorial (n - i) : <<C>>)
          * (Nat.factorial (k - n) : <<C>>)) := by
  have hfi : ((Nat.factorial i : <<N>>) : <<C>>) <<NE>> 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero i)
  have hfni : ((Nat.factorial (n - i) : <<N>>) : <<C>>) <<NE>> 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero (n - i))
  have hfkn : ((Nat.factorial (k - n) : <<N>>) : <<C>>) <<NE>> 0 :=
    Nat.cast_ne_zero.mpr (Nat.factorial_ne_zero (k - n))
  have hk : Nat.choose k n * (Nat.factorial n * Nat.factorial (k - n)) = Nat.factorial k :=
    Nat.choose_mul_factorial_mul_factorial hnk
  have hn : Nat.choose n i * (Nat.factorial i * Nat.factorial (n - i)) = Nat.factorial n :=
    Nat.choose_mul_factorial_mul_factorial hin
  rw [eq_div_iff (by exact mul_ne_zero (mul_ne_zero hfi hfni) hfkn)]
  have hkC : ((Nat.choose k n : <<N>>) : <<C>>)
      * (((Nat.factorial n : <<N>>) : <<C>>) * ((Nat.factorial (k - n) : <<N>>) : <<C>>))
      = ((Nat.factorial k : <<N>>) : <<C>>) := by
    exact_mod_cast congrArg (fun m : <<N>> => (m : <<C>>)) hk
  have hnC : ((Nat.choose n i : <<N>>) : <<C>>)
      * (((Nat.factorial i : <<N>>) : <<C>>) * ((Nat.factorial (n - i) : <<N>>) : <<C>>))
      = ((Nat.factorial n : <<N>>) : <<C>>) := by
    exact_mod_cast congrArg (fun m : <<N>> => (m : <<C>>)) hn
  calc (Nat.choose k n : <<C>>) * (Nat.choose n i : <<C>>)
        * ((Nat.factorial i : <<C>>) * (Nat.factorial (n - i) : <<C>>)
          * (Nat.factorial (k - n) : <<C>>))
      = (Nat.choose k n : <<C>>) * (Nat.factorial (k - n) : <<C>>)
        * ((Nat.choose n i : <<C>>)
          * ((Nat.factorial i : <<C>>) * (Nat.factorial (n - i) : <<C>>))) := by ring
    _ = (Nat.choose k n : <<C>>) * (Nat.factorial (k - n) : <<C>>)
        * (Nat.factorial n : <<C>>) := by rw [hnC]
    _ = (Nat.factorial k : <<C>>) := by
        rw [<<LARR>> hkC]
        ring
