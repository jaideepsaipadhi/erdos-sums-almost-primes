/-- The derivative of z -> k!/(c^k (1 - z/c)^(k+1)) is the next one in the family. -/
theorem deriv_geom_family (c : <<C>>) (hc : c <<NE>> 0) (k : <<N>>) :
    deriv (fun z : <<C>> => (Nat.factorial k : <<C>>) / (c ^ k * (1 - z / c) ^ (k + 1)))
      = fun z : <<C>> => (Nat.factorial (k + 1) : <<C>>) / (c ^ (k + 1) * (1 - z / c) ^ (k + 2)) := by
  sorry

/-- L3b core: the n-th derivative of (1 - z/c)^(-1) at 0 is n!/c^n. -/
theorem iteratedDeriv_geom (c : <<C>>) (hc : c <<NE>> 0) (n : <<N>>) :
    iteratedDeriv n (fun z : <<C>> => (1 - z / c)<<INV>>) 0
      = (Nat.factorial n : <<C>>) / c ^ n := by
  have hfam : <<ALL>> k : <<N>>,
      iteratedDeriv k (fun z : <<C>> => (1 - z / c)<<INV>>)
        = fun z : <<C>> => (Nat.factorial k : <<C>>) / (c ^ k * (1 - z / c) ^ (k + 1)) := by
    intro k
    induction k with
    | zero =>
      funext z
      simp only [iteratedDeriv_zero, Nat.factorial_zero, Nat.cast_one, pow_zero, one_mul,
        zero_add, pow_one]
      rw [one_div]
    | succ K ih =>
      rw [iteratedDeriv_succ, ih, deriv_geom_family c hc K]
  rw [hfam n]
  simp
