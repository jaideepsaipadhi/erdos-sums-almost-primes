theorem gap_five_evaluation (a : <<R>>) (ha : 0 < a) :
    <<INT>> t in Set.Ioc (0:<<R>>) 1, (Real.exp (-a * t) - 1 + a * t) / t ^ 2
      <<LE>> a * Real.log a + Real.eulerMascheroniConstant * a - a + 1 := by
  have hnorm : (<<INT>> t in Set.Ioc (0:<<R>>) 1, (Real.exp (-a * t) - 1 + a * t) / t ^ 2)
      = <<INT>> t in Set.Ioc (0:<<R>>) 1, (Real.exp (-(a * t)) - 1 + a * t) / t ^ 2 := by
    apply MeasureTheory.setIntegral_congr_fun measurableSet_Ioc
    intro t _
    simp only
    rw [show -a * t = -(a * t) by ring]
  rw [hnorm, gap_substitution a ha, gap_parts_limit a ha, ein_eq a ha]
  have htail := EinTailFun_le a ha
  have hexp : Real.exp (-a) / a * a = Real.exp (-a) := by
    field_simp
  have hkey : a * (Real.eulerMascheroniConstant + Real.log a + EinTailFun a)
      <<LE>> a * (Real.eulerMascheroniConstant + Real.log a) + Real.exp (-a) := by
    have h1 : a * EinTailFun a <<LE>> a * (Real.exp (-a) / a) :=
      mul_le_mul_of_nonneg_left htail ha.le
    have h2 : a * (Real.exp (-a) / a) = Real.exp (-a) := by
      field_simp
    nlinarith [h1, h2]
  have hbdry : a * -((Real.exp (-a) - 1 + a) / a) = -(Real.exp (-a) - 1 + a) := by
    field_simp
  have hsplit : a * (-((Real.exp (-a) - 1 + a) / a)
      + (Real.eulerMascheroniConstant + Real.log a + EinTailFun a))
      = a * -((Real.exp (-a) - 1 + a) / a)
        + a * (Real.eulerMascheroniConstant + Real.log a + EinTailFun a) := by
    ring
  rw [hsplit, hbdry]
  nlinarith [hkey]
