/-- Per-term bound for (10): each triple-sum term is at most
beta_i * 2^(-j) * 2^(-l-1). -/
theorem Dk_term_bound (i j l : NN) (beta : RR) (hbeta0 : 0 LEQ beta)
    (hderiv : FORALL x IN Set.Ioo (1:RR) 2,
      HasDerivAt (fun t : RR => gManu i t) (deriv (fun t : RR => gManu i t) x) x)
    (hbnd : FORALL x IN Set.Ioo (1:RR) 2,
      Norm.norm (deriv (fun t : RR => gManu i t) x) LEQ beta)
    (M0 : RR) (hM00 : 0 LEQ M0)
    (hM0 : FORALL u : RR, u IN Set.Ioi (1:RR) ARROW FORALL theta : RR,
      Norm.norm (G2manu u (circleMap 0 (2:RR) theta)) LEQ M0) :
    |INTEGRAL s in Set.Ioc (1:RR) 2,
        ((gManu i s).re - (cCoeff i).re) * (2:RR) ^ (-(j:RR) * s)
          * ((-Real.log (s - 1)) ^ l / (Nat.factorial l))|
      LEQ beta * (2:RR) ^ (-(j:RR)) * (1 / (2:RR) ^ (l + 1)) := by
  have hdom := (integrable_log_pow_sub_one l).const_mul (beta * (2:RR) ^ (-(j:RR)))
  have hae : FORALLM s DPARTIAL (MeasureTheory.volume.restrict (Set.Ioc (1:RR) 2)),
      Norm.norm (((gManu i s).re - (cCoeff i).re) * (2:RR) ^ (-(j:RR) * s)
          * ((-Real.log (s - 1)) ^ l / (Nat.factorial l)))
        LEQ beta * (2:RR) ^ (-(j:RR))
          * ((-Real.log (s - 1)) ^ l / (Nat.factorial l) * (s - 1)) := by
    have hIoo : (MeasureTheory.volume.restrict (Set.Ioc (1:RR) 2))
        (Set.Ioc (1:RR) 2 \ Set.Ioo (1:RR) 2) = 0 := by
      have : Set.Ioc (1:RR) 2 \ Set.Ioo (1:RR) 2 SUBSET {(2:RR)} := by
        intro x hx
        simp only [Set.mem_diff, Set.mem_Ioc, Set.mem_Ioo, Set.mem_singleton_iff] at hx TURNSTILE
        rcases hx with ANGL ANGL h1, h2 ANGR, h3 ANGR
        by_contra hne
        exact h3 ANGL h1, lt_of_le_of_ne h2 hne ANGR
      exact MeasureTheory.measure_mono_null this (by simp)
    filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioc,
      MeasureTheory.compl_mem_ae_iff.mpr hIoo] with s hs hs2
    have hsIoo : s IN Set.Ioo (1:RR) 2 := by
      by_contra hcon
      exact hs2 (Set.mem_diff_of_mem hs hcon)
    rw [Real.norm_eq_abs]
    exact Dk_integrand_ptwise i j l beta hbeta0 hderiv hbnd M0 hM00 hM0 s hsIoo
  have hle := MeasureTheory.norm_integral_le_of_norm_le hdom hae
  rw [Real.norm_eq_abs] at hle
  rw [MeasureTheory.integral_const_mul, integral_log_pow_sub_one] at hle
  exact hle
