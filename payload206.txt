/-- Each cFac factor is positive, and its argument is at most 1/3, for s > 1. -/
theorem cFac_factor_le_third (s : RR) (hs : 1 < s) (p : {p : NN // p.Prime AND 3 LEQ p}) :
    ((p:NN):RR) ^ (-s) LEQ 1/3 AND 0 < ((p:NN):RR) ^ (-s) := by
  have hp3 : (3:RR) LEQ ((p:NN):RR) := by exact_mod_cast p.2.2
  constructor
  DOT have h1 : ((p:NN):RR) ^ (-s) LEQ ((p:NN):RR) ^ (-(1:RR)) := by
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      linarith
    rw [Real.rpow_neg_one] at h1
    have h2 : (((p:NN):RR))INV LEQ 1/3 := by
      rw [inv_le_comm0 (by linarith) (by norm_num)]
      linarith
    linarith
  DOT positivity

/-- The logs of the cFac factors are summable: -log(1-x) <= 2x for x <= 1/2,
and x = p^(-s) is p-series summable. -/
theorem cFac_log_summable (s : RR) (hs : 1 < s) :
    Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      Real.log ((1 - ((p:NN):RR) ^ (-s))INV)) := by
  have hcomp : Summable (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      2 * (((p:NN):RR) ^ (-s))) := by
    have hbase : Summable (fun n : NN => (n:RR) ^ (-s)) :=
      Real.summable_nat_rpow.mpr (by linarith)
    exact (hbase.subtype _).mul_left _
  apply Summable.of_nonneg_of_le _ _ hcomp
  DOT intro p
    obtain ANGL h3, hp ANGR := cFac_factor_le_third s hs p
    apply Real.log_nonneg
    rw [le_inv_comm0 (by norm_num) (by linarith)]
    linarith
  DOT intro p
    obtain ANGL h3, hp ANGR := cFac_factor_le_third s hs p
    rw [Real.log_inv]
    have hkey := neg_log_one_sub_le (((p:NN):RR) ^ (-s)) (le_of_lt hp) (by linarith)
    linarith
