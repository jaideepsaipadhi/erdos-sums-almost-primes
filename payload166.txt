theorem iteratedDeriv_monomial_zero (m k : NN) (c : CC) :
    iteratedDeriv k (fun z : CC => z ^ m * c) 0
      = if m = k then (Nat.factorial k : CC) * c else 0 := by
  have hbase : iteratedDeriv k (fun z : CC => z ^ m) 0
      = if m = k then (Nat.factorial k : CC) else 0 := by
    rw [iteratedDeriv_pow]
    by_cases h : m = k
    DOT subst h
      simp [Nat.descFactorial_self]
    DOT rw [if_neg h]
      rcases Nat.lt_or_ge m k with hlt | hge
      DOT rw [Nat.descFactorial_eq_zero_iff_lt.mpr hlt]
        simp
      DOT have hne : m - k NEQ 0 := by omega
        rw [zero_pow hne]
        ring
  have hsplit : (fun z : CC => z ^ m * c) = fun z : CC => (z ^ m) * c := rfl
  rw [hsplit, iteratedDeriv_mul_const_field, hbase]
  by_cases h : m = k
  DOT rw [if_pos h, if_pos h]
  DOT rw [if_neg h, if_neg h, zero_mul]
