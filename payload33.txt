theorem EinTailFun_split (a : <<R>>) (ha : 0 < a) :
    EinTailFun a = EinTailFun 1 - <<INT>> v in (1:<<R>>)..a, Real.exp (-v) / v := by
  unfold EinTailFun
  rcases le_total (1:<<R>>) a with h1 | h1
  <<DOT>> have hunion : Set.Ici (1:<<R>>) = Set.Ioc 1 a <<UN>> Set.Ici a := by
      rw [Set.Ioc_union_Ici_eq_Ici h1]
    have hdisj : Disjoint (Set.Ioc (1:<<R>>) a) (Set.Ici a) := by
      rw [Set.disjoint_left]
      intro x hx hx2
      exact absurd (Set.mem_Ici.mp hx2) (not_le.mpr hx.2)
    have hint1 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ioc 1 a) :=
      (exp_div_integrableOn_Ici 1 (by norm_num)).mono
        (fun x hx => Set.mem_Ici.mpr (le_of_lt hx.1)) le_rfl
    have hint2 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ici a) :=
      exp_div_integrableOn_Ici a ha
    have hadd : (<<INT>> v in Set.Ici (1:<<R>>), Real.exp (-v) / v)
        = (<<INT>> v in Set.Ioc (1:<<R>>) a, Real.exp (-v) / v)
          + <<INT>> v in Set.Ici a, Real.exp (-v) / v := by
      rw [hunion, MeasureTheory.integral_union hdisj measurableSet_Ici hint1 hint2]
    rw [intervalIntegral.integral_of_le h1, hadd]
    ring
  <<DOT>> have hunion : Set.Ici a = Set.Ioc a 1 <<UN>> Set.Ici 1 := by
      rw [Set.Ioc_union_Ici_eq_Ici h1]
    have hdisj : Disjoint (Set.Ioc a (1:<<R>>)) (Set.Ici 1) := by
      rw [Set.disjoint_left]
      intro x hx hx2
      exact absurd (Set.mem_Ici.mp hx2) (not_le.mpr hx.2)
    have hint1 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ioc a 1) :=
      (exp_div_integrableOn_Ici a ha).mono
        (fun x hx => Set.mem_Ici.mpr (le_of_lt hx.1)) le_rfl
    have hint2 : MeasureTheory.IntegrableOn
        (fun v : <<R>> => Real.exp (-v) / v) (Set.Ici 1) :=
      exp_div_integrableOn_Ici 1 (by norm_num)
    have hadd : (<<INT>> v in Set.Ici a, Real.exp (-v) / v)
        = (<<INT>> v in Set.Ioc a (1:<<R>>), Real.exp (-v) / v)
          + <<INT>> v in Set.Ici 1, Real.exp (-v) / v := by
      rw [hunion, MeasureTheory.integral_union hdisj measurableSet_Ici hint1 hint2]
    rw [intervalIntegral.integral_of_ge h1, hadd]
    ring
