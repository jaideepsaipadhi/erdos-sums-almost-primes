/-- The pointwise bound feeding hasDerivAt_tsum_of_isPreconnected:
|z(z-1) log p / ((p^s-1)(p^s-z))| <= 27 log p p^(-2s) for p >= 3, s >= 1, |z| <= 2.
Crude route: |z(z-1)| <= 6, p^s - 1 >= (2/3)p^s, |p^s - z| >= (1/3)p^s. -/
theorem deriv_prime_term_bound (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s)
    (z : RR) (hz : |z| LEQ 2) :
    |(-(z * (z - 1) * Real.log (p:RR))
        / ((((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z)))|
      LEQ 27 * Real.log (p:RR) * ((p:RR)) ^ (-2*s) := by
  have hp3 : (3:RR) LEQ (p:RR) := by exact_mod_cast hp
  have hp0 : (0:RR) < (p:RR) := by linarith
  have hps : (3:RR) LEQ ((p:RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ ((p:RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hps0 : (0:RR) < ((p:RR)) ^ s := by linarith
  have hzabs := abs_le.mp hz
  have hlogpos : (0:RR) LEQ Real.log (p:RR) := Real.log_nonneg (by linarith)
  have hnum : |z * (z - 1) * Real.log (p:RR)| LEQ 6 * Real.log (p:RR) := by
    rw [abs_mul, abs_mul]
    rw [abs_of_nonneg hlogpos]
    have h1 : |z| LEQ 2 := hz
    have h2 : |z - 1| LEQ 3 := by
      rw [abs_le]
      constructor <;> linarith [hzabs.1, hzabs.2]
    nlinarith [abs_nonneg z, abs_nonneg (z - 1), hlogpos]
  have hd1 : (2/3) * ((p:RR)) ^ s LEQ ((p:RR)) ^ s - 1 := by linarith
  have hd2 : (1/3) * ((p:RR)) ^ s LEQ ((p:RR)) ^ s - z := by
    linarith [hzabs.2]
  have hden : (2/9) * (((p:RR)) ^ s) ^ 2
      LEQ (((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z) := by
    nlinarith [hd1, hd2, hps0]
  have hdenpos : (0:RR) < (((p:RR)) ^ s - 1) * (((p:RR)) ^ s - z) := by nlinarith [hps0]
  have hsq : (((p:RR)) ^ s) ^ 2 = ((p:RR)) ^ (2*s) := by
    rw [LARR Real.rpow_natCast (((p:RR)) ^ s) 2, LARR Real.rpow_mul (le_of_lt hp0)]
    norm_num
    ring_nf
  rw [abs_div, abs_neg]
  rw [div_le_iff0 (by rwa [abs_of_pos hdenpos])]
  rw [abs_of_pos hdenpos]
  have hpow : ((p:RR)) ^ (-2*s) * ((p:RR)) ^ (2*s) = 1 := by
    rw [LARR Real.rpow_add hp0]
    norm_num
  nlinarith [hnum, hden, hsq, hpow, hlogpos, hps0]
