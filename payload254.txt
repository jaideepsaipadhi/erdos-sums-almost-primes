/-- Expanding (k-i)^2 under the sum: the quadratic weight splits into the
three partial sums of section 2.3. -/
theorem sum_two_pow_sq_expand (k : NN) :
    SUMF i IN Finset.range (k + 1),
        (2:RR)^i * (cCoeff i).re * ((k:RR) - (i:RR))^2
      = (k:RR)^2 * (dConst - Wk k) - 2 * (k:RR) * Uk k + Vk k := by
  have hexp : FORALL i IN Finset.range (k + 1),
      (2:RR)^i * (cCoeff i).re * ((k:RR) - (i:RR))^2
        = (k:RR)^2 * ((2:RR)^i * (cCoeff i).re)
          - 2 * (k:RR) * ((i:RR) * (2:RR)^i * (cCoeff i).re)
          + ((i:RR)^2 * (2:RR)^i * (cCoeff i).re) := by
    intro i _
    ring
  rw [Finset.sum_congr rfl hexp]
  rw [Finset.sum_add_distrib, Finset.sum_sub_distrib]
  rw [LARR Finset.mul_sum, LARR Finset.mul_sum]
  rw [cCoeff_two_pow_partial_sum]
  unfold Uk Vk
  ring
