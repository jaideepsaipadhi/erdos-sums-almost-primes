/-- The s-free dominator, indexed over p >= 5 to match G2tailC.
Rebuilt from the N-indexed p-series rather than restricted from the p >= 3
version: those are different subtypes of N, neither contained in the other as
Lean sees it, and the file's idiom (line ~1126) is to restrict from N. -/
theorem deriv_bound_uniform_summable_five :
    Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      27 * Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-(2:RR))) := by
  have hbase : Summable (fun n : NN => (n:RR) ^ (-(3:RR)/2) := by
    rw [Real.summable_nat_rpow]
    norm_num
  have hsub : Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      (((p:NN):RR)) ^ (-(3:RR)/2)) := hbase.subtype _
  have h27 := hsub.mul_left 27
  apply Summable.of_nonneg_of_le _ _ h27
  DOT intro p
    have hp5 : (5:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have : (0:RR) LEQ Real.log ((p:NN):RR) := Real.log_nonneg (by linarith)
    positivity
  DOT intro p
    have hp5 : (5:RR) LEQ (((p:NN):RR)) := by exact_mod_cast p.2.2
    have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
    have hlog := log_le_sqrt (((p:NN):RR)) (by linarith)
    have hpos : (0:RR) < (((p:NN):RR)) ^ (-(2:RR)) := Real.rpow_pos_of_pos hp0 _
    have hstep : Real.log ((p:NN):RR) * (((p:NN):RR)) ^ (-(2:RR))
        LEQ ((((p:NN):RR)) ^ ((1:RR)/2)) * (((p:NN):RR)) ^ (-(2:RR)) :=
      mul_le_mul_of_nonneg_right hlog (le_of_lt hpos)
    have hcomb : ((((p:NN):RR)) ^ ((1:RR)/2)) * (((p:NN):RR)) ^ (-(2:RR))
        = (((p:NN):RR)) ^ (-(3:RR)/2) := by
      rw [LARR Real.rpow_add hp0]
      norm_num
    rw [hcomb] at hstep
    nlinarith [hstep]
