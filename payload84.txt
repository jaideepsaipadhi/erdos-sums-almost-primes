/-- Leibniz applied twice to Fbig: the k-th iterated derivative at 0 as a
double sum with binomial coefficients. -/
theorem iteratedDeriv_Fbig (k : <<N>>) (s : <<R>>) (hs : 1 < s) :
    iteratedDeriv k (fun z => Fbig s z) 0
      = <<SUM>> n <<IN>> Finset.range (k + 1),
          (Nat.choose k n : <<C>>) *
            (<<SUM>> i <<IN>> Finset.range (n + 1),
              (Nat.choose n i : <<C>>)
                * iteratedDeriv i (fun w : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-w)) 0
                * iteratedDeriv (n - i) (fun w : <<C>> => (1 - w / ((2:<<C>>) ^ (s:<<C>>)))<<INV>>) 0)
            * iteratedDeriv (k - n) (fun w : <<C>> => G2C s w) 0 := by
  have hs1 : (1:<<R>>) <<LE>> s := le_of_lt hs
  have hzne : 1 - (0:<<C>>) / ((2:<<C>>) ^ (s:<<C>>)) <<NE>> 0 := by
    simp
  have hA : <<ALL>> j : <<N>>, ContDiffAt <<C>> j
      (fun w : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-w)) 0 :=
    fun j => contDiffAt_factorA s hs j 0
  have hB : <<ALL>> j : <<N>>, ContDiffAt <<C>> j
      (fun w : <<C>> => (1 - w / ((2:<<C>>) ^ (s:<<C>>)))<<INV>>) 0 :=
    fun j => contDiffAt_factorB s j 0 hzne
  have hC : <<ALL>> j : <<N>>, ContDiffAt <<C>> j (fun w : <<C>> => G2C s w) 0 :=
    fun j => contDiffAt_factorC s hs1 j
  have hAB : <<ALL>> j : <<N>>, ContDiffAt <<C>> j
      ((fun w : <<C>> => ((s - 1 : <<R>>) : <<C>>) ^ (-w))
        * (fun w : <<C>> => (1 - w / ((2:<<C>>) ^ (s:<<C>>)))<<INV>>)) 0 :=
    fun j => (hA j).mul (hB j)
  rw [Fbig_eq_prod s]
  rw [iteratedDeriv_mul (hAB k) (hC k)]
  apply Finset.sum_congr rfl
  intro n hn
  rw [iteratedDeriv_mul (hA n) (hB n)]
  rw [Finset.sum_mul, Finset.mul_sum, Finset.sum_mul]
  apply Finset.sum_congr rfl
  intro i _
  ring
