/-- All iterated derivatives of the n = 0 term vanish. -/
theorem iteratedDeriv_term_zero (s : RR) (j : NN) (w : CC) :
    iteratedDeriv j (fun z : CC => zOmegaHom s z 0) w = 0 := by
  have hz : (fun z : CC => zOmegaHom s z 0) = fun _ : CC => (0:CC) := by
    funext z
    exact map_zero _
  rw [hz]
  rcases Nat.eq_zero_or_pos j with hj | hj
  DOT subst hj
    simp
  DOT rw [iteratedDeriv_const_of_ne (by omega)]
