theorem EinTailFun_tendsto_zero :
    Filter.Tendsto (fun n : <<N>> => EinTailFun (n:<<R>>)) Filter.atTop (nhds 0) := by
  have hmaj : Filter.Tendsto (fun n : <<N>> => Real.exp (-(n:<<R>>)))
      Filter.atTop (nhds 0) := by
    have h := Real.tendsto_exp_neg_atTop_nhds_zero
    exact h.comp tendsto_natCast_atTop_atTop
  apply squeeze_zero_norm' _ hmaj
  filter_upwards [Filter.eventually_ge_atTop 1] with n hn
  have hn1 : (1:<<R>>) <<LE>> (n:<<R>>) := by exact_mod_cast hn
  have hn0 : (0:<<R>>) < (n:<<R>>) := by linarith
  unfold EinTailFun
  have hIci : (<<INT>> v in Set.Ici (n:<<R>>), Real.exp (-v) / v)
      = <<INT>> v in Set.Ioi (n:<<R>>), Real.exp (-v) / v := by
    apply MeasureTheory.setIntegral_congr_set
    exact (MeasureTheory.Ioi_ae_eq_Ici).symm
  rw [hIci]
  have hle : |<<INT>> v in Set.Ioi (n:<<R>>), Real.exp (-v) / v|
      <<LE>> <<INT>> v in Set.Ioi (n:<<R>>), Real.exp (-v) := by
    apply MeasureTheory.abs_integral_le_integral_abs.trans
    apply MeasureTheory.setIntegral_mono_on
    <<DOT>> exact ((exp_div_integrableOn_Ici (n:<<R>>) hn0).congr_set_ae
        (MeasureTheory.Ioi_ae_eq_Ici).symm).abs
    <<DOT>> have hbase : MeasureTheory.IntegrableOn
          (fun u : <<R>> => Real.exp (-u) * u ^ ((1:<<R>>) - 1)) (Set.Ioi 0) :=
        Real.GammaIntegral_convergent (s := (1:<<R>>)) (by norm_num)
      have hb2 : MeasureTheory.IntegrableOn (fun u : <<R>> => Real.exp (-u)) (Set.Ioi 0) := by
        apply hbase.congr_fun ?_ measurableSet_Ioi
        intro u _
        simp only
        rw [show (1:<<R>>) - 1 = 0 by ring, Real.rpow_zero, mul_one]
      exact hb2.mono (fun x hx => Set.mem_Ioi.mpr (by linarith [Set.mem_Ioi.mp hx])) le_rfl
    <<DOT>> exact measurableSet_Ioi
    <<DOT>> intro v hv
      have hvn : (n:<<R>>) < v := hv
      have hv0 : (0:<<R>>) < v := by linarith
      have hE : (0:<<R>>) < Real.exp (-v) := Real.exp_pos _
      rw [abs_of_nonneg (by positivity : (0:<<R>>) <<LE>> Real.exp (-v) / v)]
      rw [div_le_iff<<S0>> hv0]
      nlinarith [hE, hv0, hn1, hvn]
  rw [integral_Ioi_exp_neg (n:<<R>>) (by linarith)] at hle
  rw [Real.norm_eq_abs, Real.norm_eq_abs, abs_of_pos (Real.exp_pos _)]
  exact hle
