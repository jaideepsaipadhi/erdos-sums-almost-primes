/-- Omega(p^n) = n for p prime. -/
lemma bigOmega_prime_pow {p : NN} (hp : p.Prime) (n : NN) : bigOmega (p ^ n) = n := by
  induction n with
  | zero =>
    simp only [pow_zero]
    unfold bigOmega
    simp
  | succ N ih =>
    have hpn : p ^ N NEQ 0 := pow_ne_zero N hp.pos.ne'
    have hp0 : p NEQ 0 := hp.pos.ne'
    rw [pow_succ, bigOmega_mul (p ^ N) p hpn hp0, ih]
    have hO : bigOmega p = 1 := by
      unfold bigOmega
      rw [hp.factorization]
      simp
    rw [hO]
