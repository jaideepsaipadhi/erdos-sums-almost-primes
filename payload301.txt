/-- The contour integrand converges pointwise in theta as s -> 1+.
The s-free factors are constant in s; only G2manu moves, and
G2manu_tendsto_G2at1 gives its limit at each point of the contour. -/
theorem contour_integrand_tendsto (i : NN) (theta : RR) :
    Filter.Tendsto (fun s : RR =>
        deriv (circleMap 0 (2:RR)) theta
          * ((1 / circleMap 0 (2:RR) theta) ^ i
            * ((circleMap 0 (2:RR) theta)INV * G2manu s (circleMap 0 (2:RR) theta))))
      (nhdsWithin 1 (Set.Ioi (1:RR)))
      (nhds (deriv (circleMap 0 (2:RR)) theta
        * ((1 / circleMap 0 (2:RR) theta) ^ i
          * ((circleMap 0 (2:RR) theta)INV * G2at1 (circleMap 0 (2:RR) theta))))) := by
  have hnorm : Norm.norm (circleMap 0 (2:RR) theta) LEQ 2 := by
    simp [circleMap]
  have hG2 := G2manu_tendsto_G2at1 (circleMap 0 (2:RR) theta) hnorm
  exact (hG2.const_mul _).const_mul _ |>.const_mul _
