/-- D4: dConst's indicator product over all primes equals the p >= 3 subtype
product, in the spelling G2C and cFac use. -/
theorem dConst_prod_eq_subtype :
    (TPRODN p : Nat.Primes,
        (if 2 < (p : NN) then (1 - 2 / ((p : NN) : RR))INV * (1 - 1 / ((p : NN) : RR)) ^ 2 else 1))
      = TPRODN p : {p : NN // p.Prime AND 3 LEQ p},
          (1 - 2 / ((p : NN) : RR))INV * (1 - 1 / ((p : NN) : RR)) ^ 2 := by
  rw [dConst_if_eq_indicator]
  rw [LARR tprod_subtype {p : Nat.Primes | (3:NN) LEQ (p : NN)}
    (fun p : Nat.Primes => (1 - 2 / ((p : NN) : RR))INV * (1 - 1 / ((p : NN) : RR)) ^ 2)]
  exact (primesEquivSubtypeGE 3).tprod_eq
    (fun p : {p : NN // p.Prime AND 3 LEQ p} =>
      (1 - 2 / ((p : NN) : RR))INV * (1 - 1 / ((p : NN) : RR)) ^ 2)
