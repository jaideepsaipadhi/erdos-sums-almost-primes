/-- A1 step 4d: the inner tsum is the closed geometric form. -/
theorem zOmegaNorm_inner_tsum (s : RR) (hs : 1 < s) (z : CC)
    (hz : Norm.norm z < (2:RR) ^ s) {p : NN} (hp : p.Prime) :
    (TSUM n : NN, zOmegaNormHom s z (p ^ n))
      = (1 - Norm.norm z * ((p:RR) ^ (-s)))INV := by
  have hratio : Norm.norm z * ((p:RR) ^ (-s)) < 1 := by
    have hp2 : (2:RR) LEQ (p:RR) := by exact_mod_cast hp.two_le
    have h2s : (0:RR) < (2:RR) ^ s := Real.rpow_pos_of_pos (by norm_num) s
    have hps : (p:RR) ^ (-s) LEQ (2:RR) ^ (-s) := by
      rw [Real.rpow_neg (by positivity : (0:RR) LEQ (p:RR)),
        Real.rpow_neg (by norm_num : (0:RR) LEQ (2:RR))]
      gcongr
    have hznn : (0:RR) LEQ Norm.norm z := norm_nonneg z
    have hinvpos : (0:RR) < ((2:RR) ^ s)INV := inv_pos.mpr h2s
    calc Norm.norm z * ((p:RR) ^ (-s)) LEQ Norm.norm z * ((2:RR) ^ (-s)) :=
          mul_le_mul_of_nonneg_left hps hznn
      _ = Norm.norm z * ((2:RR) ^ s)INV := by
          rw [Real.rpow_neg (by norm_num : (0:RR) LEQ (2:RR))]
      _ < 1 := by
          have hkey := mul_lt_mul_of_pos_right hz hinvpos
          rwa [mul_inv_cancelZ (ne_of_gt h2s)] at hkey
  have hnn : (0:RR) LEQ Norm.norm z * ((p:RR) ^ (-s)) := by positivity
  rw [tsum_congr (fun n => zOmegaNorm_prime_pow s z hp n)]
  exact tsum_geometric_of_lt_one hnn hratio
