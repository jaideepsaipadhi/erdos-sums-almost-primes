/-- The s-derivative of G2manu, as a function of the contour parameter theta,
is continuous.  This is what makes the F' measurability obligation of
Lemma 3.8 discharge: lemma_3_9_complex identifies the deriv with an explicit
expression, and every factor of that expression is continuous in theta. -/
theorem G2manu_deriv_continuous_theta (s : RR) (hsmem : s IN Set.Ioo (1:RR) 2)
    (f : RR ARROW RR)
    (hf : FORALL x IN Set.Ioo (1:RR) 2,
      HasDerivAt (fun t : RR => Real.log (zetaTilde t)) (f x) x) :
    Continuous (fun theta : RR =>
      deriv (fun u : RR => G2manu u (circleMap 0 (2:RR) theta)) s) := by
  have hs1 : 1 LEQ s := le_of_lt hsmem.1
  have hkap : 0 < kappaFac s := kappaFac_pos s hsmem.1 (cFac_pos s hsmem.1)
  have hcm : Continuous (circleMap 0 (2:RR)) := continuous_circleMap 0 _
  have hnorm : FORALL theta : RR, Norm.norm (circleMap 0 (2:RR) theta) LEQ 2 := by
    intro theta
    simp [circleMap]
  have hval : FORALL theta : RR,
      deriv (fun u : RR => G2manu u (circleMap 0 (2:RR) theta)) s
        = G2manu s (circleMap 0 (2:RR) theta) *
          (circleMap 0 (2:RR) theta * (((f s + Real.log 2 / ((2:RR) ^ s - 1) : RR)) : CC)
            + ((-(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
                  * ((Real.log ((3:NN):RR) : RR) : CC))
                / (((((((3:NN):RR)) ^ s : RR)) : CC) - 1)
                  / (((((((3:NN):RR)) ^ s : RR)) : CC) - circleMap 0 (2:RR) theta))
              + TSUMN p : {p : NN // p.Prime AND 5 LEQ p},
                  (-(circleMap 0 (2:RR) theta * (circleMap 0 (2:RR) theta - 1)
                      * ((Real.log ((p:NN):RR) : RR) : CC))
                    / (((((((p:NN):RR)) ^ s : RR)) : CC) - 1)
                      / (((((((p:NN):RR)) ^ s : RR)) : CC) - circleMap 0 (2:RR) theta)))) := by
    intro theta
    exact (lemma_3_9_complex (circleMap 0 (2:RR) theta) (hnorm theta) s hsmem f
      (hf s hsmem)).deriv
  simp only [hval]
  have hG2 : Continuous (fun theta : RR => G2manu s (circleMap 0 (2:RR) theta)) := by
    have hsub : Metric.closedBall (0:CC) (2:RR) SUBSET Metric.closedBall (0:CC) ((12:RR)/5) := by
      intro w hw
      rw [Metric.mem_closedBall, dist_zero_right] at hw TURNSTILE
      linarith
    have hmem : FORALL theta : RR, circleMap 0 (2:RR) theta IN
        Metric.closedBall (0:CC) (2:RR) := by
      intro theta
      rw [Metric.mem_closedBall, dist_zero_right]
      simp [circleMap]
    exact (((G2manu_differentiableOn_wide s hs1 hkap).mono hsub).continuousOn).comp_continuous
      hcm hmem
  apply hG2.mul
  apply Continuous.add
  DOT exact hcm.mul continuous_const
  DOT exact (clog_term_continuous_theta 3 (le_refl 3) s hs1).add
      (clog_series_continuous_theta s hs1)
