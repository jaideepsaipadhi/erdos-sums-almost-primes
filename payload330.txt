/-- Basepoint summability at s = 3/2 for the r=2 series. -/
theorem deriv2_series_summable_at_basepoint (z : CC) (hz : Norm.norm z LEQ 2) :
    Summable (fun p : {p : NN // p.Prime AND 5 LEQ p} =>
      -((z - 1) * ((Real.log ((p:NN):RR) : RR) : CC))
        / ((((((p:NN):RR)) ^ (3/2:RR) : RR) : CC) - 1)
        / ((((((p:NN):RR)) ^ (3/2:RR) : RR) : CC) - z)) := by
  apply Summable.of_norm
  apply Summable.of_nonneg_of_le
    (fun p : {p : NN // p.Prime AND 5 LEQ p} => norm_nonneg _)
    _ (deriv_bound_uniform_summable_five)
  intro p
  exact deriv_clog_term_bound_uniform (p:NN) (by omega) (3/2) (by norm_num) z hz
