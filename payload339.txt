/-- Per-term continuity in theta for the r=2 series, on the contour |z| = 2. -/
theorem clog2_term_continuous_theta (p : NN) (hp : 3 LEQ p) (s : RR) (hs : 1 LEQ s) :
    Continuous (fun theta : RR =>
      circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
          * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2 * (((((p:NN):RR)) ^ s : RR) : CC)
          * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
            + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))
        / (((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
          * (((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta) ^ 2)))) := by
  have hp3 : (3:RR) LEQ (((p:NN):RR)) := by exact_mod_cast hp
  have hp0 : (0:RR) < (((p:NN):RR)) := by linarith
  have hps : (3:RR) LEQ (((p:NN):RR)) ^ s := by
    calc (3:RR) = (3:RR) ^ (1:RR) := by norm_num
      _ LEQ (3:RR) ^ s := Real.rpow_le_rpow_of_exponent_le (by norm_num) hs
      _ LEQ (((p:NN):RR)) ^ s := Real.rpow_le_rpow (by norm_num) hp3 (by linarith)
  have hcm : Continuous (circleMap 0 (2:RR)) := continuous_circleMap 0 _
  have hnorm : FORALL theta : RR, Norm.norm (circleMap 0 (2:RR) theta) = 2 := by
    intro theta
    simp [circleMap]
  have hpsC : Norm.norm ((((((p:NN):RR)) ^ s : RR) : CC)) = (((p:NN):RR)) ^ s := by
    rw [Complex.norm_real, Real.norm_eq_abs, abs_of_pos (by linarith)]
  have hd1 : (((((((p:NN):RR)) ^ s : RR) : CC)) - 1) NEQ 0 := by
    intro hcon
    rw [sub_eq_zero] at hcon
    have : ((((p:NN):RR)) ^ s : RR) = 1 := by exact_mod_cast hcon
    linarith
  have hd2 : FORALL theta : RR,
      (((((((p:NN):RR)) ^ s : RR) : CC)) - circleMap 0 (2:RR) theta) NEQ 0 := by
    intro theta hcon
    rw [sub_eq_zero] at hcon
    have hn := hnorm theta
    rw [LARR hcon, hpsC] at hn
    linarith
  have hnum : Continuous (fun theta : RR =>
      circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
        * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2 * (((((p:NN):RR)) ^ s : RR) : CC)
        * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
          + ((((((p:NN):RR)) ^ s : RR) : CC) - 1)))) :=
    hcm.mul ((((hcm.sub continuous_const).mul continuous_const).mul continuous_const).mul
      ((continuous_const.sub hcm).add continuous_const))
  have hden : Continuous (fun theta : RR =>
      (((((((p:NN):RR)) ^ s : RR) : CC)) - 1) ^ 2
        * (((((((p:NN):RR)) ^ s : RR) : CC)) - circleMap 0 (2:RR) theta) ^ 2) :=
    continuous_const.mul ((continuous_const.sub hcm).pow 2)
  have hdne : FORALL theta : RR,
      (((((((p:NN):RR)) ^ s : RR) : CC)) - 1) ^ 2
        * (((((((p:NN):RR)) ^ s : RR) : CC)) - circleMap 0 (2:RR) theta) ^ 2 NEQ 0 := by
    intro theta
    exact mul_ne_zero (pow_ne_zero 2 hd1) (pow_ne_zero 2 (hd2 theta))
  have hfun : FORALL theta : RR,
      circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
          * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2 * (((((p:NN):RR)) ^ s : RR) : CC)
          * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
            + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))
        / (((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
          * (((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta) ^ 2)))
      = (circleMap 0 (2:RR) theta * ((circleMap 0 (2:RR) theta - 1)
          * ((Real.log ((p:NN):RR) : RR) : CC) ^ 2 * (((((p:NN):RR)) ^ s : RR) : CC)
          * ((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta
            + ((((((p:NN):RR)) ^ s : RR) : CC) - 1))))
        / ((((((((p:NN):RR)) ^ s : RR) : CC) - 1) ^ 2
          * (((((((p:NN):RR)) ^ s : RR) : CC) - circleMap 0 (2:RR) theta) ^ 2))) := by
    intro theta
    ring
  simp only [hfun]
  exact hnum.div hden hdne
